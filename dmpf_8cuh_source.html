74template <int in_bits, int max_points, int bucket_bits, typename Group, typename Prg, typename Prp, typename In = uint,
75 int kappa = 3, int ch_lambda = 80>
76 requires((std::is_unsigned_v<In> || std::is_same_v<In, __uint128_t>) && in_bits <= sizeof(In) * 8 &&
77 Groupable<Group> && Prgable<Prg, 2> && Permutable<Prp>)
78class Dmpf {
79public:
80 static_assert(max_points >= 30, "max_points must be >= 30 (Remark 1 of the paper)");
81 static constexpr int m = cuckoo_hash::ChBucket(max_points, ch_lambda);
82 static constexpr __uint128_t n = __uint128_t(1) << in_bits;
83 // b_size = ceil(n * kappa / m), use __uint128_t to avoid overflow.
84 static constexpr int b_size = static_cast<int>((static_cast<__uint128_t>(n) * kappa + m - 1) / m);
85 static_assert(b_size <= (1 << bucket_bits));
86
87 using InnerDpf = Dpf<bucket_bits, Group, Prg, uint>;

89 Prg prg;
90 Prp prp;

99 int4 sigma;
100 int m_rt;

102 typename InnerDpf::Key bks[m];

128 int Gen(Key &k0, Key &k1, int4 &sigma, cuda::std::span<cuda::std::array<int4, 2>, m> s0s,
129 cuda::std::span<const In> as, cuda::std::span<const int4> b_bufs, int t, int ch_retry = 1000,
130 bool retry = true) {
131 assert(t <= max_points);
132 assert(t >= 30);
133
134 // Compute runtime bucket count and bucket size.
135 int m_ = cuckoo_hash::ChBucket(t, ch_lambda);
136 assert(m_ <= m);
137 __uint128_t n128 = static_cast<__uint128_t>(n);
138 int b_rt = static_cast<int>((n128 * kappa + m_ - 1) / m_);
139 assert(b_rt <= (1 << bucket_bits));
140
141 k0.m_rt = m_;
142 k1.m_rt = m_;
143 k0.b_size_rt = b_rt;
144 k1.b_size_rt = b_rt;
145
146 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
147 cuckoo_hash::Compact<Prp, In, kappa> compact{prp};
148
149 for (int attempt = 0; attempt < 128; ++attempt) {
150 k0.sigma = sigma;
151 k1.sigma = sigma;
152
153 // Run Cuckoo hashing.
154 std::vector<std::pair<int, int>> table(m_, {-1, -1});
155 int ret = compact.Run(std::span<const In>(as.data(), t), m_, sigma, n, b_rt, ch_retry,
156 std::span<std::pair<int, int>>(table));
157
158 // Generate inner DPF keys for each bucket.
159 if (ret == 0) {
160 InnerDpf inner_dpf{prg};
161
162 for (int i = 0; i < m; ++i) {
163 uint a_prime = 0;
164 int4 b_buf_prime = {0, 0, 0, 0};
165
166 if (i < m_ && table[i].first != -1) {
167 int j = table[i].first; // index into as
168 int k = table[i].second; // hash function that placed it
169 auto [bucket, index] = prp_hash.Locate(sigma, as[j], k, n, b_rt);
170 a_prime = static_cast<uint>(index);
171 assert(a_prime < (1u << bucket_bits));
172 b_buf_prime = b_bufs[j];
173 }
174
175 // The inner Gen shares the correction words between the parties and differs in the seed.
176 typename InnerDpf::Key &bk = k0.bks[i];
177 inner_dpf.Gen(bk, k1.bks[i], s0s[i], static_cast<uint>(a_prime), b_buf_prime);
178 }
179
180 return 0;
181 }
182
183 if (!retry) return 1;
184
185 // Deterministically perturb the PRP seed and the initial seeds, then retry.
186 sigma.x += 1;
187 for (int i = 0; i < m; ++i) {
188 s0s[i][0].x += 1;
189 s0s[i][1].y += 1;
190 }
191 }
192
193 return 1;
194 }

206 void BatchEval(bool b, const Key &key, cuda::std::span<const In> xs, cuda::std::span<int4> ys) {
207 size_t eta = xs.size();
208 assert(ys.size() >= eta);
209
210 int m_ = key.m_rt;
211 int b_rt = key.b_size_rt;
212
213 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
214
215 // Build per-bucket input lists.
216 // inputs[i] = vector of (within_bucket_index, original_input_index).
217 std::vector<std::vector<std::pair<uint, size_t>>> inputs(m);
218
219 for (size_t omega = 0; omega < eta; ++omega) {
220 // store for easier deduplication for this omega
221 std::array<std::pair<int, uint>, kappa> seen{};
222 int num_seen = 0;
223
224 for (int k = 0; k < kappa; ++k) {
225 auto [bucket, index] = prp_hash.Locate(key.sigma, xs[omega], k, n, b_rt);
226 if (bucket >= m) continue;
227
228 uint j = static_cast<uint>(index);
229 assert(j < (1u << bucket_bits));
230
231 // Deduplicate among this omega's located records only. Permu is a permutation over
232 // [0, n * kappa), so distinct k never collide and the check is only a safeguard.
233 bool dup = false;
234 for (int s = 0; s < num_seen; ++s) {
235 if (seen[s].first == bucket && seen[s].second == j) {
236 dup = true;
237 break;
238 }
239 }
240 if (!dup) {
241 inputs[bucket].push_back({j, omega});
242 seen[num_seen++] = {bucket, j};
243 }
244 }
245 }
246
247 // Initialize outputs.
248 for (size_t i = 0; i < eta; ++i) {
249 ys[i] = {0, 0, 0, 0};
250 }
251
252 // Evaluate per bucket.
253 InnerDpf inner_dpf{prg};
254 for (int i = 0; i < m; ++i) {
255 for (auto &[j, omega] : inputs[i]) {
256 int4 y = inner_dpf.Eval(b, key.bks[i], j);
257
258 // Accumulate output.
259 ys[omega] = (Group::From(ys[omega]) + Group::From(y)).Into();
260 }
261 }
262 }

271 void EvalAll(bool b, const Key &key, cuda::std::span<int4> ys) {
272 assert(ys.size() == n);
273
274 int bucket_count = key.m_rt;
275 int b_rt = key.b_size_rt;
276
277 // build per-bucket input lists.
278 // inputs[i] = vector of (within_bucket_index, original_input_index).
279 std::vector<std::vector<std::pair<uint, size_t>>> inputs(bucket_count);
280 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
281
282 for (size_t omega = 0; omega < n; ++omega) {
283 In x = static_cast<In>(omega);
284
285 // store for easier deduplication for this omega
286 std::array<std::pair<int, uint>, kappa> seen{};
287 int num_seen = 0;
288
289 for (int k = 0; k < kappa; ++k) {
290 auto [bucket, index] = prp_hash.Locate(key.sigma, x, k, n, b_rt);
291 if (bucket >= bucket_count) continue;
292
293 uint j = static_cast<uint>(index);
294 assert(j < (1u << bucket_bits));
295
296 // Deduplicate among this omega's located records (linear scan, fine for small kappa).
297 bool dup = false;
298
299 for (int s = 0; s < num_seen; ++s) {
300 if (seen[s].first == bucket && seen[s].second == j) {
301 dup = true;
302 break;
303 }
304 }
305
306 if (!dup) {
307 inputs[bucket].push_back({j, omega});
308 seen[num_seen++] = {bucket, j};
309 }
310 }
311 }
312
313 // initialize outputs.
314 for (size_t i = 0; i < n; ++i) {
315 ys[i] = {0, 0, 0, 0};
316 }
317
318 // evaluate per bucket and merge.
319 InnerDpf inner_dpf{prg};
320 std::vector<int4> bucket_ys(1ULL << bucket_bits, {0, 0, 0, 0});
321 for (int i = 0; i < bucket_count; ++i) {
322 inner_dpf.EvalAll(b, key.bks[i], cuda::std::span<int4>(bucket_ys.data(), bucket_ys.size()));
323
324 // merge
325 for (auto &[j, omega] : inputs[i]) {
326 ys[omega] = (Group::From(ys[omega]) + Group::From(bucket_ys[j])).Into();
327 }
328 }
329 }
