79template <int in_bits, int max_points, int bucket_bits, typename Group, typename Prg, typename Prp, typename In = uint,
80 int kappa = 3, int ch_lambda = 80, int par_depth = -1>
81 requires((std::is_unsigned_v<In> || std::is_same_v<In, __uint128_t>) && in_bits <= sizeof(In) * 8 &&
82 Groupable<Group> && Prgable<Prg, 2> && Permutable<Prp>)
83class Dmpf {
84public:
85 static_assert(max_points >= 30, "max_points must be >= 30 (Remark 1 of the paper)");
86 static_assert(bucket_bits >= 1 && bucket_bits <= 30, "bucket_bits must be in [1, 30]");
87 static_assert(kappa > 0, "kappa must be positive");
88 static constexpr int m = cuckoo_hash::ChBucket(max_points, ch_lambda);
89 static_assert(m > 0, "bucket count must be positive");
90 static constexpr __uint128_t n = __uint128_t(1) << in_bits;
91 // Bound capacity before forming n * kappa, which may overflow even __uint128_t.
92 static_assert(n <= ((__uint128_t{1} << bucket_bits) * m) / kappa,
93 "bucket_bits is too small for the bucket size");
94 // b_size = ceil(n * kappa / m). The capacity bound keeps the numerator in range.
95 static constexpr __uint128_t kBucketSize = (n * kappa + m - 1) / m;
96 static constexpr int b_size = static_cast<int>(kBucketSize);
97
98 using InnerDpf = Dpf<bucket_bits, Group, Prg, uint, par_depth>;

100 Prg prg;
101 Prp prp;

110 int4 sigma;
111 int m_rt;
112 int b_size_rt;
113 typename InnerDpf::Key bks[m];

139 int Gen(Key &k0, Key &k1, int4 &sigma, cuda::std::span<cuda::std::array<int4, 2>, m> s0s,
140 cuda::std::span<const In> as, cuda::std::span<const int4> b_bufs, int t, int ch_retry = 1000,
141 bool retry = true) {
142 assert(t <= max_points);
143 assert(t >= 30);
144
145 // Compute runtime bucket count and bucket size.
146 int m_ = cuckoo_hash::ChBucket(t, ch_lambda);
147 assert(m_ <= m);
148 __uint128_t n128 = static_cast<__uint128_t>(n);
149 __uint128_t bucket_size = (n128 * kappa + m_ - 1) / m_;
150 assert(bucket_size <= (uint64_t{1} << bucket_bits));
151 int b_rt = static_cast<int>(bucket_size);
152
153 k0.m_rt = m_;
154 k1.m_rt = m_;
155 k0.b_size_rt = b_rt;
156 k1.b_size_rt = b_rt;
157
158 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
159 cuckoo_hash::Compact<Prp, In, kappa> compact{prp};
160
161 for (int attempt = 0; attempt < 128; ++attempt) {
162 k0.sigma = sigma;
163 k1.sigma = sigma;
164
165 // Run Cuckoo hashing.
166 std::vector<std::pair<int, int>> table(m_, {-1, -1});
167 int ret = compact.Run(std::span<const In>(as.data(), t), m_, sigma, n, b_rt, ch_retry,
168 std::span<std::pair<int, int>>(table));
169
170 // Generate inner DPF keys for each bucket.
171 if (ret == 0) {
172 InnerDpf inner_dpf{prg};
173
174 for (int i = 0; i < m; ++i) {
175 uint a_prime = 0;
176 int4 b_buf_prime = {0, 0, 0, 0};
177
178 if (i < m_ && table[i].first != -1) {
179 int j = table[i].first; // index into as
180 int k = table[i].second; // hash function that placed it
181 auto [bucket, index] = prp_hash.Locate(sigma, as[j], k, n, b_rt);
182 a_prime = static_cast<uint>(index);
183 assert(a_prime < (1u << bucket_bits));
184 b_buf_prime = b_bufs[j];
185 }
186
187 // The inner Gen shares the correction words between the parties and differs in the seed.
188 typename InnerDpf::Key &bk = k0.bks[i];
189 inner_dpf.Gen(bk, k1.bks[i], s0s[i], static_cast<uint>(a_prime), b_buf_prime);
190 }
191
192 return 0;
193 }
194
195 if (!retry) return 1;
196
197 // Deterministically perturb the PRP seed and the initial seeds, then retry.
198 sigma.x = util::IncrementWord(sigma.x);
199 for (int i = 0; i < m; ++i) {
200 s0s[i][0].x = util::IncrementWord(s0s[i][0].x);
201 s0s[i][1].y = util::IncrementWord(s0s[i][1].y);
202 }
203 }
204
205 return 1;
206 }

218 void BatchEval(bool b, const Key &key, cuda::std::span<const In> xs, cuda::std::span<int4> ys) {
219 size_t eta = xs.size();
220 assert(ys.size() >= eta);
221
222 int m_ = key.m_rt;
223 int b_rt = key.b_size_rt;
224
225 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
226
227 // Build per-bucket input lists.
228 // inputs[i] = vector of (within_bucket_index, original_input_index).
229 std::vector<std::vector<std::pair<uint, size_t>>> inputs(m);
230
231 for (size_t omega = 0; omega < eta; ++omega) {
232 // store for easier deduplication for this omega
233 std::array<std::pair<int, uint>, kappa> seen{};
234 int num_seen = 0;
235
236 for (int k = 0; k < kappa; ++k) {
237 auto [bucket, index] = prp_hash.Locate(key.sigma, xs[omega], k, n, b_rt);
238 if (bucket >= m) continue;
239
240 uint j = static_cast<uint>(index);
241 assert(j < (1u << bucket_bits));
242
243 // Deduplicate among this omega's located records only. Permu is a permutation over
244 // [0, n * kappa), so distinct k never collide and the check is only a safeguard.
245 bool dup = false;
246 for (int s = 0; s < num_seen; ++s) {
247 if (seen[s].first == bucket && seen[s].second == j) {
248 dup = true;
249 break;
250 }
251 }
252 if (!dup) {
253 inputs[bucket].push_back({j, omega});
254 seen[num_seen++] = {bucket, j};
255 }
256 }
257 }
258
259 // Initialize outputs.
260 for (size_t i = 0; i < eta; ++i) {
261 ys[i] = {0, 0, 0, 0};
262 }
263
264 // Evaluate per bucket.
265 InnerDpf inner_dpf{prg};
266 for (int i = 0; i < m; ++i) {
267 for (auto &[j, omega] : inputs[i]) {
268 int4 y = inner_dpf.Eval(b, key.bks[i], j);
269
270 // Accumulate output.
271 ys[omega] = (Group::From(ys[omega]) + Group::From(y)).Into();
272 }
273 }
274 }

283 void EvalAll(bool b, const Key &key, cuda::std::span<int4> ys) {
284 assert(ys.size() == n);
285
286 int bucket_count = key.m_rt;
287 int b_rt = key.b_size_rt;
288
289 // build per-bucket input lists.
290 // inputs[i] = vector of (within_bucket_index, original_input_index).
291 std::vector<std::vector<std::pair<uint, size_t>>> inputs(bucket_count);
292 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
293
294 for (size_t omega = 0; omega < n; ++omega) {
295 In x = static_cast<In>(omega);
296
297 // store for easier deduplication for this omega
298 std::array<std::pair<int, uint>, kappa> seen{};
299 int num_seen = 0;
300
301 for (int k = 0; k < kappa; ++k) {
302 auto [bucket, index] = prp_hash.Locate(key.sigma, x, k, n, b_rt);
303 if (bucket >= bucket_count) continue;
304
305 uint j = static_cast<uint>(index);
306 assert(j < (1u << bucket_bits));
307
308 // Deduplicate among this omega's located records (linear scan, fine for small kappa).
309 bool dup = false;
310
311 for (int s = 0; s < num_seen; ++s) {
312 if (seen[s].first == bucket && seen[s].second == j) {
313 dup = true;
314 break;
315 }
316 }
317
318 if (!dup) {
319 inputs[bucket].push_back({j, omega});
320 seen[num_seen++] = {bucket, j};
321 }
322 }
323 }
324
325 // initialize outputs.
326 for (size_t i = 0; i < n; ++i) {
327 ys[i] = {0, 0, 0, 0};
328 }
329
330 // evaluate per bucket and merge.
331 InnerDpf inner_dpf{prg};
332 std::vector<int4> bucket_ys(1ULL << bucket_bits, {0, 0, 0, 0});
333 for (int i = 0; i < bucket_count; ++i) {
334 inner_dpf.EvalAll(b, key.bks[i], cuda::std::span<int4>(bucket_ys.data(), bucket_ys.size()));
335
336 // merge
337 for (auto &[j, omega] : inputs[i]) {
338 ys[omega] = (Group::From(ys[omega]) + Group::From(bucket_ys[j])).Into();
339 }
340 }
341 }
