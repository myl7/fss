80template <int in_bits, int max_points, int bucket_bits, typename Group, typename Prg, typename XorHash, typename Hash,
81 typename Prp, typename In = uint, int kappa = 3, int ch_lambda = 80>
82 requires((std::is_unsigned_v<In> || std::is_same_v<In, __uint128_t>) && in_bits <= sizeof(In) * 8 &&
83 Groupable<Group> && Prgable<Prg, 2> && XorHashable<XorHash> && Hashable<Hash> && Permutable<Prp>)
84class Vdmpf {
85public:
86 static_assert(max_points >= 30, "max_points must be >= 30 (Remark 1 of the paper)");
87 static constexpr int m = cuckoo_hash::ChBucket(max_points, ch_lambda);
88 static constexpr __uint128_t n = __uint128_t(1) << in_bits;
89 // b_size = ceil(n * kappa / m), use __uint128_t to avoid overflow.
90 static constexpr int b_size = static_cast<int>((static_cast<__uint128_t>(n) * kappa + m - 1) / m);
91 static_assert(b_size <= (1 << bucket_bits));

93 using InnerVdpf = Vdpf<bucket_bits, Group, Prg, XorHash, Hash, uint>;

136 int Gen(Key &k0, Key &k1, int4 &sigma, cuda::std::span<cuda::std::array<int4, 2>, m> s0s,
137 cuda::std::span<const In> as, cuda::std::span<const int4> b_bufs, int t, int ch_retry = 1000,
138 bool retry = true) {
139 assert(t <= max_points);
140 assert(t >= 30);
141
142 // Compute runtime bucket count and bucket size.
143 int m_ = cuckoo_hash::ChBucket(t, ch_lambda);
144 assert(m_ <= m);
145 __uint128_t n128 = static_cast<__uint128_t>(n);
146 int b_rt = static_cast<int>((n128 * kappa + m_ - 1) / m_);
147 assert(b_rt <= (1 << bucket_bits));
148
149 k0.m_rt = m_;
150 k1.m_rt = m_;
151 k0.b_size_rt = b_rt;
152 k1.b_size_rt = b_rt;
153
154 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
155 cuckoo_hash::Compact<Prp, In, kappa> compact{prp};
156
157 for (int attempt = 0; attempt < 128; ++attempt) {
158 k0.sigma = sigma;
159 k1.sigma = sigma;
160
161 // Run Cuckoo hashing.
162 std::vector<std::pair<int, int>> table(m_, {-1, -1});
163 int ret = compact.Run(std::span<const In>(as.data(), t), m_, sigma, n, b_rt, ch_retry,
164 std::span<std::pair<int, int>>(table));
165
166 // Generate inner VDPF keys for each bucket.
167 if (ret == 0) {
168 InnerVdpf inner_vdpf{prg, xor_hash, hash};
169 for (int i = 0; i < m && ret == 0; ++i) {
170 uint a_prime = 0;
171 int4 b_buf_prime = {0, 0, 0, 0};
172
173 if (i < m_ && table[i].first != -1) {
174 int j = table[i].first; // index into as
175 int k = table[i].second; // hash function that placed it
176 auto [bucket, index] = prp_hash.Locate(sigma, as[j], k, n, b_rt);
177 a_prime = static_cast<uint>(index);
178 assert(a_prime < (1u << bucket_bits));
179 b_buf_prime = b_bufs[j];
180 }
181
182 // The inner Gen shares cws, cs, and ocw between the parties and differs in the seed.
183 typename InnerVdpf::Key &bk = k0.bks[i];
184 ret = inner_vdpf.Gen(bk, k1.bks[i], s0s[i], static_cast<uint>(a_prime), b_buf_prime);
185 }
186 }
187 if (ret == 0) return 0;
188
189 if (!retry) return 1;
190
191 // Deterministically perturb the PRP seed and the initial seeds, then retry.
192 sigma.x += 1;
193 for (int i = 0; i < m; ++i) {
194 s0s[i][0].x += 1;
195 s0s[i][1].y += 1;
196 }
197 }
198
199 return 1;
200 }

213 void BatchEval(bool b, const Key &key, cuda::std::span<const In> xs, cuda::std::span<int4> ys,
214 cuda::std::array<int4, 4> &pi) {
215 size_t eta = xs.size();
216 assert(ys.size() >= eta);
217
218 int m_ = key.m_rt;
219 int b_rt = key.b_size_rt;
220
221 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
222
223 // Build per-bucket input lists.
224 // inputs[i] = vector of (within_bucket_index, original_input_index).
225 std::vector<std::vector<std::pair<uint, size_t>>> inputs(m);
226 for (size_t omega = 0; omega < eta; ++omega) {
227 // store for easier deduplication for this omega
228 cuda::std::array<std::pair<int, uint>, kappa> seen{};
229 int num_seen = 0;
230
231 for (int k = 0; k < kappa; ++k) {
232 auto [bucket, index] = prp_hash.Locate(key.sigma, xs[omega], k, n, b_rt);
233 if (bucket >= m) continue;
234 uint j = static_cast<uint>(index);
235 assert(j < (1u << bucket_bits));
236 // Deduplicate among this omega's located records only. Permu is a permutation over
237 // [0, n * kappa), so distinct k never collide and the check is only a safeguard.
238 bool dup = false;
239 for (int s = 0; s < num_seen; ++s) {
240 if (seen[s].first == bucket && seen[s].second == j) {
241 dup = true;
242 break;
243 }
244 }
245 if (!dup) {
246 inputs[bucket].push_back({j, omega});
247 seen[num_seen++] = {bucket, j};
248 }
249 }
250 }
251
252 // Initialize outputs and proof.
253 for (size_t i = 0; i < eta; ++i) {
254 ys[i] = {0, 0, 0, 0};
255 }
256 pi = {int4{0, 0, 0, 0}, int4{0, 0, 0, 0}, int4{0, 0, 0, 0}, int4{0, 0, 0, 0}};
257
258 // Evaluate per bucket.
259 InnerVdpf inner_vdpf{prg, xor_hash, hash};
260 for (int i = 0; i < m; ++i) {
261 // Initialize per-bucket proof from this bucket's cs.
262 cuda::std::array<int4, 4> pi_bucket = {key.bks[i].cs[0], key.bks[i].cs[1], key.bks[i].cs[2], key.bks[i].cs[3]};
263
264 for (auto &[j, omega] : inputs[i]) {
265 auto [y, pi_tilde] = inner_vdpf.Eval(b, key.bks[i], j);
266
267 // Accumulate output.
268 ys[omega] = (Group::From(ys[omega]) + Group::From(y)).Into();
269
270 // Accumulate per-bucket proof (same as Vdpf::Prove logic).
271 cuda::std::array<int4, 4> h_input =
272 util::Xor(cuda::std::span<const int4, 4>(pi_bucket), cuda::std::span<const int4, 4>(pi_tilde));
273 auto h_out = hash.Hash(cuda::std::span<const int4, 4>(h_input));
274 pi_bucket[0] = util::Xor(pi_bucket[0], h_out[0]);
275 pi_bucket[1] = util::Xor(pi_bucket[1], h_out[1]);
276 }
277
278 // Cross-bucket proof accumulation: pi = pi XOR H'(pi XOR pi_bucket).
279 cuda::std::array<int4, 4> cross_input =
280 util::Xor(cuda::std::span<const int4, 4>(pi), cuda::std::span<const int4, 4>(pi_bucket));
281 auto cross_out = hash.Hash(cuda::std::span<const int4, 4>(cross_input));
282 pi[0] = util::Xor(pi[0], cross_out[0]);
283 pi[1] = util::Xor(pi[1], cross_out[1]);
284 }
285 }

329 void BatchEvalMerkle(bool b, const Key &key, cuda::std::span<const In> xs, cuda::std::span<int4> ys,
330 cuda::std::array<int4, 4> &pi) {
331 size_t eta = xs.size();
332 assert(ys.size() >= eta);
333
334 int m_ = key.m_rt;
335 int b_rt = key.b_size_rt;
336
337 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
338
339 // Build per-bucket input lists in the canonical (input, hash) order: the
340 // GPU path reproduces this order through a stable sort of the same
341 // per-(input, hash) records.
342 std::vector<std::vector<std::pair<uint, size_t>>> inputs(m);
343
344 for (size_t omega = 0; omega < eta; ++omega) {
345 for (int k = 0; k < kappa; ++k) {
346 auto [bucket, index] = prp_hash.Locate(key.sigma, xs[omega], k, n, b_rt);
347 if (bucket >= m) continue;
348 uint j = static_cast<uint>(index);
349 assert(j < (1u << bucket_bits));
350 inputs[bucket].push_back({j, omega});
351 }
352 }
353
354 for (size_t i = 0; i < eta; ++i) {
355 ys[i] = {0, 0, 0, 0};
356 }
357
358 InnerVdpf inner_vdpf{prg, xor_hash, hash};
359 std::vector<cuda::std::array<int4, 2>> bucket_nodes(m);
360 std::vector<cuda::std::array<int4, 2>> nodes;
361 for (int i = 0; i < m; ++i) {
362 // The bucket's node starts at its hashed cs and absorbs every chunk
363 // root of its leaf sequence left to right.
364 cuda::std::array<int4, 2> bucket_node = hash.Hash(cuda::std::span<const int4, 4>(key.bks[i].cs));
365 size_t consumed = 0;
366 while (consumed < inputs[i].size()) {
367 size_t take = inputs[i].size() - consumed;
368 if (take > static_cast<size_t>(kMerkleChunkSize)) take = kMerkleChunkSize;
369 nodes.clear();
370 nodes.reserve(take);
371 for (size_t c = 0; c < take; ++c) {
372 auto [j, omega] = inputs[i][consumed + c];
373 auto [y, pi_tilde] = inner_vdpf.Eval(b, key.bks[i], j);
374 ys[omega] = (Group::From(ys[omega]) + Group::From(y)).Into();
375 nodes.push_back(hash.Hash(cuda::std::span<const int4, 4>(pi_tilde)));
376 }
377 cuda::std::array<int4, 2> root = FoldNodes(nodes);
378 int4 msg[4] = {bucket_node[0], bucket_node[1], root[0], root[1]};
379 bucket_node = hash.Hash(cuda::std::span<const int4, 4>(msg));
380 consumed += take;
381 }
382 bucket_nodes[i] = bucket_node;
383 }
384
385 cuda::std::array<int4, 2> root = FoldNodes(bucket_nodes);
386 pi = {root[0], root[1], int4{0, 0, 0, 0}, int4{0, 0, 0, 0}};
387 }
388
389private:
390 // Level-by-level pairwise fold with an odd trailing node carrying to the
391 // next level unchanged, as in Vdpf::ProveMerkle.
392 cuda::std::array<int4, 2> FoldNodes(std::vector<cuda::std::array<int4, 2>> &nodes) {
393 size_t count = nodes.size();
394 assert(count >= 1);
395 while (count > 1) {
396 size_t pairs = count / 2;
397 for (size_t i = 0; i < pairs; ++i) {
398 int4 msg[4] = {nodes[2 * i][0], nodes[2 * i][1], nodes[2 * i + 1][0], nodes[2 * i + 1][1]};
399 nodes[i] = hash.Hash(cuda::std::span<const int4, 4>(msg));
400 }
401 if (count % 2 != 0) nodes[pairs] = nodes[count - 1];
402 count = pairs + count % 2;
403 }
404 return nodes[0];
405 }
