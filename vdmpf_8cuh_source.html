80template <int in_bits, int max_points, int bucket_bits, typename Group, typename Prg, typename XorHash, typename Hash,
81 typename Prp, typename In = uint, int kappa = 3, int ch_lambda = 80>
82 requires((std::is_unsigned_v<In> || std::is_same_v<In, __uint128_t>) && in_bits <= sizeof(In) * 8 &&
83 Groupable<Group> && Prgable<Prg, 2> && XorHashable<XorHash> && Hashable<Hash> && Permutable<Prp>)
84class Vdmpf {
85public:
86 static_assert(max_points >= 30, "max_points must be >= 30 (Remark 1 of the paper)");
87 static_assert(bucket_bits >= 1 && bucket_bits <= 30, "bucket_bits must be in [1, 30]");
88 static_assert(kappa > 0, "kappa must be positive");
89 static constexpr int m = cuckoo_hash::ChBucket(max_points, ch_lambda);
90 static_assert(m > 0, "bucket count must be positive");
91 static constexpr __uint128_t n = __uint128_t(1) << in_bits;
92 // Bound capacity before forming n * kappa, which may overflow even __uint128_t.
93 static_assert(n <= ((__uint128_t{1} << bucket_bits) * m) / kappa,
94 "bucket_bits is too small for the bucket size");
95 // b_size = ceil(n * kappa / m). The capacity bound keeps the numerator in range.
96 static constexpr __uint128_t kBucketSize = (n * kappa + m - 1) / m;
97 static constexpr int b_size = static_cast<int>(kBucketSize);

135 int Gen(Key &k0, Key &k1, int4 sigma, cuda::std::span<const cuda::std::array<int4, 2>, m> s0s, std::span<const In> as,
136 std::span<const int4> b_bufs, int t, int ch_retry = 1000) {
137 assert(t <= max_points);
138
139 k0.sigma = sigma;
140 k1.sigma = sigma;
141
142 // Compute runtime bucket count and bucket size.
143 assert(t >= 30);
144 int m_ = cuckoo_hash::ChBucket(t, ch_lambda);
145 assert(m_ <= m);
146 __uint128_t n128 = static_cast<__uint128_t>(n);
147 __uint128_t bucket_size = (n128 * kappa + m_ - 1) / m_;
148 assert(bucket_size <= (uint64_t{1} << bucket_bits));
149 int b_rt = static_cast<int>(bucket_size);
150
151 k0.m_rt = m_;
152 k1.m_rt = m_;
153 k0.b_size_rt = b_rt;
154 k1.b_size_rt = b_rt;
155
156 // Run Cuckoo hashing.
157 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
158 std::vector<std::pair<int, int>> table(m_, {-1, -1});
159 cuckoo_hash::Compact<Prp, In, kappa> compact{prp};
160 int ret = compact.Run(as.first(t), m_, sigma, n, b_rt, ch_retry, std::span<std::pair<int, int>>(table));
161 if (ret != 0) return 1;
162
163 // Generate inner VDPF keys for each bucket.
164 InnerVdpf inner_vdpf{prg, xor_hash, hash};
165 for (int i = 0; i < m; ++i) {
166 uint a_prime = 0;
167 int4 b_buf_prime = {0, 0, 0, 0};
168
169 if (i < m_ && table[i].first != -1) {
170 int j = table[i].first; // index into as
171 int k = table[i].second; // hash function that placed it
172 auto [bucket, index] = prp_hash.Locate(sigma, as[j], k, n, b_rt);
173 a_prime = static_cast<uint>(index);
174 assert(a_prime < (1u << bucket_bits));
175 b_buf_prime = b_bufs[j];
176 }
177
178 ret = inner_vdpf.Gen(k0.bks[i].cws, k0.bks[i].cs, k0.bks[i].ocw, s0s[i], static_cast<uint>(a_prime), b_buf_prime);
179 if (ret != 0) return 1;
180
181 k0.bks[i].s0 = s0s[i][0];
182 k1.bks[i].s0 = s0s[i][1];
183 // Copy shared parts.
184 for (int l = 0; l < bucket_bits; ++l) k1.bks[i].cws[l] = k0.bks[i].cws[l];
185 k1.bks[i].cs = k0.bks[i].cs;
186 k1.bks[i].ocw = k0.bks[i].ocw;
187 }
188
189 return 0;
190 }

217 int Gen(Key &k0, Key &k1, int4 &sigma, cuda::std::span<cuda::std::array<int4, 2>, m> s0s,
218 cuda::std::span<const In> as, cuda::std::span<const int4> b_bufs, int t, int ch_retry,
219 bool retry) {
220 assert(t <= max_points);
221 assert(t >= 30);
222
223 // Compute runtime bucket count and bucket size.
224 int m_ = cuckoo_hash::ChBucket(t, ch_lambda);
225 assert(m_ <= m);
226 __uint128_t n128 = static_cast<__uint128_t>(n);
227 __uint128_t bucket_size = (n128 * kappa + m_ - 1) / m_;
228 assert(bucket_size <= (uint64_t{1} << bucket_bits));
229 int b_rt = static_cast<int>(bucket_size);
230
231 k0.m_rt = m_;
232 k1.m_rt = m_;
233 k0.b_size_rt = b_rt;
234 k1.b_size_rt = b_rt;
235
236 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
237 cuckoo_hash::Compact<Prp, In, kappa> compact{prp};
238
239 for (int attempt = 0; attempt < 128; ++attempt) {
240 k0.sigma = sigma;
241 k1.sigma = sigma;
242
243 // Run Cuckoo hashing.
244 std::vector<std::pair<int, int>> table(m_, {-1, -1});
245 int ret = compact.Run(std::span<const In>(as.data(), t), m_, sigma, n, b_rt, ch_retry,
246 std::span<std::pair<int, int>>(table));
247
248 // Generate inner VDPF keys for each bucket.
249 if (ret == 0) {
250 InnerVdpf inner_vdpf{prg, xor_hash, hash};
251 for (int i = 0; i < m && ret == 0; ++i) {
252 uint a_prime = 0;
253 int4 b_buf_prime = {0, 0, 0, 0};
254
255 if (i < m_ && table[i].first != -1) {
256 int j = table[i].first; // index into as
257 int k = table[i].second; // hash function that placed it
258 auto [bucket, index] = prp_hash.Locate(sigma, as[j], k, n, b_rt);
259 a_prime = static_cast<uint>(index);
260 assert(a_prime < (1u << bucket_bits));
261 b_buf_prime = b_bufs[j];
262 }
263
264 // The inner Gen shares cws, cs, and ocw between the parties and differs in the seed.
265 typename InnerVdpf::Key bk0, bk1;
266 ret = inner_vdpf.Gen(bk0, bk1, s0s[i], static_cast<uint>(a_prime), b_buf_prime);
267 if (ret == 0) {
268 for (int l = 0; l < bucket_bits; ++l) {
269 k0.bks[i].cws[l] = bk0.cws[l];
270 k1.bks[i].cws[l] = bk1.cws[l];
271 }
272 k0.bks[i].cs = bk0.cs;
273 k1.bks[i].cs = bk1.cs;
274 k0.bks[i].ocw = bk0.ocw;
275 k1.bks[i].ocw = bk1.ocw;
276 k0.bks[i].s0 = bk0.s0;
277 k1.bks[i].s0 = bk1.s0;
278 }
279 }
280 }
281 if (ret == 0) return 0;
282
283 if (!retry) return 1;
284
285 // Deterministically perturb the PRP seed and the initial seeds, then retry.
286 sigma.x = util::IncrementWord(sigma.x);
287 for (int i = 0; i < m; ++i) {
288 s0s[i][0].x = util::IncrementWord(s0s[i][0].x);
289 s0s[i][1].y = util::IncrementWord(s0s[i][1].y);
290 }
291 }
292
293 return 1;
294 }

313 template <size_t x_extent, size_t y_extent>
314 void BatchEval(bool b, const Key &key, cuda::std::span<const In, x_extent> xs,
315 cuda::std::span<int4, y_extent> ys, cuda::std::array<int4, 4> &pi) {
316 size_t eta = xs.size();
317 assert(ys.size() >= eta);
318
319 int m_ = key.m_rt;
320 int b_rt = key.b_size_rt;
321
322 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
323
324 // Build per-bucket input lists.
325 // inputs[i] = vector of (within_bucket_index, original_input_index).
326 std::vector<std::vector<std::pair<uint, size_t>>> inputs(m);
327 for (size_t omega = 0; omega < eta; ++omega) {
328 // store for easier deduplication for this omega
329 cuda::std::array<std::pair<int, uint>, kappa> seen{};
330 int num_seen = 0;
331
332 for (int k = 0; k < kappa; ++k) {
333 auto [bucket, index] = prp_hash.Locate(key.sigma, xs[omega], k, n, b_rt);
334 if (bucket >= m) continue;
335 uint j = static_cast<uint>(index);
336 assert(j < (1u << bucket_bits));
337 // Deduplicate among this omega's located records only. Permu is a permutation over
338 // [0, n * kappa), so distinct k never collide and the check is only a safeguard.
339 bool dup = false;
340 for (int s = 0; s < num_seen; ++s) {
341 if (seen[s].first == bucket && seen[s].second == j) {
342 dup = true;
343 break;
344 }
345 }
346 if (!dup) {
347 inputs[bucket].push_back({j, omega});
348 seen[num_seen++] = {bucket, j};
349 }
350 }
351 }
352
353 // Initialize outputs and proof.
354 for (size_t i = 0; i < eta; ++i) {
355 ys[i] = {0, 0, 0, 0};
356 }
357 pi = {int4{0, 0, 0, 0}, int4{0, 0, 0, 0}, int4{0, 0, 0, 0}, int4{0, 0, 0, 0}};
358
359 // Evaluate per bucket.
360 InnerVdpf inner_vdpf{prg, xor_hash, hash};
361 for (int i = 0; i < m; ++i) {
362 // Initialize per-bucket proof from this bucket's cs.
363 cuda::std::array<int4, 4> pi_bucket = {key.bks[i].cs[0], key.bks[i].cs[1], key.bks[i].cs[2], key.bks[i].cs[3]};
364
365 for (auto &[j, omega] : inputs[i]) {
366 int4 y;
367 auto pi_tilde = inner_vdpf.Eval(b, key.bks[i].s0, key.bks[i].cws, key.bks[i].cs, key.bks[i].ocw, j, y);
368
369 // Accumulate output.
370 ys[omega] = (Group::From(ys[omega]) + Group::From(y)).Into();
371
372 // Accumulate per-bucket proof (same as Vdpf::Prove logic).
373 cuda::std::array<int4, 4> h_input =
374 util::Xor(cuda::std::span<const int4, 4>(pi_bucket), cuda::std::span<const int4, 4>(pi_tilde));
375 auto h_out = hash.Hash(cuda::std::span<const int4, 4>(h_input));
376 pi_bucket[0] = util::Xor(pi_bucket[0], h_out[0]);
377 pi_bucket[1] = util::Xor(pi_bucket[1], h_out[1]);
378 }
379
380 // Cross-bucket proof accumulation: pi = pi XOR H'(pi XOR pi_bucket).
381 cuda::std::array<int4, 4> cross_input =
382 util::Xor(cuda::std::span<const int4, 4>(pi), cuda::std::span<const int4, 4>(pi_bucket));
383 auto cross_out = hash.Hash(cuda::std::span<const int4, 4>(cross_input));
384 pi[0] = util::Xor(pi[0], cross_out[0]);
385 pi[1] = util::Xor(pi[1], cross_out[1]);
386 }
387 }

431 void BatchEvalMerkle(bool b, const Key &key, cuda::std::span<const In> xs, cuda::std::span<int4> ys,
432 cuda::std::array<int4, 4> &pi) {
433 size_t eta = xs.size();
434 assert(ys.size() >= eta);
435
436 int m_ = key.m_rt;
437 int b_rt = key.b_size_rt;
438
439 cuckoo_hash::PrpHash<Prp, In, kappa> prp_hash{prp};
440
441 // Build per-bucket input lists in the canonical (input, hash) order: the
442 // GPU path reproduces this order through a stable sort of the same
443 // per-(input, hash) records.
444 std::vector<std::vector<std::pair<uint, size_t>>> inputs(m);
445
446 for (size_t omega = 0; omega < eta; ++omega) {
447 for (int k = 0; k < kappa; ++k) {
448 auto [bucket, index] = prp_hash.Locate(key.sigma, xs[omega], k, n, b_rt);
449 if (bucket >= m) continue;
450 uint j = static_cast<uint>(index);
451 assert(j < (1u << bucket_bits));
452 inputs[bucket].push_back({j, omega});
453 }
454 }
455
456 for (size_t i = 0; i < eta; ++i) {
457 ys[i] = {0, 0, 0, 0};
458 }
459
460 InnerVdpf inner_vdpf{prg, xor_hash, hash};
461 std::vector<cuda::std::array<int4, 2>> bucket_nodes(m);
462 std::vector<cuda::std::array<int4, 2>> nodes;
463 for (int i = 0; i < m; ++i) {
464 // The bucket's node starts at its hashed cs and absorbs every chunk
465 // root of its leaf sequence left to right.
466 cuda::std::array<int4, 2> bucket_node = hash.Hash(cuda::std::span<const int4, 4>(key.bks[i].cs));
467 size_t consumed = 0;
468 while (consumed < inputs[i].size()) {
469 size_t take = inputs[i].size() - consumed;
470 if (take > static_cast<size_t>(kMerkleChunkSize)) take = kMerkleChunkSize;
471 nodes.clear();
472 nodes.reserve(take);
473 for (size_t c = 0; c < take; ++c) {
474 auto [j, omega] = inputs[i][consumed + c];
475 int4 y;
476 auto pi_tilde = inner_vdpf.Eval(b, key.bks[i].s0, key.bks[i].cws, key.bks[i].cs, key.bks[i].ocw, j, y);
477 ys[omega] = (Group::From(ys[omega]) + Group::From(y)).Into();
478 nodes.push_back(hash.Hash(cuda::std::span<const int4, 4>(pi_tilde)));
479 }
480 cuda::std::array<int4, 2> root = FoldNodes(nodes);
481 int4 msg[4] = {bucket_node[0], bucket_node[1], root[0], root[1]};
482 bucket_node = hash.Hash(cuda::std::span<const int4, 4>(msg));
483 consumed += take;
484 }
485 bucket_nodes[i] = bucket_node;
486 }
487
488 cuda::std::array<int4, 2> root = FoldNodes(bucket_nodes);
489 pi = {root[0], root[1], int4{0, 0, 0, 0}, int4{0, 0, 0, 0}};
490 }
491
492private:
493 // Level-by-level pairwise fold with an odd trailing node carrying to the
494 // next level unchanged, as in Vdpf::ProveMerkle.
495 cuda::std::array<int4, 2> FoldNodes(std::vector<cuda::std::array<int4, 2>> &nodes) {
496 size_t count = nodes.size();
497 assert(count >= 1);
498 while (count > 1) {
499 size_t pairs = count / 2;
500 for (size_t i = 0; i < pairs; ++i) {
501 int4 msg[4] = {nodes[2 * i][0], nodes[2 * i][1], nodes[2 * i + 1][0], nodes[2 * i + 1][1]};
502 nodes[i] = hash.Hash(cuda::std::span<const int4, 4>(msg));
503 }
504 if (count % 2 != 0) nodes[pairs] = nodes[count - 1];
505 count = pairs + count % 2;
506 }
507 return nodes[0];
508 }
509};
