68// Split correction-word view shared by the host and device split paths. Layer i is
69// read from cws[i*stride] and the right-control correction is bit i of
70// packed_tr. The terminal is rebuilt as a plain Cw with no control bit, so bits of
71// packed_tr at or above in_bits are never read; the shift by in_bits would be
72// undefined when in_bits equals the width of In. The layer LSB keeps the
73// left-control correction exactly as the Cw layout does. operator[] recovers the Cw
74// the shared algorithms consume, so their bodies stay identical on the Cw and split
75// paths.
76template <typename Cw, typename In, int in_bits>
77struct DpfCwsView {
78 const int4 *s;
79 In packed_tr;
80 size_t stride;
81
82 __host__ __device__ inline Cw operator[](int i) const {
83 const int4 s_cw = s[static_cast<size_t>(i) * stride];
84 if (i >= in_bits) return Cw{s_cw, false};
85 return Cw{s_cw, ((packed_tr >> i) & 1) != 0};
86 }
87};

571// The single-key and batch kernels share the same block-local frontier and tail.
572// Each node keeps the former full-domain helper's by-value backend lifetime.
573// Cws is a Cw pointer on the array path and a kernel-boundary split view
574// on the split path, so the traversal body does not depend on the layout.
575template <typename Scheme, int frontier_depth, int block_root_depth, bool range = false, typename Cws>
576__device__ __forceinline__ void DpfEvalAllBlock(bool b, int4 s0, Cws cws,
577 int4 *ys, typename DpfGpuSchemeTraits<Scheme>::Backend &prg, int subtree_block_index, int64_t thread_index, int4 *shared_frontier,
578 __uint128_t first = 0, int subtree_depth = DpfGpuSchemeTraits<Scheme>::kInputBits,
579 int range_frontier = frontier_depth, int range_block_root = block_root_depth) {
580 constexpr int kInputBits = DpfGpuSchemeTraits<Scheme>::kInputBits;
581 constexpr int kTailCapacity = range ? 8 : kInputBits - frontier_depth;
582 const int start_depth = range ? kInputBits - subtree_depth : 0;
583 const int frontier = range ? start_depth + range_frontier : frontier_depth;
584 const int root_depth = range ? start_depth + range_block_root : block_root_depth;
585 const int tail_depth = kInputBits - frontier;
586 const int block_subtree_depth = frontier - root_depth;
587 // Phase 0: block root walk (top block_root_depth levels) by thread 0.
588 if (threadIdx.x == 0) {
589 int4 st = util::SetLsb(s0, b);
590#pragma unroll
591 for (int i = 0; i < root_depth; ++i) {
592 bool x_bit = i < start_depth
593 ? ((first >> (kInputBits - 1 - i)) & 1) != 0
594 : ((subtree_block_index >> (root_depth - 1 - i)) & 1) != 0;
595 {
596 // Preserve the old by-value path helper's copy and destruction boundary.
597 auto node_prg = prg;
598 st = fss::detail::SchemeAccess::DpfPathStep<Scheme>(node_prg, st, cws[i].s, cws[i].tr, x_bit);
599 }
600 }
601 shared_frontier[0] = st;
602 }
603 __syncthreads();
604
605 // Phase 1: breadth-parallel expansion of the block's subtree, levels block_root_depth..frontier_depth.
606#pragma unroll
607 for (int i = 0; i < block_subtree_depth; ++i) {
608 const bool active = threadIdx.x < (1 << (i + 1));
609 int4 n = {0, 0, 0, 0};
610 if (active) n = shared_frontier[threadIdx.x >> 1];
611 // All parent reads must finish before any child overwrites the frontier.
612 __syncthreads();
613 if (active) {
614 int4 left, right;
615 {
616 // Preserve the old by-value expansion helper's copy and destruction boundary.
617 auto node_prg = prg;
618 fss::detail::SchemeAccess::DpfExpandNode<Scheme>(node_prg, n, cws[root_depth + i].s, cws[root_depth + i].tr, left, right);
619 }
620 shared_frontier[threadIdx.x] = (threadIdx.x & 1) ? right : left;
621 }
622 __syncthreads();
623 }
624
625 if constexpr (range) {
626 int4 pending[8];
627 int4 node = shared_frontier[threadIdx.x];
628 int level = frontier;
629 const int count = 1 << tail_depth;
630 const size_t offset = static_cast<size_t>(thread_index) << tail_depth;
631 for (int index = 0; index < count; ++index) {
632 while (level < kInputBits) {
633 int4 left, right;
634 {
635 auto node_prg = prg;
636 fss::detail::SchemeAccess::DpfExpandNode<Scheme>(node_prg, node,
637 cws[level].s, cws[level].tr, left, right);
638 }
639 pending[level - frontier] = right;
640 node = left;
641 ++level;
642 }
643 ys[offset + index] = fss::detail::SchemeAccess::DpfConvertLeaf<Scheme>(
644 b, node, cws[kInputBits].s);
645 if (index + 1 < count) {
646 int next = index + 1;
647 level = kInputBits - 1;
648 while ((next & 1) == 0) {
649 --level;
650 next >>= 1;
651 }
652 node = pending[level - frontier];
653 ++level;
654 }
655 }
656 } else {
657 // Phase 2: expand the subtree below depth frontier_depth, then convert to leaves.
658 int4 nodes[1 << kTailCapacity];
659 nodes[0] = shared_frontier[threadIdx.x];
660 int count = 1;
661#pragma unroll
662 for (int lev = frontier; lev < kInputBits; ++lev) {
663#pragma unroll
664 for (int k = count - 1; k >= 0; --k) {
665 int4 left, right;
666 {
667 // Preserve the old by-value expansion helper's copy and destruction boundary.
668 auto node_prg = prg;
669 fss::detail::SchemeAccess::DpfExpandNode<Scheme>(node_prg, nodes[k], cws[lev].s, cws[lev].tr, left, right);
670 }
671 nodes[2 * k] = left;
672 nodes[2 * k + 1] = right;
673 }
674 count *= 2;
675 }
676
677 int4 v_cw_np1 = cws[kInputBits].s;
678 const size_t output_offset = static_cast<size_t>(thread_index) << tail_depth;
679#pragma unroll
680 for (int k = 0; k < count; ++k) {
681 ys[output_offset + k] = fss::detail::SchemeAccess::DpfConvertLeaf<Scheme>(b, nodes[k], v_cw_np1);
682 }
683 }
684}

688 int4 *ys, typename DpfGpuSchemeTraits<Scheme>::Backend prg) {

690 // Cw per record, so the typed view is a plain cast.

693 constexpr int kBlocksPerKey = 1 << block_root_depth;

695 const int subtree_block_index = blockIdx.x - key_index * kBlocksPerKey;
696 const int64_t thread_index = static_cast<int64_t>(subtree_block_index) * blockDim.x + threadIdx.x;
697 if (thread_index >= (1LL << frontier_depth)) return;
698 const auto *key_cws = cws + static_cast<size_t>(key_index) * (kInputBits + 1);
699 int4 *key_outputs = ys + static_cast<size_t>(key_index) * (size_t{1} << kInputBits);

701 DpfEvalAllBlock<Scheme, frontier_depth, block_root_depth>(b, s0s[key_index], key_cws, key_outputs, prg, subtree_block_index, thread_index, shared_frontier);

900// Correction words arrive in the dense level-major split layout: layer i of
901// key k is cws[i * key_count + k], the terminal layer is
902// cws[in_bits * key_count + k], and tr[k] holds key k's packed right-control
903// bits, bit i for layer i.
904template <typename Scheme, int unroll, bool b, typename Control, typename Query>
905__global__ void DpfCoreEvalPointKernel(int4 *ys, const int4 *seeds, const int4 *cws,
906 const Control *tr, const int4 *out_cw,
907 const Query *xs, int key_count,
908 typename DpfGpuSchemeTraits<Scheme>::Backend prg) {
909 using In = typename DpfGpuSchemeTraits<Scheme>::Input;
910 constexpr int kInputBits = DpfGpuSchemeTraits<Scheme>::kInputBits;
911 static_assert(kInputBits <= sizeof(In) * 8, "In too narrow for kInputBits");
912 int thread_index = blockIdx.x * blockDim.x + threadIdx.x;
913 if (thread_index >= key_count) return;
914 int4 state = util::SetLsb(seeds[thread_index], b);
915 In x = static_cast<In>(xs[thread_index]);
916 Control ex = tr[thread_index];
917#pragma unroll unroll
918 for (int i = 0; i < kInputBits; ++i) {
919 bool take_right = ((x >> (kInputBits - 1 - i)) & 1) != 0;
920 bool right_control = ((ex >> i) & 1) != 0;
921 state = fss::detail::SchemeAccess::DpfPathStep<Scheme>(
922 prg, state, cws[static_cast<size_t>(i) * key_count + thread_index], right_control, take_right);
923 }
924 ys[thread_index] = fss::detail::SchemeAccess::DpfConvertLeaf<Scheme>(b, state, out_cw[thread_index]);
925}

1032template <int in_bits, typename Group, typename Prg, typename In>
1033 requires(ValidForInputBits<In, in_bits> &&
1034 Groupable<Group> && Prgable<Prg, 2>)
1035template <typename Writer>
1036__host__ __device__ inline void Dpf<in_bits, Group, Prg, In>::Generate(
1037 Writer writer, const int4 *s0s, In a, int4 b_buf) {
1038 int4 s0 = s0s[0];
1039 s0 = util::SetLsb(s0, false);
1040 int4 s1 = s0s[1];
1041 s1 = util::SetLsb(s1, false);
1042 bool t0 = false;
1043 bool t1 = true;
1044 b_buf = util::SetLsb(b_buf, false);
1045
1046 for (int i = 0; i < in_bits; ++i) {
1047 auto [s0l, s0r] = prg_.Gen(s0);
1048 auto [s1l, s1r] = prg_.Gen(s1);
1049
1050 bool t0l = util::GetLsb(s0l);
1051 s0l = util::SetLsb(s0l, false);
1052 bool t0r = util::GetLsb(s0r);
1053 s0r = util::SetLsb(s0r, false);
1054 bool t1l = util::GetLsb(s1l);
1055 s1l = util::SetLsb(s1l, false);
1056 bool t1r = util::GetLsb(s1r);
1057 s1r = util::SetLsb(s1r, false);
1058
1059 bool a_bit = (a >> (in_bits - 1 - i)) & 1;
1060
1061 int4 s_cw;
1062 if (!a_bit) s_cw = util::Xor(s0r, s1r);
1063 else s_cw = util::Xor(s0l, s1l);
1064
1065 bool tl_cw = t0l ^ t1l ^ a_bit ^ 1;
1066 bool tr_cw = t0r ^ t1r ^ a_bit;
1067
1068 if (!a_bit) {
1069 s0 = s0l;
1070 if (t0) s0 = util::Xor(s0, s_cw);
1071 s1 = s1l;
1072 if (t1) s1 = util::Xor(s1, s_cw);
1073
1074 if (t0) t0 = t0l ^ tl_cw;
1075 else t0 = t0l;
1076 if (t1) t1 = t1l ^ tl_cw;
1077 else t1 = t1l;
1078 } else {
1079 s0 = s0r;
1080 if (t0) s0 = util::Xor(s0, s_cw);
1081 s1 = s1r;
1082 if (t1) s1 = util::Xor(s1, s_cw);
1083
1084 if (t0) t0 = t0r ^ tr_cw;
1085 else t0 = t0r;
1086 if (t1) t1 = t1r ^ tr_cw;
1087 else t1 = t1r;
1088 }
1089
1090 s_cw = util::SetLsb(s_cw, tl_cw);
1091 writer.Write(i, s_cw, tr_cw);
1092 }
1093
1094 auto v_cw_np1 = Group::From(b_buf) + (-Group::From(s0)) + Group::From(s1);
1095 if (t1) v_cw_np1 = -v_cw_np1;
1096 writer.Finish(v_cw_np1.Into());
1097}

1177template <int in_bits, typename Group, typename Prg, typename In>

1180template <typename Cws>
1181inline void Dpf<in_bits, Group, Prg, In>::EvalRangeSerial(bool b, In first, In last,
1182 int4 s0, Cws cws, int4 ys[]) {
1183 constexpr In domain_last = [] {
1184 if constexpr (in_bits == sizeof(In) * 8) return static_cast<In>(~In{0});
1185 else return static_cast<In>((In{1} << in_bits) - 1);
1186 }();
1187 struct Frame {
1188 int4 state;
1189 In low;
1190 In high;
1191 int level;
1192 };
1193 cuda::std::array<Frame, in_bits + 1> stack;
1194 size_t pending = 1;
1195 stack[0] = {util::SetLsb(s0, b), In{0}, domain_last, 0};
1196 while (pending != 0) {
1197 const Frame node = stack[--pending];
1198 if (node.level == in_bits) {
1199 const int4 terminal = cws[in_bits].s;
1200 ys[static_cast<size_t>(node.low - first)] = ConvertLeaf(b, node.state, terminal);
1201 continue;
1202 }
1203 const In middle = node.low + ((node.high - node.low) >> 1);
1204 int4 left;
1205 int4 right;
1206 ExpandNode(prg_, node.state, cws[node.level].s, cws[node.level].tr, left, right);
1207 if (last > middle) {
1208 assert(pending < stack.size());
1209 stack[pending++] = {right, static_cast<In>(middle + 1), node.high, node.level + 1};
1210 }
1211 if (first <= middle) {
1212 assert(pending < stack.size());
1213 stack[pending++] = {left, node.low, middle, node.level + 1};
1214 }
1215 }
1216}

1233template <int in_bits, typename Group, typename Prg, typename In>
1234 requires(ValidForInputBits<In, in_bits> &&
1235 Groupable<Group> && Prgable<Prg, 2>)
1236template <typename Cws>
1237inline void Dpf<in_bits, Group, Prg, In>::EvalAllCws(
1238 bool b, int4 s0, Cws cws, int4 ys[], int par_depth) {
1239 assert(par_depth >= 0);
1240 static_assert(detail::FitsFullDomain<int4, in_bits>(),
1241 "full-domain output count and byte size must fit in size_t");
1242
1243 // The terminal LSB is a caller precondition. Check it on the calling thread,
1244 // so a Debug violation aborts there instead of inside an OpenMP worker;
1245 // Release builds do not check it.
1246 util::RequireArg((cws[in_bits].s.w & 1) == 0,
1247 "Dpf::EvalAll: the terminal group value must have a zero LSB");
1248
1249 if constexpr (in_bits < kBatchPrefix + kBatchTail) {
1250 // Domains too small to fill an 8-wide frontier evaluate per input; the
1251 // point walk is cheaper than the batched machinery at this size.
1252 // The Walk call is host-only (it runs inside EvalAllCws's host ParallelFor),
1253 // so nvcc's #20013-D on the host lambda passed to the host/device Walk
1254 // template below is a known false alarm for this instantiation.
1255 constexpr size_t n = size_t{1} << in_bits;
1256 const int4 terminal = cws[in_bits].s;
1257 fss::detail::ParallelFor(par_depth, n, [&](size_t x) {
1258 const int4 state = Walk(b, s0, cws, static_cast<In>(x), [](bool, bool) {});
1259 ys[x] = ConvertLeaf(b, state, terminal);
1260 });
1261 return;
1262 }
1263
1264 int4 st = s0;
1265 st = util::SetLsb(st, b);
1266
1267 int par_depth_ = fss::detail::ResolveParDepth(par_depth, in_bits);
1268
1269 int split_depth = par_depth_;
1270 if (split_depth > in_bits - kBatchPrefix - kBatchTail) split_depth = in_bits - kBatchPrefix - kBatchTail;
1271 if (split_depth > 0) {
1272 // Split at a frontier of 2 ** split_depth subtree roots instead of per-node tasks,
1273 // so the sequential batched walk stays the fast default path.
1274 size_t num_tasks = size_t{1} << split_depth;
1275 std::vector<int4> frontier = BuildParallelFrontier(st, cws, split_depth);
1276 size_t span = (size_t{1} << in_bits) / num_tasks;
1277 fss::detail::ParallelFor(par_depth_, num_tasks, [&](size_t task) {
1278 EvalBatched8(b, frontier[task], cws, ys, task * span, split_depth);
1279 });
1280 } else {
1281 EvalBatched8(b, st, cws, ys, 0, 0);
1282 }
1283}

1388template <int in_bits, typename Group, typename Prg, typename In>
1389 requires(ValidForInputBits<In, in_bits> &&
1390 Groupable<Group> && Prgable<Prg, 2>)
1391template <typename Cws>
1392inline void Dpf<in_bits, Group, Prg, In>::EvalBatched8Group(
1393 bool b, const int4 (&frontier)[8], Cws cws, int4 ys[], size_t l, int i) {
1394 if (i == in_bits - kBatchTail) {
1395 // Expand the group to its 8 << kBatchTail leaves in local memory, then convert.
1396 int4 bufs[2][8 << kBatchTail];
1397 int4 *cur = bufs[0];
1398 int4 *next = bufs[1];
1399 for (int j = 0; j < 8; ++j) cur[j] = frontier[j];
1400 int w = 8;
1401 for (int lv = i; lv < in_bits; ++lv) {
1402 int4 s_cw = util::SetLsb(cws[lv].s, false);
1403 int tl_cw = util::GetLsb(cws[lv].s);
1404 int tr_cw = cws[lv].tr;
1405 for (int g = 0; g < w / 8; ++g) {
1406 cuda::std::array<int4, 8> seeds;
1407 int ts[8];
1408 for (int j = 0; j < 8; ++j) {
1409 ts[j] = static_cast<int>(util::GetLsb(cur[8 * g + j]));
1410 seeds[j] = util::SetLsb(cur[8 * g + j], false);
1411 }
1412 auto out = detail::GenBatch<2>(prg_, seeds);
1413 for (int j = 0; j < 8; ++j) {
1414 int m = -ts[j];
1415 next[16 * g + 2 * j] = CorrectChild(out[j][0], s_cw, m, ts[j] & tl_cw);
1416 next[16 * g + 2 * j + 1] = CorrectChild(out[j][1], s_cw, m, ts[j] & tr_cw);
1417 }
1418 }
1419 int4 *swap = cur;
1420 cur = next;
1421 next = swap;
1422 w *= 2;
1423 }
1424
1425 int4 v_cw_np1 = cws[in_bits].s;
1426 // The shared entry validated the terminal before parallel dispatch.
1427 assert((v_cw_np1.w & 1) == 0);
1428 Group v_cw = Group::From(v_cw_np1);
1429 for (int j = 0; j < w; ++j) ys[l + j] = ConvertLeaf(b, cur[j], v_cw);
1430 return;
1431 }
1432
1433 int4 s_cw = util::SetLsb(cws[i].s, false);
1434 int tl_cw = util::GetLsb(cws[i].s);
1435 int tr_cw = cws[i].tr;
1436
1437 cuda::std::array<int4, 8> seeds;
1438 int ts[8];
1439 for (int j = 0; j < 8; ++j) {
1440 ts[j] = static_cast<int>(util::GetLsb(frontier[j]));
1441 seeds[j] = util::SetLsb(frontier[j], false);
1442 }
1443 auto out = detail::GenBatch<2>(prg_, seeds);
1444 // The 16 children interleave in leaf order: node j's children land at 2j, 2j+1 of the
1445 // next level. De-interleave them into two consecutive groups of 8.
1446 int4 low[8];
1447 int4 high[8];
1448 for (int j = 0; j < 4; ++j) {
1449 int m = -ts[j];
1450 low[2 * j] = CorrectChild(out[j][0], s_cw, m, ts[j] & tl_cw);
1451 low[2 * j + 1] = CorrectChild(out[j][1], s_cw, m, ts[j] & tr_cw);
1452 int m4 = -ts[j + 4];
1453 high[2 * j] = CorrectChild(out[j + 4][0], s_cw, m4, ts[j + 4] & tl_cw);
1454 high[2 * j + 1] = CorrectChild(out[j + 4][1], s_cw, m4, ts[j + 4] & tr_cw);
1455 }
1456
1457 size_t half = size_t{8} << (in_bits - i - 1);
1458 EvalBatched8Group(b, low, cws, ys, l, i + 1);
1459 EvalBatched8Group(b, high, cws, ys, l + half, i + 1);
1460}
