146 // EvalAll() shape: a scalar prefix of kBatchPrefix levels grows the frontier to
147 // 8 nodes, detail::GenBatch() expands 8 parents per level (pipelined through
148 // Gen8() when the PRG offers a batched entry), and a kBatchTail-level tail
149 // converts each 8-subtree group of 8 << kBatchTail leaves in local memory.
150 static constexpr int kBatchPrefix = 3;

154 __host__ __device__ void Gen(Cw cws[], const int4 s0s[2], In a, int4 b_buf) {
155 int4 s0 = s0s[0];
156 s0 = util::SetLsb(s0, false);
157 int4 s1 = s0s[1];
158 s1 = util::SetLsb(s1, false);
159 bool t0 = false;
160 bool t1 = true;
161 b_buf = util::SetLsb(b_buf, false);
162
163 for (int i = 0; i < in_bits; ++i) {
164 auto [s0l, s0r] = prg.Gen(s0);
165 auto [s1l, s1r] = prg.Gen(s1);
166
167 bool t0l = util::GetLsb(s0l);
168 s0l = util::SetLsb(s0l, false);
169 bool t0r = util::GetLsb(s0r);
170 s0r = util::SetLsb(s0r, false);
171 bool t1l = util::GetLsb(s1l);
172 s1l = util::SetLsb(s1l, false);
173 bool t1r = util::GetLsb(s1r);
174 s1r = util::SetLsb(s1r, false);
175
176 bool a_bit = (a >> (in_bits - 1 - i)) & 1;
177
178 int4 s_cw;
179 if (!a_bit) s_cw = util::Xor(s0r, s1r);
180 else s_cw = util::Xor(s0l, s1l);
181
182 bool tl_cw = t0l ^ t1l ^ a_bit ^ 1;
183 bool tr_cw = t0r ^ t1r ^ a_bit;
184
185 if (!a_bit) {
186 s0 = s0l;
187 if (t0) s0 = util::Xor(s0, s_cw);
188 s1 = s1l;
189 if (t1) s1 = util::Xor(s1, s_cw);
190
191 if (t0) t0 = t0l ^ tl_cw;
192 else t0 = t0l;
193 if (t1) t1 = t1l ^ tl_cw;
194 else t1 = t1l;
195 } else {
196 s0 = s0r;
197 if (t0) s0 = util::Xor(s0, s_cw);
198 s1 = s1r;
199 if (t1) s1 = util::Xor(s1, s_cw);
200
201 if (t0) t0 = t0r ^ tr_cw;
202 else t0 = t0r;
203 if (t1) t1 = t1r ^ tr_cw;
204 else t1 = t1r;
205 }
206
207 s_cw = util::SetLsb(s_cw, tl_cw);
208 // Write both 16B halves so the full 32B slot is covered by unmasked stores.
209 // {s_cw, tr_cw} alone leaves the padding undefined: NVCC emits a masked
210 // 256-bit store, and the partial-sector write forces a write-allocate
211 // round trip per level on GPU (measured ~2x Gen time).
212 int4 *cw_slot = reinterpret_cast<int4 *>(&cws[i]);
213 cw_slot[0] = s_cw;
214 cw_slot[1] = make_int4(tr_cw, 0, 0, 0);
215 }
216
217 auto v_cw_np1 = Group::From(b_buf) + (-Group::From(s0)) + Group::From(s1);
218 if (t1) v_cw_np1 = -v_cw_np1;
219 cws[in_bits] = {v_cw_np1.Into(), false};
220 }

222 // Seed + correction-words form used by the internal recursion and tests.
223 __host__ __device__ int4 Eval(bool b, int4 s0, const Cw cws[], In x) {
224 int4 st = Walk(b, s0, cws, x, [](bool, bool) {});
225 bool t = util::GetLsb(st);
226 int4 s = util::SetLsb(st, false);
227
228 auto y = Group::From(s);
229 int4 v_cw_np1 = cws[in_bits].s;
230 assert((v_cw_np1.w & 1) == 0);
231 if (t) y = y + Group::From(v_cw_np1);
232 if (b) y = -y;
233
234 return y.Into();
235 }

237 void EvalAll(bool b, int4 s0, const Cw cws[], int4 ys[]) {
238 assert(in_bits < sizeof(size_t) * 8);
239
240 if constexpr (in_bits < kBatchPrefix + kBatchTail) {
241 // Domains too small to fill an 8-wide frontier evaluate per input; the
242 // point walk is cheaper than the batched machinery at this size.
243 constexpr size_t n = size_t{1} << in_bits;
244 util::ParallelFor(par_depth, n, [&](size_t x) { ys[x] = Eval(b, s0, cws, static_cast<In>(x)); });
245 return;
246 }
247
248 int4 st = s0;
249 st = util::SetLsb(st, b);
250
251 int par_depth_ = util::ResolveParDepth(par_depth, in_bits);
252
253 int split_depth = par_depth_;
254 if (split_depth > in_bits - kBatchPrefix - kBatchTail) split_depth = in_bits - kBatchPrefix - kBatchTail;
255 if (split_depth > 0) {
256 // Split at a frontier of 2 ** split_depth subtree roots instead of per-node tasks,
257 // so the sequential batched walk stays the fast default path.
258 size_t num_tasks = size_t{1} << split_depth;
259 std::vector<int4> frontier = BuildParallelFrontier(st, cws, split_depth);
260 size_t span = (size_t{1} << in_bits) / num_tasks;
261 util::ParallelFor(par_depth_, num_tasks, [&](size_t task) {
262 EvalBatched8(b, frontier[task], cws, ys, task * span, split_depth);
263 });
264 } else {
265 EvalBatched8(b, st, cws, ys, 0, 0);
266 }
267 }

272 // Walks x's path from the root and returns the leaf seed with its control bit in the LSB.
273 // visit(t, x_bit) sees the control bit of every expanded node and the branch x takes from it.
274 template <typename Visit>
275 __host__ __device__ int4 Walk(bool b, int4 s0, const Cw cws[], In x, Visit &&visit) {
276 int4 s = s0;
277 s = util::SetLsb(s, false);
278 bool t = b;
279
280 for (int i = 0; i < in_bits; ++i) {
281 bool x_bit = (x >> (in_bits - 1 - i)) & 1;
282 visit(t, x_bit);
283
284 Cw cw = cws[i];
285 int4 s_cw = cw.s;
286 bool tl_cw = util::GetLsb(s_cw);
287 s_cw = util::SetLsb(s_cw, false);
288 bool tr_cw = cw.tr;
289
290 auto [sl, sr] = prg.Gen(s);
291
292 bool tl = util::GetLsb(sl);
293 sl = util::SetLsb(sl, false);
294 bool tr = util::GetLsb(sr);
295 sr = util::SetLsb(sr, false);
296
297 if (t) {
298 sl = util::Xor(sl, s_cw);
299 sr = util::Xor(sr, s_cw);
300 tl = tl ^ tl_cw;
301 tr = tr ^ tr_cw;
302 }
303
304 if (!x_bit) {
305 s = sl;
306 t = tl;
307 } else {
308 s = sr;
309 t = tr;
310 }
311 }
312
313 return util::SetLsb(s, t);
314 }

316 // GCC 12.4.0 can discard these writes before the OpenMP walk after IPA
317 // effect analysis. Keep this boundary opaque on GCC 12 releases before 12.5.
318 // GCC PR115033 describes a related closure escape-analysis defect.
319#if defined(__GNUC__) && !defined(__clang__) && __GNUC__ == 12 && __GNUC_MINOR__ <= 4
320 __attribute__((noipa))
321#endif
322 std::vector<int4> BuildParallelFrontier(int4 st, const Cw cws[], int split_depth) {
323 std::vector<int4> frontier(size_t{1} << split_depth);
324 frontier[0] = st;
325 for (int lv = 0; lv < split_depth; ++lv) ExpandFrontier(frontier.data(), size_t{1} << lv, cws, lv);
326 return frontier;
327 }

329 // Expands w scalar frontier nodes (t in LSB) by one level with cws[lv].
330 // The children overwrite the frontier in place: iteration is in reverse so
331 // the writes at 2j, 2j+1 never clobber nodes still to be read.
332 void ExpandFrontier(int4 frontier[], size_t w, const Cw cws[], int lv) {
333 int4 s_cw = util::SetLsb(cws[lv].s, false);
334 bool tl_cw = util::GetLsb(cws[lv].s);
335 bool tr_cw = cws[lv].tr;
336
337 for (size_t j = w; j-- > 0;) {
338 bool t = util::GetLsb(frontier[j]);
339 int4 s = util::SetLsb(frontier[j], false);
340
341 auto [sl, sr] = prg.Gen(s);
342
343 bool tl = util::GetLsb(sl);
344 sl = util::SetLsb(sl, false);
345 bool tr = util::GetLsb(sr);
346 sr = util::SetLsb(sr, false);
347
348 if (t) {
349 sl = util::Xor(sl, s_cw);
350 sr = util::Xor(sr, s_cw);
351 tl = tl ^ tl_cw;
352 tr = tr ^ tr_cw;
353 }
354
355 frontier[2 * j] = util::SetLsb(sl, tl);
356 frontier[2 * j + 1] = util::SetLsb(sr, tr);
357 }
358 }

360 // Applies the level correction to one expanded child branchlessly.
361 // m is 0 or -1 from the parent control bit t, and t_cw is the cw control bit
362 // masked by t. The cw seed is clamped, so the XOR leaves the child LSB free
363 // for the packed control bit.
364 static int4 CorrectChild(int4 child, int4 s_cw, int m, int t_cw) {
365 int4 res = {child.x ^ (s_cw.x & m), child.y ^ (s_cw.y & m), child.z ^ (s_cw.z & m), child.w ^ (s_cw.w & m)};
366 res.w = (res.w & ~1) | ((child.w ^ t_cw) & 1);
367 return res;
368 }

370 // Full-domain evaluation of the subtree rooted at st (t in LSB) at level i through an
371 // 8-wide batched frontier: a kBatchPrefix-level scalar prefix, batched expansions, and
372 // a kBatchTail-level tail converted in local memory.
373 // Requires in_bits - i >= kBatchPrefix + kBatchTail.
374 void EvalBatched8(bool b, int4 st, const Cw cws[], int4 ys[], size_t l, int i) {
375 int4 frontier[8];
376 frontier[0] = st;
377 for (int lv = i; lv < i + kBatchPrefix; ++lv) ExpandFrontier(frontier, size_t{1} << (lv - i), cws, lv);
378 EvalBatched8Group(b, frontier, cws, ys, l, i + kBatchPrefix);
379 }

381 // Evaluates a group of 8 consecutive frontier nodes (t in LSB) at level i, covering the
382 // leaves [l, l + 8 << (in_bits - i)) in order. Requires in_bits - i >= kBatchTail.
383 void EvalBatched8Group(bool b, const int4 (&frontier)[8], const Cw cws[], int4 ys[], size_t l, int i) {
384 if (i == in_bits - kBatchTail) {
385 // Expand the group to its 8 << kBatchTail leaves in local memory, then convert.
386 int4 bufs[2][8 << kBatchTail];
387 int4 *cur = bufs[0];
388 int4 *next = bufs[1];
389 for (int j = 0; j < 8; ++j) cur[j] = frontier[j];
390 int w = 8;
391 for (int lv = i; lv < in_bits; ++lv) {
392 int4 s_cw = util::SetLsb(cws[lv].s, false);
393 int tl_cw = util::GetLsb(cws[lv].s);
394 int tr_cw = cws[lv].tr;
395 for (int g = 0; g < w / 8; ++g) {
396 cuda::std::array<int4, 8> seeds;
397 int ts[8];
398 for (int j = 0; j < 8; ++j) {
399 ts[j] = static_cast<int>(util::GetLsb(cur[8 * g + j]));
400 seeds[j] = util::SetLsb(cur[8 * g + j], false);
401 }
402 auto out = detail::GenBatch<2>(prg, seeds);
403 for (int j = 0; j < 8; ++j) {
404 int m = -ts[j];
405 next[16 * g + 2 * j] = CorrectChild(out[j][0], s_cw, m, ts[j] & tl_cw);
406 next[16 * g + 2 * j + 1] = CorrectChild(out[j][1], s_cw, m, ts[j] & tr_cw);
407 }
408 }
409 int4 *swap = cur;
410 cur = next;
411 next = swap;
412 w *= 2;
413 }
414
415 int4 v_cw_np1 = cws[in_bits].s;
416 assert((v_cw_np1.w & 1) == 0);
417 Group v_cw = Group::From(v_cw_np1);
418 for (int j = 0; j < w; ++j) {
419 bool t = util::GetLsb(cur[j]);
420 int4 s = util::SetLsb(cur[j], false);
421 auto y = Group::From(s);
422 if (t) y = y + v_cw;
423 if (b) y = -y;
424 ys[l + j] = y.Into();
425 }
426 return;
427 }
428
429 int4 s_cw = util::SetLsb(cws[i].s, false);
430 int tl_cw = util::GetLsb(cws[i].s);
431 int tr_cw = cws[i].tr;
432
433 cuda::std::array<int4, 8> seeds;
434 int ts[8];
435 for (int j = 0; j < 8; ++j) {
436 ts[j] = static_cast<int>(util::GetLsb(frontier[j]));
437 seeds[j] = util::SetLsb(frontier[j], false);
438 }
439 auto out = detail::GenBatch<2>(prg, seeds);
440 // The 16 children interleave in leaf order: node j's children land at 2j, 2j+1 of the
441 // next level. De-interleave them into two consecutive groups of 8.
442 int4 low[8];
443 int4 high[8];
444 for (int j = 0; j < 4; ++j) {
445 int m = -ts[j];
446 low[2 * j] = CorrectChild(out[j][0], s_cw, m, ts[j] & tl_cw);
447 low[2 * j + 1] = CorrectChild(out[j][1], s_cw, m, ts[j] & tr_cw);
448 int m4 = -ts[j + 4];
449 high[2 * j] = CorrectChild(out[j + 4][0], s_cw, m4, ts[j + 4] & tl_cw);
450 high[2 * j + 1] = CorrectChild(out[j + 4][1], s_cw, m4, ts[j + 4] & tr_cw);
451 }
452
453 size_t half = size_t{8} << (in_bits - i - 1);
454 EvalBatched8Group(b, low, cws, ys, l, i + 1);
455 EvalBatched8Group(b, high, cws, ys, l + half, i + 1);
456 }
