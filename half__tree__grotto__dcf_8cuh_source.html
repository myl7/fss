114 static __host__ __device__ bool LeafParity(int4 block, In x) {
115 int lane = static_cast<int>(x) & (DpfType::kLanes - 1);
116 bool right = ((x >> (in_bits - kDepth)) & 1) != 0;
117 int word = lane >> 5;
118 unsigned int partial = 0xffffffffu >> (31 - (lane & 31));
119 unsigned int masks[4];
120 for (int j = 0; j < 4; ++j) {
121 unsigned int upto = j < word ? 0xffffffffu : (j == word ? partial : 0u);
122 masks[j] = right ? ~upto : upto;
123 }
124 unsigned int acc = (static_cast<unsigned int>(block.x) & masks[0]) ^
125 (static_cast<unsigned int>(block.y) & masks[1]) ^ (static_cast<unsigned int>(block.z) & masks[2]) ^
126 (static_cast<unsigned int>(block.w) & masks[3]);
127 return (cuda::std::popcount(acc) & 1) != 0;
128 }
129
130private:
131 __host__ __device__ DpfType MakeDpf() const {
132 return DpfType{prg, hash_key};
133 }
