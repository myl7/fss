32// below is reachable: naming Uint<__uint128_t, ...> checks this requires-

69 __host__ __device__ static Uint From(int4 buf) {
70 assert((buf.w & 1) == 0);
71
72 T value = 0;
73 if constexpr (sizeof(T) < 4) value = buf.x & ((1 << 8 * sizeof(T)) - 1);
74 // Cast to unsigned int first to prevent sign extension when promoting to larger types
75 else if constexpr (sizeof(T) == 4) value = static_cast<unsigned int>(buf.x);
76 else if constexpr (sizeof(T) == 8)
77 value = static_cast<T>(static_cast<unsigned int>(buf.x)) | static_cast<T>(static_cast<unsigned int>(buf.y)) << 32;
78 else __builtin_unreachable();
79
80 if constexpr (mod > 0) value %= mod;
81

83 }

89 else if constexpr (sizeof(T) == 8) {
