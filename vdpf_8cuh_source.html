65// Split correction-word view over the caller's arrays. Layer i is read from
66// cws[i*stride] and the right-control correction is bit i of packed_tr. The
67// terminal is rebuilt as a plain Cw with no control bit, so bits of packed_tr at
68// or above in_bits are never read. The layer LSB keeps the left-control
69// correction exactly as the Cw layout does. operator[] recovers the Cw the shared
70// evaluation consumes, so its body stays identical on the Cw and split paths.
71template <typename Cw, typename In, int in_bits>
72struct VdpfCwsView {
73 const int4 *s;
74 In packed_tr;
75 size_t stride;
76
77 __host__ __device__ inline Cw operator[](int i) const {
78 const int4 s_cw = s[static_cast<size_t>(i) * stride];
79 if (i >= in_bits) return Cw{s_cw, false};
80 return Cw{s_cw, ((packed_tr >> i) & 1) != 0};
81 }
82};

679 struct AosWriter {

682 __host__ __device__ void Write(int i, int4 s_cw, bool tr_cw) const {
683 // Both 16B halves written so the full 32B slot gets unmasked stores.
684 // {s_cw, tr_cw} alone leaves padding undefined: NVCC emits a masked
685 // 256-bit store, and the partial-sector write forces a write-allocate

693 cws[in_bits].s = terminal;

786template <int in_bits, int threads_per_block, typename Group, typename Prg, typename XorHash,
787 typename Hash, typename In>
788__device__ void VdpfRangeMerkleLeavesWork(bool b, In first,
789 int subtree_depth, int frontier_depth, int block_root_depth, int4 s0, const int4 *cws,
790 In packed_tr, size_t stride, const int4 *cs,
791 int4 *ys, cuda::std::array<int4, 2> *leaves, Prg prg,
792 XorHash xor_hash, Hash hash) {
793 using MathScheme = Dpf<in_bits, Group, Prg, In>;
794 const int root_depth = in_bits - subtree_depth;
795 const int block_subtree_depth = frontier_depth - block_root_depth;
796 const int tail_depth = subtree_depth - frontier_depth;
797 __shared__ int4 frontier[threads_per_block];
798 if (threadIdx.x == 0) {
799 int4 st = util::SetLsb(s0, b);
800 for (int level = 0; level < root_depth; ++level) {
801 const bool right = (first >> (in_bits - level - 1)) & 1;
802 st = fss::detail::SchemeAccess::DpfPathStep<MathScheme>(
803 prg, st, VdpfCwSeed(cws, stride, level), VdpfCwRight(packed_tr, level), right);
804 }
805 for (int level = 0; level < block_root_depth; ++level) {
806 const bool right = (blockIdx.x >> (block_root_depth - level - 1)) & 1;
807 st = fss::detail::SchemeAccess::DpfPathStep<MathScheme>(prg, st,
808 VdpfCwSeed(cws, stride, root_depth + level),
809 VdpfCwRight(packed_tr, root_depth + level), right);
810 }
811 frontier[0] = st;
812 }
813 __syncthreads();
814 for (int level = 0; level < block_subtree_depth; ++level) {
815 const bool active = threadIdx.x < (1u << (level + 1));
816 int4 st = {0, 0, 0, 0};
817 if (active) st = frontier[threadIdx.x >> 1];
818 __syncthreads();
819 if (active) {
820 int4 left, right;
821 const int cw_index = root_depth + block_root_depth + level;
822 fss::detail::SchemeAccess::DpfExpandNode<MathScheme>(prg, st,
823 VdpfCwSeed(cws, stride, cw_index), VdpfCwRight(packed_tr, cw_index), left, right);
824 frontier[threadIdx.x] = (threadIdx.x & 1) ? right : left;
825 }
826 __syncthreads();
827 }
828 int4 pending[8];
829 int4 node = frontier[threadIdx.x];
830 const int frontier_level = root_depth + frontier_depth;
831 int level = frontier_level;
832 const int count = 1 << tail_depth;
833 const size_t thread_index = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
834 const size_t offset = thread_index << tail_depth;
835 const auto cw = Group::From(cws[static_cast<size_t>(in_bits) * stride]);
836 for (int i = 0; i < count; ++i) {
837 while (level < in_bits) {
838 int4 left, right;
839 fss::detail::SchemeAccess::DpfExpandNode<MathScheme>(prg, node,
840 VdpfCwSeed(cws, stride, level), VdpfCwRight(packed_tr, level), left, right);
841 pending[level - frontier_level] = right;
842 node = left;
843 ++level;
844 }
845 const bool t = util::GetLsb(node);
846 const int4 seed = util::SetLsb(node, false);
847 auto value = Group::From(seed);
848 if (t) value = value + cw;
849 if (b) value = -value;
850 ys[offset + i] = value.Into();
851 const In x = static_cast<In>(first + offset + i);
852 auto corrected = xor_hash.Hash(cuda::std::tuple<int4, const int4>{fss::detail::Pack(x), seed});
853 if (t) corrected = util::Xor(cuda::std::span<const int4, 4>(corrected),
854 cuda::std::span<const int4, 4>(cs, 4));
855 leaves[offset + i] = hash.Hash(cuda::std::span<const int4, 4>(corrected));
856 if (i + 1 < count) {
857 int next = i + 1;
858 level = in_bits - 1;
859 while ((next & 1) == 0) {
860 --level;
861 next >>= 1;
862 }
863 node = pending[level - frontier_level];
864 ++level;
865 }
866 }

966template <int in_bits, int frontier_depth, int block_root_depth, typename Group, typename Prg,
967 typename XorHash, typename Hash, typename In>
968__device__ void VdpfMerkleFrontierWork(bool b, int4 s0, const int4 *cws,
969 In packed_tr, size_t stride, const int4 *cs,
970 int4 *ys, cuda::std::array<int4, 2> *roots, Prg prg,
971 XorHash xor_hash, Hash hash) {
972 using MathScheme = Dpf<in_bits, Group, Prg, In>;
973 constexpr int kTailDepth = in_bits - frontier_depth;
974 constexpr int kBlockSubtreeDepth = frontier_depth - block_root_depth;
975 const int thread_index = blockIdx.x * blockDim.x + threadIdx.x;
976 if (thread_index >= (1 << frontier_depth)) return;
977
978 // Phase 0: block root walk (top block_root_depth levels) by thread 0.
979 __shared__ int4 shared_frontier[1 << kBlockSubtreeDepth];
980 if (threadIdx.x == 0) {
981 int4 st = util::SetLsb(s0, b);
982#pragma unroll
983 for (int i = 0; i < block_root_depth; ++i) {
984 bool x_bit = (blockIdx.x >> (block_root_depth - 1 - i)) & 1;
985 {
986 auto node_prg = prg;
987 st = fss::detail::SchemeAccess::DpfPathStep<MathScheme>(
988 node_prg, st, VdpfCwSeed(cws, stride, i), VdpfCwRight(packed_tr, i), x_bit);
989 }
990 }
991 shared_frontier[0] = st;
992 }
993 __syncthreads();
994
995 // Phase 1: breadth-parallel expansion of the block's subtree, levels block_root_depth..frontier_depth.
996#pragma unroll
997 for (int i = 0; i < kBlockSubtreeDepth; ++i) {
998 const bool active = threadIdx.x < (1 << (i + 1));
999 int4 n = {0, 0, 0, 0};
1000 if (active) n = shared_frontier[threadIdx.x >> 1];
1001 // All parent reads must finish before any child overwrites the frontier.
1002 __syncthreads();
1003 if (active) {
1004 int4 left, right;
1005 {
1006 auto node_prg = prg;
1007 fss::detail::SchemeAccess::DpfExpandNode<MathScheme>(node_prg, n,
1008 VdpfCwSeed(cws, stride, block_root_depth + i),
1009 VdpfCwRight(packed_tr, block_root_depth + i), left, right);
1010 }
1011 shared_frontier[threadIdx.x] = (threadIdx.x & 1) ? right : left;
1012 }
1013 __syncthreads();
1014 }
1015
1016 // Phase 2: expand the subtree below depth frontier_depth, convert each leaf to the output
1017 // share and the corrected verification hash, and fold the leaf hashes into
1018 // the thread's Merkle subtree on the fly.
1019 int4 nodes[1 << kTailDepth];
1020 nodes[0] = shared_frontier[threadIdx.x];
1021 int count = 1;
1022#pragma unroll
1023 for (int lev = frontier_depth; lev < in_bits; ++lev) {
1024#pragma unroll
1025 for (int k = count - 1; k >= 0; --k) {
1026 int4 left, right;
1027 {
1028 auto node_prg = prg;
1029 fss::detail::SchemeAccess::DpfExpandNode<MathScheme>(node_prg, nodes[k],
1030 VdpfCwSeed(cws, stride, lev), VdpfCwRight(packed_tr, lev), left, right);
1031 }
1032 nodes[2 * k] = left;
1033 nodes[2 * k + 1] = right;
1034 }
1035 count *= 2;
1036 }
1037
1038 auto ocw_group = Group::From(cws[static_cast<size_t>(in_bits) * stride]);
1039 const int output_offset = thread_index << kTailDepth;
1040 cuda::std::array<int4, 2> stack[kTailDepth + 1];
1041#pragma unroll
1042 for (int k = 0; k < count; ++k) {
1043 int4 st_cur = nodes[k];
1044 bool t_cur = util::GetLsb(st_cur);
1045 int4 s_cur = util::SetLsb(st_cur, false);
1046
1047 // Output share
1048 auto g = Group::From(s_cur);
1049 if (t_cur) g = g + ocw_group;
1050 if (b) g = -g;
1051 ys[output_offset + k] = g.Into();
1052
1053 // Leaf: the corrected verification hash of the point
1054 int4 x_buf = fss::detail::Pack(static_cast<In>(output_offset + k));
1055 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, s_cur});
1056 if (t_cur) {
1057 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(cs, 4));
1058 }
1059 int4 leaf_msg[4] = {pi_tilde[0], pi_tilde[1], pi_tilde[2], pi_tilde[3]};
1060 cuda::std::array<int4, 2> node = hash.Hash(cuda::std::span<const int4, 4>(leaf_msg));
1061
1062 // Stream the leaf into the thread's subtree: combine with the pending
1063 // sibling stack entries while the leaf index has trailing one bits.
1064 int rest = k;
1065 int lvl = 0;
1066#pragma unroll
1067 while (rest != 0 && (rest & 1) == 1) {
1068 int4 fold_msg[4] = {stack[lvl][0], stack[lvl][1], node[0], node[1]};
1069 node = hash.Hash(cuda::std::span<const int4, 4>(fold_msg));
1070 rest >>= 1;
1071 ++lvl;
1072 }
1073 stack[lvl] = node;
1074 }
1075
1076 // Block fold of the threads' subtree roots, in thread order: adjacent pairs
1077 // at each stride, slot idx combining with idx + stride.
1078 __shared__ cuda::std::array<int4, 2> s_node[1 << kBlockSubtreeDepth];
1079 s_node[threadIdx.x] = stack[kTailDepth];
1080 __syncthreads();
1081#pragma unroll
1082 for (int stride = 1; stride < (1 << kBlockSubtreeDepth); stride <<= 1) {
1083 int idx = 2 * stride * threadIdx.x;
1084 if (idx < (1 << kBlockSubtreeDepth)) {
1085 int4 fold_msg[4] = {s_node[idx][0], s_node[idx][1], s_node[idx + stride][0], s_node[idx + stride][1]};
1086 s_node[idx] = hash.Hash(cuda::std::span<const int4, 4>(fold_msg));
1087 }
1088 __syncthreads();
1089 }
1090 if (threadIdx.x == 0) roots[blockIdx.x] = s_node[0];
1091}

1139// Point-evaluate in_bits-level VDPF keys, one thread per key. Same seed walk as
1140// the DPF kernel plus the per-point verification hash; the walk carries no hash
1141// cost, and the tail applies the verification hash and, when t, folds in cs. cws
1142// holds dense level-major correction words, the layout the split form of Gen()
1143// writes: layer i of key k is cws[i*nkeys + k] and the terminal is
1144// cws[in_bits*nkeys + k]. tr[k] carries key k's packed right-control bits, bit i
1145// for layer i. cs stays per-key: it is read once per evaluation, not once per level.
1146template <int in_bits, int unroll, bool b, typename Group, typename Prg, typename XorHash,
1147 typename In, typename Query>
1148__global__ void VdpfEvalPointKernel(int4 *ys, cuda::std::array<int4, 4> *pi, const int4 *seeds,
1149 const int4 *cws, const In *tr, const cuda::std::array<int4, 4> *cs, const Query *xs,
1150 int nkeys, Prg prg, XorHash xor_hash) {
1151 static_assert(in_bits <= sizeof(In) * 8, "In too narrow for in_bits");
1152 int tid = blockIdx.x * blockDim.x + threadIdx.x;
1153 if (tid >= nkeys) return;
1154 int4 s = fss::util::SetLsb(seeds[tid], false);
1155 bool t = b;
1156 In x = static_cast<In>(xs[tid]);
1157 const In packed_tr = tr[tid];
1158
1159#pragma unroll unroll
1160 for (int i = 0; i < in_bits; ++i) {
1161 int4 s_cw = cws[static_cast<size_t>(i) * nkeys + tid];
1162 bool tl_cw = fss::util::GetLsb(s_cw);
1163 s_cw = fss::util::SetLsb(s_cw, false);
1164 bool tr_cw = ((packed_tr >> i) & 1) != 0;
1165
1166 auto g = prg.Gen(s);
1167 int4 sl = g[0], sr = g[1];
1168 bool tl = fss::util::GetLsb(sl);
1169 sl = fss::util::SetLsb(sl, false);
1170 bool tr = fss::util::GetLsb(sr);
1171 sr = fss::util::SetLsb(sr, false);
1172
1173 if (t) {
1174 sl = fss::util::Xor(sl, s_cw);
1175 sr = fss::util::Xor(sr, s_cw);
1176 tl = tl ^ tl_cw;
1177 tr = tr ^ tr_cw;
1178 }
1179
1180 bool x_bit = GetBit(x, in_bits - 1 - i);
1181 if (x_bit) {
1182 s = sr;
1183 t = tr;
1184 } else {
1185 s = sl;
1186 t = tl;
1187 }
1188 }
1189
1190 auto y = Group::From(s);
1191 if (t) y = y + Group::From(cws[static_cast<size_t>(in_bits) * nkeys + tid]);
1192 if constexpr (b) y = -y;
1193 ys[tid] = y.Into();
1194
1195 int4 x_buf = fss::detail::Pack(x);
1196 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, s});
1197 if (t) {
1198 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(cs[tid]));
1199 }
1200#pragma unroll
1201 for (int j = 0; j < 4; ++j) pi[tid][j] = pi_tilde[j];
1202}

1454 bool b, int4 s0, Cws cws, cuda::std::span<const int4, 4> cs, In x, int4 &y) {
1455 int4 s = s0;
1456 s = util::SetLsb(s, false);
1457 bool t = b;
1458
1459 for (int i = 0; i < in_bits; ++i) {
1460 Cw cw = cws[i];
1461 int4 s_cw = cw.s;
1462 bool tl_cw = util::GetLsb(s_cw);
1463 s_cw = util::SetLsb(s_cw, false);
1464 bool tr_cw = cw.tr;
1465
1466 bool x_bit = (x >> (in_bits - 1 - i)) & 1;
1467 auto [child] = detail::GenHalf<2>(prg_, s, x_bit);
1468 bool child_t = util::GetLsb(child);
1469 child = util::SetLsb(child, false);
1470 if (t) {
1471 child = util::Xor(child, s_cw);
1472 child_t = child_t ^ (x_bit ? tr_cw : tl_cw);
1473 }
1474 s = child;
1475 t = child_t;
1476 }
1477
1478 // Output share
1479 auto g = Group::From(s);
1480 int4 ocw = cws[in_bits].s;
1481 util::RequireArg((ocw.w & 1) == 0,
1482 "Vdpf::Eval: the terminal group value must have a zero LSB");

1484 if (b) g = -g;
1485
1486 // Corrected verification hash
1487 int4 x_buf = fss::detail::Pack(x);
1488
1489 auto pi_tilde = xor_hash_.Hash(cuda::std::tuple<int4, const int4>{x_buf, s});
1490 if (t) {
1491 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(cs));
1492 }
1493 y = g.Into();
1494 return pi_tilde;
1495}

1561 cuda::std::span<int4> ys, cuda::std::span<int4, 4> pi, int par_depth) {
1562 assert(par_depth >= 0);
1563 In max_input = static_cast<In>(~In{0});
1564 if constexpr (in_bits < sizeof(In) * 8) {
1565 max_input = static_cast<In>((In{1} << in_bits) - 1);
1566 }
1567 if (first > last)
1568 throw std::invalid_argument("Vdpf::EvalRange: first must not exceed last");
1569 if (last > max_input)
1570 throw std::invalid_argument("Vdpf::EvalRange: last must stay inside the input domain");
1571 const __uint128_t distance = static_cast<__uint128_t>(last) - first;
1572 if (distance >= static_cast<size_t>(-1) / sizeof(int4))
1573 throw std::invalid_argument("Vdpf::EvalRange: the interval is too large for size_t");

1576 throw std::invalid_argument("Vdpf::EvalRange: ys must hold last-first+1 entries");

1579 // traversal, so a Debug violation aborts on the calling thread; Release
1580 // builds do not check it.

1592 } else {
1593 const int threads = fss::detail::ResolveParThreads(par_depth_, num_inputs);

1595#pragma omp parallel num_threads(threads)
1596#pragma omp single
1597 EvalPrunedTree(st, cws, ys, In{0}, max_input, 0, par_depth_, first, last);
1598 }

1600 pi[0] = cs[0];
1601 pi[1] = cs[1];
1602 pi[2] = cs[2];
1603 pi[3] = cs[3];

1606 const In x = static_cast<In>(first + offset);
1607 int4 sj = ys[offset];
1608 bool tj = util::GetLsb(sj);
1609 sj = util::SetLsb(sj, false);
1610
1611 // Output share
1612 auto g = Group::From(sj);
1613 if (tj) g = g + ocw_group;
1614 if (b) g = -g;
1615 ys[offset] = g.Into();

1617 // Proof accumulation
1618 int4 x_buf = fss::detail::Pack(x);
1619
1620 auto pi_tilde = xor_hash_.Hash(cuda::std::tuple<int4, const int4>{x_buf, sj});
1621 if (tj) {
1622 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(cs));
1623 }
1624
1625 cuda::std::array<int4, 4> h_input =
1626 util::Xor(cuda::std::span<const int4, 4>(pi), cuda::std::span<const int4, 4>(pi_tilde));
1627 auto h_out = hash_.Hash(cuda::std::span<const int4, 4>(h_input));
1628 pi[0] = util::Xor(pi[0], h_out[0]);
1629 pi[1] = util::Xor(pi[1], h_out[1]);
1630 }
1631}

1637 cuda::std::span<const cuda::std::array<int4, 4>> pi_tildes, cuda::std::span<const int4, 4> cs,
1638 cuda::std::span<int4, 4> pi, int par_depth) {
1639 assert(par_depth >= 0);
1640 size_t count = pi_tildes.size();
1641 if (count < 1)
1642 throw std::invalid_argument("Vdpf::ProveMerkle: pi_tildes must not be empty");
1643
1644 std::vector<cuda::std::array<int4, 2>> nodes(count);
1645 if constexpr (HasHash4<Hash>) {
1646 const size_t groups = count / 4;
1647 fss::detail::ParallelFor(par_depth, groups, [&](size_t g) {
1648 auto out =
1649 hash_.Hash4(cuda::std::span<const cuda::std::array<int4, 4>, 4>(pi_tildes.data() + 4 * g, 4));
1650 for (size_t j = 0; j < 4; ++j) nodes[4 * g + j] = out[j];
1651 });

1653 nodes[i] = hash_.Hash(cuda::std::span<const int4, 4>(pi_tildes[i]));
1654 }
1655 } else {
1656 fss::detail::ParallelFor(par_depth, count, [&](size_t i) {
1657 nodes[i] = hash_.Hash(cuda::std::span<const int4, 4>(pi_tildes[i]));
1658 });
1659 }
1660
1661 while (count > 1) {
1662 const size_t next_count = count / 2 + count % 2;
1663 std::vector<cuda::std::array<int4, 2>> next(next_count);
1664 if constexpr (HasHash4<Hash>) {
1665 const size_t pairs = count / 2;
1666 const size_t groups = pairs / 4;
1667 fss::detail::ParallelFor(par_depth, groups, [&](size_t g) {
1668 cuda::std::array<cuda::std::array<int4, 4>, 4> msgs;
1669 for (size_t j = 0; j < 4; ++j) {
1670 const auto *p = nodes.data() + 8 * g + 2 * j;
1671 msgs[j] = {p[0][0], p[0][1], p[1][0], p[1][1]};
1672 }
1673 auto out = hash_.Hash4(cuda::std::span<const cuda::std::array<int4, 4>, 4>(msgs));
1674 for (size_t j = 0; j < 4; ++j) next[4 * g + j] = out[j];
1675 });
1676 // Remaining complete pairs; an odd final node is carried by the helper.
1677 for (size_t i = 4 * groups; i < next_count; ++i) {
1678 next[i] = detail::VdpfMerkleFoldNode(hash_, nodes.data(), count, i);
1679 }
1680 } else {
1681 fss::detail::ParallelFor(par_depth, next_count, [&](size_t i) {
1682 next[i] = detail::VdpfMerkleFoldNode(hash_, nodes.data(), count, i);
1683 });
1684 }
1685 nodes = std::move(next);
1686 count = nodes.size();
1687 }
1688
1689 pi[0] = nodes[0][0];
1690 pi[1] = nodes[0][1];
1691 pi[2] = cs[2];
1692 pi[3] = cs[3];
1693}

1885 bool t1 = true;

1898 bool t1r = util::GetLsb(s1r);
1899 s1r = util::SetLsb(s1r, false);

1902
1903 int4 s_cw;

1905 else s_cw = util::Xor(s0l, s1l);
1906
1907 bool tl_cw = t0l ^ t1l ^ a_bit ^ 1;
1908 bool tr_cw = t0r ^ t1r ^ a_bit;
1909
1910 if (!a_bit) {
1911 s0 = s0l;
1912 if (t0) s0 = util::Xor(s0, s_cw);
1913 s1 = s1l;
1914 if (t1) s1 = util::Xor(s1, s_cw);
1915
1916 if (t0) t0 = t0l ^ tl_cw;
1917 else t0 = t0l;
1918 if (t1) t1 = t1l ^ tl_cw;
1919 else t1 = t1l;
1920 } else {
1921 s0 = s0r;
1922 if (t0) s0 = util::Xor(s0, s_cw);
1923 s1 = s1r;
1924 if (t1) s1 = util::Xor(s1, s_cw);
1925
1926 if (t0) t0 = t0r ^ tr_cw;
1927 else t0 = t0r;
1928 if (t1) t1 = t1r ^ tr_cw;
1929 else t1 = t1r;
1930 }
1931
1932 s_cw = util::SetLsb(s_cw, tl_cw);
1933 writer.Write(i, s_cw, tr_cw);
1934 }
1935
1936 // Verification hash
1937 int4 a_buf = fss::detail::Pack(a);
1938
1939 auto pi_tilde_0 = xor_hash_.Hash(cuda::std::tuple<int4, const int4>{a_buf, s0});
1940 auto pi_tilde_1 = xor_hash_.Hash(cuda::std::tuple<int4, const int4>{a_buf, s1});
1941 for (int i = 0; i < 4; ++i) cs[i] = util::Xor(pi_tilde_0[i], pi_tilde_1[i]);

1944 // their control bits differ everywhere (see Assumption 2 of doc/vdpf_merkle_fold_security.md).

1947 return 1;
1948 }
1949
1950 // Reject a failed generation attempt
1951 if (t0 == t1) return 1;
1952
1953 // Output correction word
1954 auto v_cw = Group::From(b_buf) + (-Group::From(s0)) + Group::From(s1);
1955 if (t1) v_cw = -v_cw;
1956 writer.Finish(v_cw.Into());

1958 return 0;
