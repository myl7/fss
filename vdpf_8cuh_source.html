123 __host__ __device__ int Gen(Key &k0, Key &k1, cuda::std::span<int4, 2> s0s, In a, int4 b_buf, bool retry = true) {
124 for (int attempt = 0; attempt < 128; ++attempt) {
125 int ret = Gen(k0.cws, k0.cs, k0.ocw, s0s, a, b_buf);
126 if (ret != 0) {
127 if (!retry) return 1;
128 s0s[0].x += 1;
129 s0s[1].y += 1;
130 continue;
131 }
132 k0.s0 = s0s[0];
133 for (int i = 0; i < in_bits; ++i) k1.cws[i] = k0.cws[i];
134 k1.cs = k0.cs;
135 k1.ocw = k0.ocw;
136 k1.s0 = s0s[1];
137 return 0;
138 }
139 return 1;
140 }

158 __host__ __device__ EvalResult Eval(bool b, const Key &key, In x) {
159 int4 s = key.s0;
160 s = util::SetLsb(s, false);
161 bool t = b;
162
163 for (int i = 0; i < in_bits; ++i) {
164 Cw cw = key.cws[i];
165 int4 s_cw = cw.s;
166 bool tl_cw = util::GetLsb(s_cw);
167 s_cw = util::SetLsb(s_cw, false);
168 bool tr_cw = cw.tr;
169
170 auto [sl, sr] = prg.Gen(s);
171
172 bool tl = util::GetLsb(sl);
173 sl = util::SetLsb(sl, false);
174 bool tr = util::GetLsb(sr);
175 sr = util::SetLsb(sr, false);
176
177 if (t) {
178 sl = util::Xor(sl, s_cw);
179 sr = util::Xor(sr, s_cw);
180 tl = tl ^ tl_cw;
181 tr = tr ^ tr_cw;
182 }
183
184 bool x_bit = (x >> (in_bits - 1 - i)) & 1;
185
186 if (!x_bit) {
187 s = sl;
188 t = tl;
189 } else {
190 s = sr;
191 t = tr;
192 }
193 }
194
195 // Output share
196 auto g = Group::From(s);
197 assert((key.ocw.w & 1) == 0);
198 if (t) g = g + Group::From(key.ocw);
199 if (b) g = -g;
200
201 // Corrected verification hash
202 int4 x_buf = util::Pack(x);
203
204 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, s});
205 if (t) {
206 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(key.cs));
207 }
208 return {g.Into(), pi_tilde};
209 }

220 void Prove(cuda::std::span<const cuda::std::array<int4, 4>> pi_tildes, cuda::std::span<const int4, 4> cs,
221 cuda::std::array<int4, 4> &pi) {
222 pi = {cs[0], cs[1], cs[2], cs[3]};
223 for (size_t i = 0; i < pi_tildes.size(); ++i) {
224 cuda::std::array<int4, 4> h_input =
225 util::Xor(cuda::std::span<const int4, 4>(pi), cuda::std::span<const int4, 4>(pi_tildes[i]));
226 auto h_out = hash.Hash(cuda::std::span<const int4, 4>(h_input));
227 pi[0] = util::Xor(pi[0], h_out[0]);
228 pi[1] = util::Xor(pi[1], h_out[1]);
229 }
230 }

260 void EvalAll(bool b, const Key &key, cuda::std::span<int4> ys, cuda::std::array<int4, 4> &pi) {
261 int4 st = key.s0;
262 bool t = b;
263 st = util::SetLsb(st, t);
264
265 assert(in_bits < sizeof(size_t) * 8);
266 size_t l = 0;
267 size_t r = 1ULL << in_bits;
268 int i = 0;
269
270 int par_depth_ = util::ResolveParDepth(par_depth);
271
272 // Phase 1: tree traversal, store (s, t) packed into ys temporarily
273#pragma omp parallel
274#pragma omp single
275 EvalTree(st, key.cws, ys, l, r, i, par_depth_);
276
277 // Phase 2: sequential output computation and proof accumulation
278 pi = {key.cs[0], key.cs[1], key.cs[2], key.cs[3]};
279 size_t n = 1ULL << in_bits;
280 assert((key.ocw.w & 1) == 0);
281 auto ocw_group = Group::From(key.ocw);
282 for (size_t j = 0; j < n; ++j) {
283 int4 sj = ys[j];
284 bool tj = util::GetLsb(sj);
285 sj = util::SetLsb(sj, false);
286
287 // Output share
288 auto g = Group::From(sj);
289 if (tj) g = g + ocw_group;
290 if (b) g = -g;
291 ys[j] = g.Into();
292
293 // Proof accumulation
294 int4 x_buf = util::Pack(static_cast<In>(j));
295
296 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, sj});
297 if (tj) {
298 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(key.cs));
299 }
300
301 cuda::std::array<int4, 4> h_input =
302 util::Xor(cuda::std::span<const int4, 4>(pi), cuda::std::span<const int4, 4>(pi_tilde));
303 auto h_out = hash.Hash(cuda::std::span<const int4, 4>(h_input));
304 pi[0] = util::Xor(pi[0], h_out[0]);
305 pi[1] = util::Xor(pi[1], h_out[1]);
306 }

323 void ProveMerkle(cuda::std::span<const cuda::std::array<int4, 4>> pi_tildes, cuda::std::span<const int4, 4> cs,
324 cuda::std::array<int4, 4> &pi) {
325 size_t count = pi_tildes.size();
326 assert(count >= 1);
327
328 int par_depth_ = util::ResolveParDepth(par_depth);
329
330 std::vector<cuda::std::array<int4, 2>> nodes(count);
331#pragma omp parallel for if (par_depth_ > 0)
332 for (size_t i = 0; i < count; ++i) {
333 nodes[i] = hash.Hash(cuda::std::span<const int4, 4>(pi_tildes[i]));
334 }
335
336 while (count > 1) {
337 size_t pairs = count / 2;
338 std::vector<cuda::std::array<int4, 2>> next(pairs + count % 2);
339#pragma omp parallel for if (par_depth_ > 0)
340 for (size_t i = 0; i < pairs; ++i) {
341 int4 msg[4] = {nodes[2 * i][0], nodes[2 * i][1], nodes[2 * i + 1][0], nodes[2 * i + 1][1]};
342 next[i] = hash.Hash(cuda::std::span<const int4, 4>(msg));
343 }
344 if (count % 2 != 0) next[pairs] = nodes[count - 1];
345 nodes = std::move(next);
346 count = nodes.size();
347 }
348
349 pi = {nodes[0][0], nodes[0][1], cs[2], cs[3]};
350 }

424 __host__ __device__ int Gen(
425 Cw cws[], cuda::std::array<int4, 4> &cs, int4 &ocw, cuda::std::span<const int4, 2> s0s, In a, int4 b_buf) {
426 int4 s0 = s0s[0];
427 s0 = util::SetLsb(s0, false);
428 int4 s1 = s0s[1];
429 s1 = util::SetLsb(s1, false);
430 bool t0 = false;
431 bool t1 = true;
432 b_buf = util::SetLsb(b_buf, false);
433
434 for (int i = 0; i < in_bits; ++i) {
435 auto [s0l, s0r] = prg.Gen(s0);
436 auto [s1l, s1r] = prg.Gen(s1);
437
438 bool t0l = util::GetLsb(s0l);
439 s0l = util::SetLsb(s0l, false);
440 bool t0r = util::GetLsb(s0r);
441 s0r = util::SetLsb(s0r, false);
442 bool t1l = util::GetLsb(s1l);
443 s1l = util::SetLsb(s1l, false);
444 bool t1r = util::GetLsb(s1r);
445 s1r = util::SetLsb(s1r, false);
446
447 bool a_bit = (a >> (in_bits - 1 - i)) & 1;
448
449 int4 s_cw;
450 if (!a_bit) s_cw = util::Xor(s0r, s1r);
451 else s_cw = util::Xor(s0l, s1l);
452
453 bool tl_cw = t0l ^ t1l ^ a_bit ^ 1;
454 bool tr_cw = t0r ^ t1r ^ a_bit;
455
456 if (!a_bit) {
457 s0 = s0l;
458 if (t0) s0 = util::Xor(s0, s_cw);
459 s1 = s1l;
460 if (t1) s1 = util::Xor(s1, s_cw);
461
462 if (t0) t0 = t0l ^ tl_cw;
463 else t0 = t0l;
464 if (t1) t1 = t1l ^ tl_cw;
465 else t1 = t1l;
466 } else {
467 s0 = s0r;
468 if (t0) s0 = util::Xor(s0, s_cw);
469 s1 = s1r;
470 if (t1) s1 = util::Xor(s1, s_cw);
471
472 if (t0) t0 = t0r ^ tr_cw;
473 else t0 = t0r;
474 if (t1) t1 = t1r ^ tr_cw;
475 else t1 = t1r;
476 }
477
478 s_cw = util::SetLsb(s_cw, tl_cw);
479 // Both 16B halves written so the full 32B slot gets unmasked stores.
480 // {s_cw, tr_cw} alone leaves padding undefined: NVCC emits a masked
481 // 256-bit store, and the partial-sector write forces a write-allocate
482 // round trip per level on GPU (measured ~2x Gen time).
483 int4 *cw_slot = reinterpret_cast<int4 *>(&cws[i]);
484 cw_slot[0] = s_cw;
485 cw_slot[1] = make_int4(tr_cw, 0, 0, 0);
486 }
487
488 // Verification hash
489 int4 a_buf = util::Pack(a);
490
491 auto pi_tilde_0 = xor_hash.Hash(cuda::std::tuple<int4, const int4>{a_buf, s0});
492 auto pi_tilde_1 = xor_hash.Hash(cuda::std::tuple<int4, const int4>{a_buf, s1});
493 cs = util::Xor(cuda::std::span<const int4, 4>(pi_tilde_0), cuda::std::span<const int4, 4>(pi_tilde_1));
494
495 // A zero cs would let adversarial keys keep both parties' per-point hashes equal while
496 // their control bits differ everywhere (see Assumption 2 of doc/vdpf_merkle_fold_security.md).
497 if ((cs[0].x | cs[0].y | cs[0].z | cs[0].w | cs[1].x | cs[1].y | cs[1].z | cs[1].w | cs[2].x | cs[2].y |
498 cs[2].z | cs[2].w | cs[3].x | cs[3].y | cs[3].z | cs[3].w) == 0) {
499 return 1;
500 }
501
502 // Check retry condition
503 if (t0 == t1) return 1;
504
505 // Output correction word
506 auto v_cw = Group::From(b_buf) + (-Group::From(s0)) + Group::From(s1);
507 if (t1) v_cw = -v_cw;
508 ocw = v_cw.Into();
509
510 return 0;
511 }

513 void EvalTree(
514 int4 st, cuda::std::span<const Cw> cws, cuda::std::span<int4> ys, size_t l, size_t r, int i, int par_depth_) {
515 if (i == in_bits) {
516 assert(l + 1 == r);
517 ys[l] = st;
518 return;
519 }
520
521 bool t = util::GetLsb(st);
522 int4 s = st;
523 s = util::SetLsb(s, false);
524
525 Cw cw = cws[i];
526 int4 s_cw = cw.s;
527 bool tl_cw = util::GetLsb(s_cw);
528 s_cw = util::SetLsb(s_cw, false);
529 bool tr_cw = cw.tr;
530
531 auto [sl, sr] = prg.Gen(s);
532
533 bool tl = util::GetLsb(sl);
534 sl = util::SetLsb(sl, false);
535 bool tr = util::GetLsb(sr);
536 sr = util::SetLsb(sr, false);
537
538 if (t) {
539 sl = util::Xor(sl, s_cw);
540 sr = util::Xor(sr, s_cw);
541 tl = tl ^ tl_cw;
542 tr = tr ^ tr_cw;
543 }
544
545 int4 stl = sl;
546 stl = util::SetLsb(stl, tl);
547 int4 str = sr;
548 str = util::SetLsb(str, tr);
549
550 size_t mid = (l + r) / 2;
551
552 if (i < par_depth_) {
553#pragma omp task
554 EvalTree(stl, cws, ys, l, mid, i + 1, par_depth_);
555#pragma omp task
556 EvalTree(str, cws, ys, mid, r, i + 1, par_depth_);
557#pragma omp taskwait
558 } else {
559 EvalTree(stl, cws, ys, l, mid, i + 1, par_depth_);
560 EvalTree(str, cws, ys, mid, r, i + 1, par_depth_);
561 }
562 }
