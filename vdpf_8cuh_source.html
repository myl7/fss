123 __host__ __device__ int Gen(Key &k0, Key &k1, cuda::std::span<int4, 2> s0s, In a, int4 b_buf, bool retry = true) {
124 for (int attempt = 0; attempt < 128; ++attempt) {
125 int ret = Gen(k0.cws, k0.cs, k0.ocw, s0s, a, b_buf);
126 if (ret != 0) {
127 if (!retry) return 1;
128 s0s[0].x = util::IncrementWord(s0s[0].x);
129 s0s[1].y = util::IncrementWord(s0s[1].y);
130 continue;
131 }
132 k0.s0 = s0s[0];
133 for (int i = 0; i < in_bits; ++i) k1.cws[i] = k0.cws[i];
134 k1.cs = k0.cs;
135 k1.ocw = k0.ocw;
136 k1.s0 = s0s[1];
137 return 0;
138 }
139 return 1;

164 __host__ __device__ cuda::std::array<int4, 4> Eval(bool b, int4 s0, cuda::std::span<const Cw> cws,
165 cuda::std::span<const int4, 4> cs, int4 ocw, In x, int4 &y) {
166 int4 s = s0;
167 s = util::SetLsb(s, false);
168 bool t = b;
169
170 for (int i = 0; i < in_bits; ++i) {
171 Cw cw = cws[i];
172 int4 s_cw = cw.s;
173 bool tl_cw = util::GetLsb(s_cw);
174 s_cw = util::SetLsb(s_cw, false);
175 bool tr_cw = cw.tr;
176
177 auto [sl, sr] = prg.Gen(s);
178
179 bool tl = util::GetLsb(sl);
180 sl = util::SetLsb(sl, false);
181 bool tr = util::GetLsb(sr);
182 sr = util::SetLsb(sr, false);
183
184 if (t) {
185 sl = util::Xor(sl, s_cw);
186 sr = util::Xor(sr, s_cw);
187 tl = tl ^ tl_cw;
188 tr = tr ^ tr_cw;
189 }
190
191 bool x_bit = (x >> (in_bits - 1 - i)) & 1;
192
193 if (!x_bit) {
194 s = sl;
195 t = tl;
196 } else {
197 s = sr;
198 t = tr;
199 }
200 }
201
202 // Output share
203 auto g = Group::From(s);
204 assert((ocw.w & 1) == 0);
205 if (t) g = g + Group::From(ocw);
206 if (b) g = -g;
207
208 // Corrected verification hash
209 int4 x_buf = util::Pack(x);
210
211 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, s});
212 if (t) {
213 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(cs));
214 }
215 y = g.Into();
216 return pi_tilde;
217 }

297 void EvalRange(bool b, size_t L, size_t R, int4 s0, cuda::std::span<const Cw> cws, cuda::std::span<const int4, 4> cs,
298 int4 ocw, cuda::std::span<int4> ys, cuda::std::array<int4, 4> &pi) {
299 assert(in_bits < sizeof(size_t) * 8);
300 size_t domain_size = 1ULL << in_bits;
301 // check that [L, R] is contained in [0, domain_size - 1].
302 assert(L <= R);
303 assert(R < domain_size);
304
305 size_t num_inputs = R - L + 1;
306 // check enough space is allocated for the output
307 assert(ys.size() >= num_inputs);
308
309 const int4 st = util::SetLsb(s0, b);
310 const int par_depth_ = util::ResolveParDepth(par_depth, in_bits);
311
312 // Phase 1: tree traversal, store (s, t) packed into ys temporarily
313 if (par_depth_ == 0) {
314 EvalPrunedTree(st, cws, ys, 0, domain_size, 0, 0, L, R);
315 } else {
316 const int threads = util::ResolveParThreads(par_depth_, num_inputs);
317
318#pragma omp parallel num_threads(threads)
319#pragma omp single
320 EvalPrunedTree(st, cws, ys, 0, domain_size, 0, par_depth_, L, R);
321 }
322 // Phase 2: sequential output computation and proof accumulation
323 pi = {cs[0], cs[1], cs[2], cs[3]};
324
325 assert((ocw.w & 1) == 0);
326 auto ocw_group = Group::From(ocw);
327 for (size_t j = L; j <= R; ++j) {
328 int4 sj = ys[j - L];
329 bool tj = util::GetLsb(sj);
330 sj = util::SetLsb(sj, false);
331
332 // Output share
333 auto g = Group::From(sj);
334 if (tj) g = g + ocw_group;
335 if (b) g = -g;
336 ys[j - L] = g.Into();
337
338 // Proof accumulation
339 int4 x_buf = util::Pack(static_cast<In>(j));
340
341 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, sj});
342 if (tj) {
343 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(cs));
344 }
345
346 cuda::std::array<int4, 4> h_input =
347 util::Xor(cuda::std::span<const int4, 4>(pi), cuda::std::span<const int4, 4>(pi_tilde));
348 auto h_out = hash.Hash(cuda::std::span<const int4, 4>(h_input));
349 pi[0] = util::Xor(pi[0], h_out[0]);
350 pi[1] = util::Xor(pi[1], h_out[1]);
351 }
352 }

368 void ProveMerkle(cuda::std::span<const cuda::std::array<int4, 4>> pi_tildes, cuda::std::span<const int4, 4> cs,
369 cuda::std::array<int4, 4> &pi) {
370 size_t count = pi_tildes.size();
371 assert(count >= 1);
372
373 std::vector<cuda::std::array<int4, 2>> nodes(count);
374 util::ParallelFor(par_depth, count, [&](size_t i) {
375 nodes[i] = hash.Hash(cuda::std::span<const int4, 4>(pi_tildes[i]));
376 });
377
378 while (count > 1) {
379 size_t pairs = count / 2;
380 std::vector<cuda::std::array<int4, 2>> next(pairs + count % 2);
381 util::ParallelFor(par_depth, pairs, [&](size_t i) {
382 int4 msg[4] = {nodes[2 * i][0], nodes[2 * i][1], nodes[2 * i + 1][0], nodes[2 * i + 1][1]};
383 next[i] = hash.Hash(cuda::std::span<const int4, 4>(msg));
384 });
385 if (count % 2 != 0) next[pairs] = nodes[count - 1];
386 nodes = std::move(next);
387 count = nodes.size();
388 }
389
390 pi = {nodes[0][0], nodes[0][1], cs[2], cs[3]};
391 }

405 void EvalAllMerkle(bool b, const Key &key, cuda::std::span<int4> ys, cuda::std::array<int4, 4> &pi) {
406 int4 st = key.s0;
407 bool t = b;
408 st = util::SetLsb(st, t);
409
410 assert(in_bits < sizeof(size_t) * 8);
411 size_t l = 0;
412 size_t r = 1ULL << in_bits;
413 int i = 0;
414
415 int par_depth_ = util::ResolveParDepth(par_depth, in_bits);
416
417 // Phase 1: tree traversal, store (s, t) packed into ys temporarily
418 if (par_depth_ == 0) {
419 EvalTree(st, key.cws, ys, l, r, i, 0);
420 } else {
421 int threads = util::ResolveParThreads(par_depth_, r);
422#pragma omp parallel num_threads(threads)
423#pragma omp single
424 EvalTree(st, key.cws, ys, l, r, i, par_depth_);
425 }
426
427 // Phase 2: parallel output conversion and leaf hashing, then the Merkle fold
428 pi = {key.cs[0], key.cs[1], key.cs[2], key.cs[3]};
429 size_t n = 1ULL << in_bits;
430 assert((key.ocw.w & 1) == 0);
431 auto ocw_group = Group::From(key.ocw);
432 std::vector<cuda::std::array<int4, 4>> leaves(n);
433 util::ParallelFor(par_depth, n, [&](size_t j) {
434 int4 sj = ys[j];
435 bool tj = util::GetLsb(sj);
436 sj = util::SetLsb(sj, false);
437
438 auto g = Group::From(sj);
439 if (tj) g = g + ocw_group;
440 if (b) g = -g;
441 ys[j] = g.Into();
442
443 int4 x_buf = util::Pack(static_cast<In>(j));
444
445 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, sj});
446 if (tj) {
447 leaves[j] = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(key.cs));
448 } else {
449 leaves[j] = pi_tilde;
450 }
451 });
452 ProveMerkle(cuda::std::span<const cuda::std::array<int4, 4>>(leaves), cuda::std::span<const int4, 4>(key.cs), pi);
453 }

469 __host__ __device__ int Gen(
470 Cw cws[], cuda::std::array<int4, 4> &cs, int4 &ocw, cuda::std::span<const int4, 2> s0s, In a, int4 b_buf) {
471 int4 s0 = s0s[0];
472 s0 = util::SetLsb(s0, false);
473 int4 s1 = s0s[1];
474 s1 = util::SetLsb(s1, false);
475 bool t0 = false;
476 bool t1 = true;
477 b_buf = util::SetLsb(b_buf, false);
478
479 for (int i = 0; i < in_bits; ++i) {
480 auto [s0l, s0r] = prg.Gen(s0);
481 auto [s1l, s1r] = prg.Gen(s1);
482
483 bool t0l = util::GetLsb(s0l);
484 s0l = util::SetLsb(s0l, false);
485 bool t0r = util::GetLsb(s0r);
486 s0r = util::SetLsb(s0r, false);
487 bool t1l = util::GetLsb(s1l);
488 s1l = util::SetLsb(s1l, false);
489 bool t1r = util::GetLsb(s1r);
490 s1r = util::SetLsb(s1r, false);
491
492 bool a_bit = (a >> (in_bits - 1 - i)) & 1;
493
494 int4 s_cw;
495 if (!a_bit) s_cw = util::Xor(s0r, s1r);
496 else s_cw = util::Xor(s0l, s1l);
497
498 bool tl_cw = t0l ^ t1l ^ a_bit ^ 1;
499 bool tr_cw = t0r ^ t1r ^ a_bit;
500
501 if (!a_bit) {
502 s0 = s0l;
503 if (t0) s0 = util::Xor(s0, s_cw);
504 s1 = s1l;
505 if (t1) s1 = util::Xor(s1, s_cw);
506
507 if (t0) t0 = t0l ^ tl_cw;
508 else t0 = t0l;
509 if (t1) t1 = t1l ^ tl_cw;
510 else t1 = t1l;
511 } else {
512 s0 = s0r;
513 if (t0) s0 = util::Xor(s0, s_cw);
514 s1 = s1r;
515 if (t1) s1 = util::Xor(s1, s_cw);
516
517 if (t0) t0 = t0r ^ tr_cw;
518 else t0 = t0r;
519 if (t1) t1 = t1r ^ tr_cw;
520 else t1 = t1r;
521 }
522
523 s_cw = util::SetLsb(s_cw, tl_cw);
524 // Both 16B halves written so the full 32B slot gets unmasked stores.
525 // {s_cw, tr_cw} alone leaves padding undefined: NVCC emits a masked
526 // 256-bit store, and the partial-sector write forces a write-allocate
527 // round trip per level on GPU (measured ~2x Gen time).
528 int4 *cw_slot = reinterpret_cast<int4 *>(&cws[i]);
529 cw_slot[0] = s_cw;
530 cw_slot[1] = make_int4(tr_cw, 0, 0, 0);
531 }
532
533 // Verification hash
534 int4 a_buf = util::Pack(a);
535
536 auto pi_tilde_0 = xor_hash.Hash(cuda::std::tuple<int4, const int4>{a_buf, s0});
537 auto pi_tilde_1 = xor_hash.Hash(cuda::std::tuple<int4, const int4>{a_buf, s1});
538 cs = util::Xor(cuda::std::span<const int4, 4>(pi_tilde_0), cuda::std::span<const int4, 4>(pi_tilde_1));
539
540 // A zero cs would let adversarial keys keep both parties' per-point hashes equal while
541 // their control bits differ everywhere (see Assumption 2 of doc/vdpf_merkle_fold_security.md).
542 if ((cs[0].x | cs[0].y | cs[0].z | cs[0].w | cs[1].x | cs[1].y | cs[1].z | cs[1].w | cs[2].x | cs[2].y |
543 cs[2].z | cs[2].w | cs[3].x | cs[3].y | cs[3].z | cs[3].w) == 0) {
544 return 1;
545 }
546
547 // Check retry condition
548 if (t0 == t1) return 1;
549
550 // Output correction word
551 auto v_cw = Group::From(b_buf) + (-Group::From(s0)) + Group::From(s1);
552 if (t1) v_cw = -v_cw;
553 ocw = v_cw.Into();
554
555 return 0;
556 }

559 void EvalTree(
560 int4 st, cuda::std::span<const Cw> cws, cuda::std::span<int4> ys, size_t l, size_t r, int i, int par_depth_) {
561 if (i == in_bits) {
562 assert(l + 1 == r);
563 ys[l] = st;
564 return;
565 }
566
567 bool t = util::GetLsb(st);
568 int4 s = st;
569 s = util::SetLsb(s, false);
570
571 Cw cw = cws[i];
572 int4 s_cw = cw.s;
573 bool tl_cw = util::GetLsb(s_cw);
574 s_cw = util::SetLsb(s_cw, false);
575 bool tr_cw = cw.tr;
576
577 auto [sl, sr] = prg.Gen(s);
578
579 bool tl = util::GetLsb(sl);
580 sl = util::SetLsb(sl, false);
581 bool tr = util::GetLsb(sr);
582 sr = util::SetLsb(sr, false);
583
584 if (t) {
585 sl = util::Xor(sl, s_cw);
586 sr = util::Xor(sr, s_cw);
587 tl = tl ^ tl_cw;
588 tr = tr ^ tr_cw;
589 }
590
591 int4 stl = sl;
592 stl = util::SetLsb(stl, tl);
593 int4 str = sr;
594 str = util::SetLsb(str, tr);
595
596 size_t mid = (l + r) / 2;
597
598 if (i < par_depth_) {
599#pragma omp task
600 EvalTree(stl, cws, ys, l, mid, i + 1, par_depth_);
601#pragma omp task
602 EvalTree(str, cws, ys, mid, r, i + 1, par_depth_);
603#pragma omp taskwait
604 } else {
605 EvalTree(stl, cws, ys, l, mid, i + 1, par_depth_);
606 EvalTree(str, cws, ys, mid, r, i + 1, par_depth_);
607 }
608 }

610 void EvalPrunedTree(int4 st, cuda::std::span<const Cw> cws, cuda::std::span<int4> ys, size_t l, size_t r, int i,
611 int par_depth_, size_t L, size_t R) {
612 // this subtree is outside the range [L, R]
613 if (r <= L || l > R) {
614 return;
615 }
616 // this is a full subtree
617 if (l >= L && r - 1 <= R) {
618 EvalTree(st, cws, ys.subspan(l - L, r - l), 0, r - l, i, par_depth_);
619 return;
620 }
621 if (i == in_bits) {
622 assert(l + 1 == r);
623 // we only write leaves from [L, R]
624 ys[l - L] = st;
625 return;
626 }
627
628 bool t = util::GetLsb(st);
629 int4 s = st;
630 s = util::SetLsb(s, false);
631
632 Cw cw = cws[i];
633 int4 s_cw = cw.s;
634 bool tl_cw = util::GetLsb(s_cw);
635 s_cw = util::SetLsb(s_cw, false);
636 bool tr_cw = cw.tr;
637
638 auto [sl, sr] = prg.Gen(s);
639
640 bool tl = util::GetLsb(sl);
641 sl = util::SetLsb(sl, false);
642 bool tr = util::GetLsb(sr);
643 sr = util::SetLsb(sr, false);
644
645 if (t) {
646 sl = util::Xor(sl, s_cw);
647 sr = util::Xor(sr, s_cw);
648 tl = tl ^ tl_cw;
649 tr = tr ^ tr_cw;
650 }
651
652 int4 stl = sl;
653 stl = util::SetLsb(stl, tl);
654 int4 str = sr;
655 str = util::SetLsb(str, tr);
656
657 size_t mid = (l + r) / 2;
658
659 if (i < par_depth_) {
660 if (mid > L) {
661#pragma omp task
662 EvalPrunedTree(stl, cws, ys, l, mid, i + 1, par_depth_, L, R);
663 }
664 if (mid <= R) {
665#pragma omp task
666 EvalPrunedTree(str, cws, ys, mid, r, i + 1, par_depth_, L, R);
667 }
668#pragma omp taskwait
669 } else {
670 if (mid > L) {
671 EvalPrunedTree(stl, cws, ys, l, mid, i + 1, par_depth_, L, R);
672 }
673 if (mid <= R) {
674 EvalPrunedTree(str, cws, ys, mid, r, i + 1, par_depth_, L, R);
675 }
676 }
677 }
