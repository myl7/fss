66// Split correction-word view over the caller's arrays. Layer i is read from
67// cws[i*stride] and the right-control correction is bit i of packed_tr. The
68// terminal is rebuilt as a plain Cw with no control bit, so bits of packed_tr at
69// or above in_bits are never read. The layer LSB keeps the left-control
70// correction exactly as the Cw layout does. operator[] recovers the Cw the shared
71// evaluation consumes, so its body stays identical on the Cw and split paths.
72template <typename Cw, typename In, int in_bits>
73struct VdpfCwsView {
74 const int4 *s;
75 In packed_tr;
76 size_t stride;
77
78 __host__ __device__ inline Cw operator[](int i) const {
79 const int4 s_cw = s[static_cast<size_t>(i) * stride];
80 if (i >= in_bits) return Cw{s_cw, false};
81 return Cw{s_cw, ((packed_tr >> i) & 1) != 0};
82 }
83};

751 // Writes Cw slots in the legacy AoS layout, covering each full 32B slot.
752 struct AosWriter {
753 Cw *cws;
754
755 __host__ __device__ void Write(int i, int4 s_cw, bool tr_cw) const {
756 // Both 16B halves written so the full 32B slot gets unmasked stores.
757 // {s_cw, tr_cw} alone leaves padding undefined: NVCC emits a masked
758 // 256-bit store, and the partial-sector write forces a write-allocate
759 // round trip per level on GPU (measured ~2x Gen time).
760 int4 *cw_slot = reinterpret_cast<int4 *>(&cws[i]);
761 cw_slot[0] = s_cw;
762 cw_slot[1] = make_int4(tr_cw, 0, 0, 0);
763 }

766 cws[in_bits].s = terminal;
767 cws[in_bits].tr = false;
768 }

859template <int in_bits, int threads_per_block, typename Group, typename Prg, typename XorHash,
860 typename Hash, typename In>
861__device__ void VdpfRangeMerkleLeavesWork(bool b, In first,
862 int subtree_depth, int frontier_depth, int block_root_depth, int4 s0, const int4 *cws,
863 In packed_tr, size_t stride, const int4 *cs,
864 int4 *ys, cuda::std::array<int4, 2> *leaves, Prg prg,
865 XorHash xor_hash, Hash hash) {
866 using MathScheme = Dpf<in_bits, Group, Prg, In>;
867 const int root_depth = in_bits - subtree_depth;
868 const int block_subtree_depth = frontier_depth - block_root_depth;
869 const int tail_depth = subtree_depth - frontier_depth;
870 __shared__ int4 frontier[threads_per_block];
871 if (threadIdx.x == 0) {
872 int4 st = util::SetLsb(s0, b);
873 for (int level = 0; level < root_depth; ++level) {
874 const bool right = (first >> (in_bits - level - 1)) & 1;
875 st = fss::detail::SchemeAccess::DpfPathStep<MathScheme>(
876 prg, st, VdpfCwSeed(cws, stride, level), VdpfCwRight(packed_tr, level), right);
877 }
878 for (int level = 0; level < block_root_depth; ++level) {
879 const bool right = (blockIdx.x >> (block_root_depth - level - 1)) & 1;
880 st = fss::detail::SchemeAccess::DpfPathStep<MathScheme>(prg, st,
881 VdpfCwSeed(cws, stride, root_depth + level),
882 VdpfCwRight(packed_tr, root_depth + level), right);
883 }
884 frontier[0] = st;
885 }
886 __syncthreads();
887 for (int level = 0; level < block_subtree_depth; ++level) {
888 const bool active = threadIdx.x < (1u << (level + 1));
889 int4 st = {0, 0, 0, 0};
890 if (active) st = frontier[threadIdx.x >> 1];
891 __syncthreads();
892 if (active) {
893 int4 left, right;
894 const int cw_index = root_depth + block_root_depth + level;
895 fss::detail::SchemeAccess::DpfExpandNode<MathScheme>(prg, st,
896 VdpfCwSeed(cws, stride, cw_index), VdpfCwRight(packed_tr, cw_index), left, right);
897 frontier[threadIdx.x] = (threadIdx.x & 1) ? right : left;
898 }
899 __syncthreads();
900 }
901 int4 pending[8];
902 int4 node = frontier[threadIdx.x];
903 const int frontier_level = root_depth + frontier_depth;
904 int level = frontier_level;
905 const int count = 1 << tail_depth;
906 const size_t thread_index = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
907 const size_t offset = thread_index << tail_depth;
908 const auto cw = Group::From(cws[static_cast<size_t>(in_bits) * stride]);
909 for (int i = 0; i < count; ++i) {
910 while (level < in_bits) {
911 int4 left, right;
912 fss::detail::SchemeAccess::DpfExpandNode<MathScheme>(prg, node,
913 VdpfCwSeed(cws, stride, level), VdpfCwRight(packed_tr, level), left, right);
914 pending[level - frontier_level] = right;
915 node = left;
916 ++level;
917 }
918 const bool t = util::GetLsb(node);
919 const int4 seed = util::SetLsb(node, false);
920 auto value = Group::From(seed);
921 if (t) value = value + cw;
922 if (b) value = -value;
923 ys[offset + i] = value.Into();
924 const In x = static_cast<In>(first + offset + i);
925 auto corrected = xor_hash.Hash(cuda::std::tuple<int4, const int4>{fss::detail::Pack(x), seed});
926 if (t) corrected = util::Xor(cuda::std::span<const int4, 4>(corrected),
927 cuda::std::span<const int4, 4>(cs, 4));
928 leaves[offset + i] = hash.Hash(cuda::std::span<const int4, 4>(corrected));
929 if (i + 1 < count) {
930 int next = i + 1;
931 level = in_bits - 1;
932 while ((next & 1) == 0) {
933 --level;
934 next >>= 1;
935 }
936 node = pending[level - frontier_level];
937 ++level;
938 }
939 }

985// One thread per query: walks the raw-key path of xs[index] from the root and
986// writes the output share and the hashed corrected per-point hash as the leaf
987// of the batch Merkle fold. The same per-point math as Vdpf::Eval, fed by the
988// split cws/packed-tr layout.
989template <int in_bits, typename Group, typename Prg, typename XorHash, typename Hash, typename In>
990__device__ void VdpfBatchMerkleLeavesWork(bool b, int4 s0, const int4 *cws,
991 In packed_tr, size_t stride, const int4 *cs,
992 const In *xs, int4 *ys, cuda::std::array<int4, 2> *leaves, size_t count,
993 Prg prg, XorHash xor_hash,
994 Hash hash) {
995 using MathScheme = Dpf<in_bits, Group, Prg, In>;
996 const size_t index = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
997 if (index >= count) return;
998 const In x = xs[index];
999 int4 node = util::SetLsb(s0, b);
1000 for (int level = 0; level < in_bits; ++level) {
1001 const bool right = (x >> (in_bits - level - 1)) & 1;
1002 node = fss::detail::SchemeAccess::DpfPathStep<MathScheme>(
1003 prg, node, VdpfCwSeed(cws, stride, level), VdpfCwRight(packed_tr, level), right);
1004 }
1005 const bool t = util::GetLsb(node);
1006 const int4 seed = util::SetLsb(node, false);
1007 auto value = Group::From(seed);
1008 const auto cw = Group::From(cws[static_cast<size_t>(in_bits) * stride]);
1009 if (t) value = value + cw;
1010 if (b) value = -value;

1039template <int in_bits, int frontier_depth, int block_root_depth, typename Group, typename Prg,
1040 typename XorHash, typename Hash, typename In>
1041__device__ void VdpfMerkleFrontierWork(bool b, int4 s0, const int4 *cws,
1042 In packed_tr, size_t stride, const int4 *cs,
1043 int4 *ys, cuda::std::array<int4, 2> *roots, Prg prg,
1044 XorHash xor_hash, Hash hash) {
1045 using MathScheme = Dpf<in_bits, Group, Prg, In>;
1046 constexpr int kTailDepth = in_bits - frontier_depth;
1047 constexpr int kBlockSubtreeDepth = frontier_depth - block_root_depth;
1048 const int thread_index = blockIdx.x * blockDim.x + threadIdx.x;
1049 if (thread_index >= (1 << frontier_depth)) return;
1050
1051 // Phase 0: block root walk (top block_root_depth levels) by thread 0.
1052 __shared__ int4 shared_frontier[1 << kBlockSubtreeDepth];
1053 if (threadIdx.x == 0) {
1054 int4 st = util::SetLsb(s0, b);
1055#pragma unroll
1056 for (int i = 0; i < block_root_depth; ++i) {
1057 bool x_bit = (blockIdx.x >> (block_root_depth - 1 - i)) & 1;
1058 {
1059 auto node_prg = prg;
1060 st = fss::detail::SchemeAccess::DpfPathStep<MathScheme>(
1061 node_prg, st, VdpfCwSeed(cws, stride, i), VdpfCwRight(packed_tr, i), x_bit);
1062 }
1063 }
1064 shared_frontier[0] = st;
1065 }
1066 __syncthreads();
1067
1068 // Phase 1: breadth-parallel expansion of the block's subtree, levels block_root_depth..frontier_depth.
1069#pragma unroll
1070 for (int i = 0; i < kBlockSubtreeDepth; ++i) {
1071 const bool active = threadIdx.x < (1 << (i + 1));
1072 int4 n = {0, 0, 0, 0};
1073 if (active) n = shared_frontier[threadIdx.x >> 1];
1074 // All parent reads must finish before any child overwrites the frontier.
1075 __syncthreads();
1076 if (active) {
1077 int4 left, right;
1078 {
1079 auto node_prg = prg;
1080 fss::detail::SchemeAccess::DpfExpandNode<MathScheme>(node_prg, n,
1081 VdpfCwSeed(cws, stride, block_root_depth + i),
1082 VdpfCwRight(packed_tr, block_root_depth + i), left, right);
1083 }
1084 shared_frontier[threadIdx.x] = (threadIdx.x & 1) ? right : left;
1085 }
1086 __syncthreads();
1087 }
1088
1089 // Phase 2: expand the subtree below depth frontier_depth, convert each leaf to the output
1090 // share and the corrected verification hash, and fold the leaf hashes into
1091 // the thread's Merkle subtree on the fly.
1092 int4 nodes[1 << kTailDepth];
1093 nodes[0] = shared_frontier[threadIdx.x];
1094 int count = 1;
1095#pragma unroll
1096 for (int lev = frontier_depth; lev < in_bits; ++lev) {
1097#pragma unroll
1098 for (int k = count - 1; k >= 0; --k) {
1099 int4 left, right;
1100 {
1101 auto node_prg = prg;
1102 fss::detail::SchemeAccess::DpfExpandNode<MathScheme>(node_prg, nodes[k],
1103 VdpfCwSeed(cws, stride, lev), VdpfCwRight(packed_tr, lev), left, right);
1104 }
1105 nodes[2 * k] = left;
1106 nodes[2 * k + 1] = right;
1107 }
1108 count *= 2;
1109 }
1110
1111 auto ocw_group = Group::From(cws[static_cast<size_t>(in_bits) * stride]);
1112 const int output_offset = thread_index << kTailDepth;
1113 cuda::std::array<int4, 2> stack[kTailDepth + 1];
1114#pragma unroll
1115 for (int k = 0; k < count; ++k) {
1116 int4 st_cur = nodes[k];
1117 bool t_cur = util::GetLsb(st_cur);
1118 int4 s_cur = util::SetLsb(st_cur, false);
1119
1120 // Output share
1121 auto g = Group::From(s_cur);
1122 if (t_cur) g = g + ocw_group;
1123 if (b) g = -g;
1124 ys[output_offset + k] = g.Into();
1125
1126 // Leaf: the corrected verification hash of the point
1127 int4 x_buf = fss::detail::Pack(static_cast<In>(output_offset + k));
1128 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, s_cur});
1129 if (t_cur) {
1130 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(cs, 4));
1131 }
1132 int4 leaf_msg[4] = {pi_tilde[0], pi_tilde[1], pi_tilde[2], pi_tilde[3]};
1133 cuda::std::array<int4, 2> node = hash.Hash(cuda::std::span<const int4, 4>(leaf_msg));
1134
1135 // Stream the leaf into the thread's subtree: combine with the pending
1136 // sibling stack entries while the leaf index has trailing one bits.
1137 int rest = k;
1138 int lvl = 0;
1139#pragma unroll
1140 while (rest != 0 && (rest & 1) == 1) {
1141 int4 fold_msg[4] = {stack[lvl][0], stack[lvl][1], node[0], node[1]};
1142 node = hash.Hash(cuda::std::span<const int4, 4>(fold_msg));
1143 rest >>= 1;
1144 ++lvl;
1145 }
1146 stack[lvl] = node;
1147 }
1148
1149 // Block fold of the threads' subtree roots, in thread order: adjacent pairs
1150 // at each stride, slot idx combining with idx + stride.
1151 __shared__ cuda::std::array<int4, 2> s_node[1 << kBlockSubtreeDepth];
1152 s_node[threadIdx.x] = stack[kTailDepth];
1153 __syncthreads();
1154#pragma unroll
1155 for (int stride = 1; stride < (1 << kBlockSubtreeDepth); stride <<= 1) {
1156 int idx = 2 * stride * threadIdx.x;
1157 if (idx < (1 << kBlockSubtreeDepth)) {
1158 int4 fold_msg[4] = {s_node[idx][0], s_node[idx][1], s_node[idx + stride][0], s_node[idx + stride][1]};
1159 s_node[idx] = hash.Hash(cuda::std::span<const int4, 4>(fold_msg));
1160 }
1161 __syncthreads();
1162 }
1163 if (threadIdx.x == 0) roots[blockIdx.x] = s_node[0];
1164}

1212// Point-evaluate in_bits-level VDPF keys, one thread per key. Same seed walk as
1213// the DPF kernel plus the per-point verification hash; the walk carries no hash
1214// cost, and the tail applies the verification hash and, when t, folds in cs. cws
1215// holds dense level-major correction words, the layout the split form of Gen()
1216// writes: layer i of key k is cws[i*nkeys + k] and the terminal is
1217// cws[in_bits*nkeys + k]. tr[k] carries key k's packed right-control bits, bit i
1218// for layer i. cs stays per-key: it is read once per evaluation, not once per level.
1219template <int in_bits, int unroll, bool b, typename Group, typename Prg, typename XorHash,
1220 typename In, typename Query>
1221__global__ void VdpfEvalPointKernel(int4 *ys, cuda::std::array<int4, 4> *pi, const int4 *seeds,
1222 const int4 *cws, const In *tr, const cuda::std::array<int4, 4> *cs, const Query *xs,
1223 int nkeys, Prg prg, XorHash xor_hash) {
1224 static_assert(in_bits <= sizeof(In) * 8, "In too narrow for in_bits");
1225 int tid = blockIdx.x * blockDim.x + threadIdx.x;
1226 if (tid >= nkeys) return;
1227 int4 s = fss::util::SetLsb(seeds[tid], false);
1228 bool t = b;
1229 In x = static_cast<In>(xs[tid]);
1230 const In packed_tr = tr[tid];
1231
1232#pragma unroll unroll
1233 for (int i = 0; i < in_bits; ++i) {
1234 int4 s_cw = cws[static_cast<size_t>(i) * nkeys + tid];
1235 bool tl_cw = fss::util::GetLsb(s_cw);
1236 s_cw = fss::util::SetLsb(s_cw, false);
1237 bool tr_cw = ((packed_tr >> i) & 1) != 0;
1238
1239 auto g = prg.Gen(s);
1240 int4 sl = g[0], sr = g[1];
1241 bool tl = fss::util::GetLsb(sl);
1242 sl = fss::util::SetLsb(sl, false);
1243 bool tr = fss::util::GetLsb(sr);
1244 sr = fss::util::SetLsb(sr, false);
1245
1246 if (t) {
1247 sl = fss::util::Xor(sl, s_cw);
1248 sr = fss::util::Xor(sr, s_cw);
1249 tl = tl ^ tl_cw;
1250 tr = tr ^ tr_cw;
1251 }
1252
1253 bool x_bit = GetBit(x, in_bits - 1 - i);
1254 if (x_bit) {
1255 s = sr;
1256 t = tr;
1257 } else {
1258 s = sl;
1259 t = tl;
1260 }
1261 }
1262
1263 auto y = Group::From(s);
1264 if (t) y = y + Group::From(cws[static_cast<size_t>(in_bits) * nkeys + tid]);
1265 if constexpr (b) y = -y;
1266 ys[tid] = y.Into();
1267
1268 int4 x_buf = fss::detail::Pack(x);
1269 auto pi_tilde = xor_hash.Hash(cuda::std::tuple<int4, const int4>{x_buf, s});
1270 if (t) {
1271 pi_tilde = util::Xor(cuda::std::span<const int4, 4>(pi_tilde), cuda::std::span<const int4, 4>(cs[tid]));
1272 }
1273#pragma unroll
1274 for (int j = 0; j < 4; ++j) pi[tid][j] = pi_tilde[j];
1275}
