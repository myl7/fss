112 static constexpr int kDepth = in_bits - [] {
113 int r = 0;
114 int v = kLanes;
115 while (v > 1) {
116 v >>= 1;

189 static __host__ __device__ int4 PackBeta(In a, uint64_t beta) {
190 int lane = static_cast<int>(a) & (kLanes - 1);
191 int4 packed = {0, 0, 0, 0};
192 if constexpr (out_bits == 64) {
193 packed = lane == 0 ? int4{static_cast<int>(beta), static_cast<int>(beta >> 32), 0, 0}
194 : int4{0, 0, static_cast<int>(beta), static_cast<int>(beta >> 32)};
195 } else {
196 constexpr int lanes_per_word = 32 / out_bits;
197 int word = lane / lanes_per_word;
198 int off = (lane % lanes_per_word) * out_bits;
199 unsigned int v = static_cast<unsigned int>(beta) << off;
200 if (word == 0) packed.x = static_cast<int>(v);
201 else if (word == 1) packed.y = static_cast<int>(v);
202 else if (word == 2) packed.z = static_cast<int>(v);
203 else packed.w = static_cast<int>(v);
204 }
205 return packed;
206 }

215 static __host__ __device__ uint64_t Extract(int4 block, In x) {
216 int lane = static_cast<int>(x) & (kLanes - 1);
217 if constexpr (out_bits == 64) {
218 uint64_t lo = lane == 0 ? static_cast<unsigned int>(block.x) : static_cast<unsigned int>(block.z);
219 uint64_t hi = lane == 0 ? static_cast<unsigned int>(block.y) : static_cast<unsigned int>(block.w);
220 return lo | (hi << 32);
221 } else {
222 constexpr int lanes_per_word = 32 / out_bits;
223 int word = lane / lanes_per_word;
224 int off = (lane % lanes_per_word) * out_bits;
225 unsigned int w = word == 0 ? static_cast<unsigned int>(block.x)
226 : word == 1 ? static_cast<unsigned int>(block.y)
227 : word == 2 ? static_cast<unsigned int>(block.z)
228 : static_cast<unsigned int>(block.w);
229 return (w >> off) & static_cast<unsigned int>((1ULL << out_bits) - 1);
230 }
231 }

254 __host__ __device__ void Gen(Cw cws[], int4 &fcw, const int4 s0s[2], In a, uint64_t beta) {
255 assert(beta <= kBetaMax);
256
257 int4 packed_beta = PackBeta(a, beta);
258
259 // node1 starts with the control bit set, so the path difference delta keeps LSB 1 and
260 // exactly one party has t=1 on the path at every level.
261 int4 node0 = util::SetLsb(s0s[0], false);
262 int4 node1 = util::SetLsb(s0s[1], true);
263 int4 delta = util::Xor(node0, node1); // LSB = 0^1 = 1
264
265 for (int i = 0; i < kDepth; ++i) {
266 int4 h0 = prg.Gen(util::Xor(hash_key, node0))[0];
267 int4 h1 = prg.Gen(util::Xor(hash_key, node1))[0];
268
269 bool a_bit = (a >> (in_bits - 1 - i)) & 1;
270
271 // CW = h0 ^ h1 ^ (!a_bit ? delta : 0)
272 // When a_bit=0 (go left): non-alpha is right, CW = h0^h1^delta makes right0=right1
273 // When a_bit=1 (go right): non-alpha is left, CW = h0^h1 makes left0=left1
274 int4 cw = util::Xor(h0, h1);
275 if (!a_bit) cw = util::Xor(cw, delta);
276
277 cws[i] = {cw};
278
279 bool t0 = util::GetLsb(node0);
280 bool t1 = util::GetLsb(node1);
281
282 // node_b = h_b ^ (a_bit ? node_b : 0) ^ (t_b ? cw : 0)
283 int4 zero4 = {0, 0, 0, 0};
284 int4 ab_mask0 = a_bit ? node0 : zero4;
285 int4 ab_mask1 = a_bit ? node1 : zero4;
286 int4 t0_mask = t0 ? cw : zero4;
287 int4 t1_mask = t1 ? cw : zero4;
288
289 node0 = util::Xor(util::Xor(h0, ab_mask0), t0_mask);
290 node1 = util::Xor(util::Xor(h1, ab_mask1), t1_mask);
291
292 // delta = node0 ^ node1 for the next level; stays the root seed XOR
293 delta = util::Xor(node0, node1);
294 }
295
296 // The leaf hash masks the constant path difference, see the file-level details.
297 int4 leaf0 = prg.Gen(util::Xor(hash_key, node0))[0];
298 int4 leaf1 = prg.Gen(util::Xor(hash_key, node1))[0];
299 fcw = util::Xor(util::Xor(packed_beta, leaf0), leaf1);
300 }

307 // Walks x's path to its level-kDepth node. visit(t, x_bit) sees the control bit of every
308 // expanded node and the branch x takes from it.
309 template <typename Visit>
310 __host__ __device__ int4 Walk(bool b, int4 s0, const Cw cws[], In x, Visit &&visit) {
311 int4 node = util::SetLsb(s0, b);
312
313 for (int i = 0; i < kDepth; ++i) {
314 bool x_bit = (x >> (in_bits - 1 - i)) & 1;
315 bool t = util::GetLsb(node);
316 visit(t, x_bit);
317
318 int4 h = prg.Gen(util::Xor(hash_key, node))[0];
319
320 int4 zero4 = {0, 0, 0, 0};
321 int4 xb_mask = x_bit ? node : zero4;
322 int4 t_mask = t ? cws[i].s : zero4;
323
324 node = util::Xor(util::Xor(h, xb_mask), t_mask);
325 }
326
327 return node;
328 }

330 void EvalAll(bool b, int4 s0, const Cw cws[], int4 fcw, int4 ys[]) {
331 if constexpr (kDepth > kTopDepth) {
332 EvalAllFrontier(b, s0, cws, fcw, ys);
333 } else {
334 // Domains too shallow to fill an 8-wide frontier evaluate one point per
335 // block through the tested point walk.
336 constexpr size_t blocks = size_t{1} << kDepth;
337 util::ParallelFor(par_depth, blocks, [&](size_t j) {
338 ys[j] = Eval(b, s0, cws, fcw, static_cast<In>(j * kLanes));
339 });
340 }
341 }

343 // lhs ^ (rhs & mask) with mask being 0 or -1 in every lane, so the per-node corrections
344 // controlled by the seed LSB apply branchlessly.
345 static __host__ __device__ int4 MaskXor(int4 lhs, int4 rhs, int mask) {
346 return {lhs.x ^ (rhs.x & mask), lhs.y ^ (rhs.y & mask), lhs.z ^ (rhs.z & mask), lhs.w ^ (rhs.w & mask)};
347 }

349 // Expands one node with a scalar PRG call: left = H_S(node) ^ (t ? cw : 0), right = left ^ node.
350 __host__ __device__ void ExpandNode(int4 node, int4 cw, int4 &left, int4 &right) {
351 int4 h = prg.Gen(util::Xor(hash_key, node))[0];
352 int t_mask = -static_cast<int>(util::GetLsb(node));
353 left = MaskXor(h, cw, t_mask);
354 right = util::Xor(left, node);
355 }

357 // Level-order full-domain evaluation over a batched frontier. The frontier of
358 // level i lives in bufs[i & 1]; the final expansion fuses the leaf conversion and writes
359 // each ys block exactly once, in the same order the DFS walk produces. The level-order
360 // node index equals the in-order leaf index, so the ys layout is unchanged.
361 void EvalAllFrontier(bool b, int4 s0, const Cw cws[], int4 fcw, int4 ys[]) {
362 assert(kDepth < sizeof(size_t) * 8);
363 size_t num_blocks = 1ULL << kDepth;
364
365 // Frontier ping-pong scratch, grown on demand and reused across calls of this thread.
366 // Two halves of num_blocks / 2 each: no frontier before the last level exceeds that size.
367 // thread_local isolates concurrent calls: each thread pays the first-call allocation and
368 // retains the peak 16 * 2 ** kDepth bytes, about one packed output buffer, until it exits.
369 static thread_local std::vector<int4> scratch;
370 if (scratch.size() < num_blocks) scratch.resize(num_blocks);
371 int4 *bufs[2] = {scratch.data(), scratch.data() + num_blocks / 2};
372
373 // Scalar top: the root becomes the 2^kTopDepth level-kTopDepth frontier in bufs[kTopDepth & 1].
374 bufs[0][0] = util::SetLsb(s0, b);
375 for (int i = 0; i < kTopDepth; ++i) {
376 size_t count = size_t{1} << i;
377 for (size_t j = 0; j < count; ++j) {
378 ExpandNode(bufs[i & 1][j], cws[i].s, bufs[(i + 1) & 1][2 * j], bufs[(i + 1) & 1][2 * j + 1]);
379 }
380 }
381
382 // Each level bounds its independent partitions by 2^par_depth. The
383 // sequential branch avoids an inactive OpenMP region as well as workers.
384 int depth = util::ResolveParDepth(par_depth, kDepth);
385 if (depth == 0) {
386 for (int i = kTopDepth; i < kDepth; ++i) {
387 const int4 *src = bufs[i & 1];
388 int4 *dst = i + 1 == kDepth ? ys : bufs[(i + 1) & 1];
389 size_t batches = (size_t{1} << i) / 8;
390 for (size_t k = 0; k < batches; ++k) {
391 if (i + 1 == kDepth) ExpandConvert8(src, dst, k, cws[i].s, fcw);
392 else Expand8(src, dst, k, cws[i].s);
393 }
394 }
395 return;
396 }
397 int threads = util::ResolveParThreads(depth, num_blocks / 16);
398#pragma omp parallel num_threads(threads)
399 {
400 for (int i = kTopDepth; i < kDepth; ++i) {
401 const int4 *src = bufs[i & 1];
402 int4 *dst = i + 1 == kDepth ? ys : bufs[(i + 1) & 1];
403 size_t batches = (size_t{1} << i) / 8;
404 size_t partitions = util::ResolveParPartitions(depth, batches);
405#pragma omp for schedule(static)
406 for (size_t partition = 0; partition < partitions; ++partition) {
407 size_t begin = partition * (batches / partitions) +
408 (partition < batches % partitions ? partition : batches % partitions);
409 size_t end = begin + batches / partitions + (partition < batches % partitions);
410 for (size_t k = begin; k < end; ++k) {
411 if (i + 1 == kDepth) ExpandConvert8(src, dst, k, cws[i].s, fcw);
412 else Expand8(src, dst, k, cws[i].s);
413 }
414 }
415 }
416 }
417 }

419 // Expands the 8 frontier parents src[8k..8k+8) into their 16 children with one batch.
420 // Parent p has children 2p and 2p + 1, so the children of a batch interleave in pairs.
421 void Expand8(const int4 *src, int4 *dst, size_t k, int4 cw) {
422 int4 parents[8];
423 cuda::std::array<int4, 8> seeds;
424 for (int j = 0; j < 8; ++j) {
425 parents[j] = src[8 * k + j];
426 seeds[j] = util::Xor(hash_key, parents[j]);
427 }
428 auto hs = detail::GenBatch<1>(prg, seeds);
429 for (int j = 0; j < 8; ++j) {
430 int4 left = MaskXor(hs[j][0], cw, -static_cast<int>(util::GetLsb(parents[j])));
431 dst[16 * k + 2 * j] = left;
432 dst[16 * k + 2 * j + 1] = util::Xor(left, parents[j]);
433 }
434 }
435
436 // Last-level Expand8() with the leaf conversion fused in: the children are the level
437 // kDepth nodes, so they are hashed straight from registers into two more batches
438 // (even and odd children), fcw applies through the child LSB mask, and each final block
439 // is stored exactly once.
440 void ExpandConvert8(const int4 *src, int4 *ys, size_t k, int4 cw, int4 fcw) {
441 int4 parents[8];
442 cuda::std::array<int4, 8> seeds;
443 for (int j = 0; j < 8; ++j) {
444 parents[j] = src[8 * k + j];
445 seeds[j] = util::Xor(hash_key, parents[j]);
446 }
447 auto hs = detail::GenBatch<1>(prg, seeds);
448
449 int4 children[16];
450 cuda::std::array<int4, 8> conv_seeds[2];
451 for (int j = 0; j < 8; ++j) {
452 int4 left = MaskXor(hs[j][0], cw, -static_cast<int>(util::GetLsb(parents[j])));
453 children[2 * j] = left;
454 children[2 * j + 1] = util::Xor(left, parents[j]);
455 conv_seeds[0][j] = util::Xor(hash_key, children[2 * j]);
456 conv_seeds[1][j] = util::Xor(hash_key, children[2 * j + 1]);
457 }
458 auto leaves_even = detail::GenBatch<1>(prg, conv_seeds[0]);
459 auto leaves_odd = detail::GenBatch<1>(prg, conv_seeds[1]);
460 for (int j = 0; j < 8; ++j) {
461 ys[16 * k + 2 * j] = MaskXor(leaves_even[j][0], fcw, -static_cast<int>(util::GetLsb(children[2 * j])));
462 ys[16 * k + 2 * j + 1] =
463 MaskXor(leaves_odd[j][0], fcw, -static_cast<int>(util::GetLsb(children[2 * j + 1])));
464 }
465 }

467 __host__ __device__ int4 ConvertLeaf(int4 node, int4 fcw) {
468 int4 leaf = prg.Gen(util::Xor(hash_key, node))[0];
469 if (util::GetLsb(node)) leaf = util::Xor(leaf, fcw);
470 return leaf;
