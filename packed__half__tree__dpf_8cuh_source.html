93class PackedHalfTreeDpf {

104 static constexpr int kLanes = 128 / out_bits;

109 static constexpr int kDepth = in_bits - [] {
110 int r = 0;
111 int v = kLanes;
112 while (v > 1) {
113 v >>= 1;
114 ++r;
115 }
116 return r;
117 }();
118 static_assert(kDepth >= 1, "in_bits must exceed log2(kLanes) so that the tree has at least one level");

123 static constexpr uint64_t kBetaMax = out_bits == 64 ? ~0ULL : (1ULL << out_bits) - 1;

186 static __host__ __device__ int4 PackBeta(In a, uint64_t beta) {
187 int lane = static_cast<int>(a) & (kLanes - 1);
188 int4 packed = {0, 0, 0, 0};
189 if constexpr (out_bits == 64) {
190 packed = lane == 0 ? int4{static_cast<int>(beta), static_cast<int>(beta >> 32), 0, 0}
191 : int4{0, 0, static_cast<int>(beta), static_cast<int>(beta >> 32)};
192 } else {
193 constexpr int lanes_per_word = 32 / out_bits;
194 int word = lane / lanes_per_word;
195 int off = (lane % lanes_per_word) * out_bits;
196 unsigned int v = static_cast<unsigned int>(beta) << off;
197 if (word == 0) packed.x = static_cast<int>(v);
198 else if (word == 1) packed.y = static_cast<int>(v);
199 else if (word == 2) packed.z = static_cast<int>(v);
200 else packed.w = static_cast<int>(v);
201 }
202 return packed;
203 }

212 static __host__ __device__ uint64_t Extract(int4 block, In x) {
213 int lane = static_cast<int>(x) & (kLanes - 1);
214 if constexpr (out_bits == 64) {
215 uint64_t lo = lane == 0 ? static_cast<unsigned int>(block.x) : static_cast<unsigned int>(block.z);
216 uint64_t hi = lane == 0 ? static_cast<unsigned int>(block.y) : static_cast<unsigned int>(block.w);
217 return lo | (hi << 32);
218 } else {
219 constexpr int lanes_per_word = 32 / out_bits;
220 int word = lane / lanes_per_word;
221 int off = (lane % lanes_per_word) * out_bits;
222 unsigned int w = word == 0 ? static_cast<unsigned int>(block.x)
223 : word == 1 ? static_cast<unsigned int>(block.y)
224 : word == 2 ? static_cast<unsigned int>(block.z)
225 : static_cast<unsigned int>(block.w);
226 return (w >> off) & static_cast<unsigned int>((1ULL << out_bits) - 1);
227 }
228 }

248 __host__ __device__ void Gen(Cw cws[], int4 &fcw, const int4 s0s[2], In a, uint64_t beta) {
249 assert(beta <= kBetaMax);
250
251 int4 packed_beta = PackBeta(a, beta);
252
253 // node1 starts with the control bit set, so the path difference delta keeps LSB 1 and
254 // exactly one party has t=1 on the path at every level.
255 int4 node0 = util::SetLsb(s0s[0], false);
256 int4 node1 = util::SetLsb(s0s[1], true);
257 int4 delta = util::Xor(node0, node1); // LSB = 0^1 = 1
258
259 for (int i = 0; i < kDepth; ++i) {
260 int4 h0 = prg.Gen(util::Xor(hash_key, node0))[0];
261 int4 h1 = prg.Gen(util::Xor(hash_key, node1))[0];
262
263 bool a_bit = (a >> (in_bits - 1 - i)) & 1;
264
265 // CW = h0 ^ h1 ^ (!a_bit ? delta : 0)
266 // When a_bit=0 (go left): non-alpha is right, CW = h0^h1^delta makes right0=right1
267 // When a_bit=1 (go right): non-alpha is left, CW = h0^h1 makes left0=left1
268 int4 cw = util::Xor(h0, h1);
269 if (!a_bit) cw = util::Xor(cw, delta);
270
271 cws[i] = {cw};
272
273 bool t0 = util::GetLsb(node0);
274 bool t1 = util::GetLsb(node1);
275
276 // node_b = h_b ^ (a_bit ? node_b : 0) ^ (t_b ? cw : 0)
277 int4 zero4 = {0, 0, 0, 0};
278 int4 ab_mask0 = a_bit ? node0 : zero4;
279 int4 ab_mask1 = a_bit ? node1 : zero4;
280 int4 t0_mask = t0 ? cw : zero4;
281 int4 t1_mask = t1 ? cw : zero4;
282
283 node0 = util::Xor(util::Xor(h0, ab_mask0), t0_mask);
284 node1 = util::Xor(util::Xor(h1, ab_mask1), t1_mask);
285
286 // delta = node0 ^ node1 for the next level; stays the root seed XOR
287 delta = util::Xor(node0, node1);
288 }
289
290 // The leaf hash masks the constant path difference, see the file-level details.
291 int4 leaf0 = prg.Gen(util::Xor(hash_key, node0))[0];
292 int4 leaf1 = prg.Gen(util::Xor(hash_key, node1))[0];
293 fcw = util::Xor(util::Xor(packed_beta, leaf0), leaf1);
294 }

296 // Seed + correction-words form used by the internal recursion and tests.
297 __host__ __device__ int4 Eval(bool b, int4 s0, const Cw cws[], int4 fcw, In x) {
298 int4 node = util::SetLsb(s0, b);
299
300 for (int i = 0; i < kDepth; ++i) {
301 bool x_bit = (x >> (in_bits - 1 - i)) & 1;
302 bool t = util::GetLsb(node);
303
304 int4 h = prg.Gen(util::Xor(hash_key, node))[0];
305
306 int4 zero4 = {0, 0, 0, 0};
307 int4 xb_mask = x_bit ? node : zero4;
308 int4 t_mask = t ? cws[i].s : zero4;
309
310 node = util::Xor(util::Xor(h, xb_mask), t_mask);
311 }
312
313 return ConvertLeaf(node, fcw);
314 }

316 void EvalAll(bool b, int4 s0, const Cw cws[], int4 fcw, int4 ys[]) {
317 if constexpr (kDepth > kTopDepth) {
318 EvalAllFrontier(b, s0, cws, fcw, ys);
319 } else {
320 // Domains too shallow to fill an 8-wide frontier evaluate one point per
321 // block through the tested point walk.
322 constexpr size_t blocks = size_t{1} << kDepth;
323 util::ParallelFor(par_depth, blocks, [&](size_t j) {
324 ys[j] = Eval(b, s0, cws, fcw, static_cast<In>(j * kLanes));
325 });
326 }
327 }

329 // lhs ^ (rhs & mask) with mask being 0 or -1 in every lane, so the per-node corrections
330 // controlled by the seed LSB apply branchlessly.
331 static __host__ __device__ int4 MaskXor(int4 lhs, int4 rhs, int mask) {
332 return {lhs.x ^ (rhs.x & mask), lhs.y ^ (rhs.y & mask), lhs.z ^ (rhs.z & mask), lhs.w ^ (rhs.w & mask)};
333 }

335 // Expands one node with a scalar PRG call: left = H_S(node) ^ (t ? cw : 0), right = left ^ node.
336 __host__ __device__ void ExpandNode(int4 node, int4 cw, int4 &left, int4 &right) {
337 int4 h = prg.Gen(util::Xor(hash_key, node))[0];
338 int t_mask = -static_cast<int>(util::GetLsb(node));
339 left = MaskXor(h, cw, t_mask);
340 right = util::Xor(left, node);
341 }

343 // Level-order full-domain evaluation over a batched frontier. The frontier of
344 // level i lives in bufs[i & 1]; the final expansion fuses the leaf conversion and writes
345 // each ys block exactly once, in the same order the DFS walk produces. The level-order
346 // node index equals the in-order leaf index, so the ys layout is unchanged.
347 void EvalAllFrontier(bool b, int4 s0, const Cw cws[], int4 fcw, int4 ys[]) {
348 assert(kDepth < sizeof(size_t) * 8);
349 size_t num_blocks = 1ULL << kDepth;
350
351 // Frontier ping-pong scratch, grown on demand and reused across calls of this thread.
352 // Two halves of num_blocks / 2 each: no frontier before the last level exceeds that size.
353 // thread_local isolates concurrent calls: each thread pays the first-call allocation and
354 // retains the peak 16 * 2 ** kDepth bytes, about one packed output buffer, until it exits.
355 static thread_local std::vector<int4> scratch;
356 if (scratch.size() < num_blocks) scratch.resize(num_blocks);
357 int4 *bufs[2] = {scratch.data(), scratch.data() + num_blocks / 2};
358
359 // Scalar top: the root becomes the 2^kTopDepth level-kTopDepth frontier in bufs[kTopDepth & 1].
360 bufs[0][0] = util::SetLsb(s0, b);
361 for (int i = 0; i < kTopDepth; ++i) {
362 size_t count = size_t{1} << i;
363 for (size_t j = 0; j < count; ++j) {
364 ExpandNode(bufs[i & 1][j], cws[i].s, bufs[(i + 1) & 1][2 * j], bufs[(i + 1) & 1][2 * j + 1]);
365 }
366 }
367
368 // Each level bounds its independent partitions by 2^par_depth. The
369 // sequential branch avoids an inactive OpenMP region as well as workers.
370 int depth = util::ResolveParDepth(par_depth, kDepth);
371 if (depth == 0) {
372 for (int i = kTopDepth; i < kDepth; ++i) {
373 const int4 *src = bufs[i & 1];
374 int4 *dst = i + 1 == kDepth ? ys : bufs[(i + 1) & 1];
375 size_t batches = (size_t{1} << i) / 8;
376 for (size_t k = 0; k < batches; ++k) {
377 if (i + 1 == kDepth) ExpandConvert8(src, dst, k, cws[i].s, fcw);
378 else Expand8(src, dst, k, cws[i].s);
379 }
380 }
381 return;
382 }
383 int threads = util::ResolveParThreads(depth, num_blocks / 16);
384#pragma omp parallel num_threads(threads)
385 {
386 for (int i = kTopDepth; i < kDepth; ++i) {
387 const int4 *src = bufs[i & 1];
388 int4 *dst = i + 1 == kDepth ? ys : bufs[(i + 1) & 1];
389 size_t batches = (size_t{1} << i) / 8;
390 size_t partitions = util::ResolveParPartitions(depth, batches);
391#pragma omp for schedule(static)
392 for (size_t partition = 0; partition < partitions; ++partition) {
393 size_t begin = partition * (batches / partitions) +
394 (partition < batches % partitions ? partition : batches % partitions);
395 size_t end = begin + batches / partitions + (partition < batches % partitions);
396 for (size_t k = begin; k < end; ++k) {
397 if (i + 1 == kDepth) ExpandConvert8(src, dst, k, cws[i].s, fcw);
398 else Expand8(src, dst, k, cws[i].s);
399 }
400 }
401 }
402 }
403 }

405 // Expands the 8 frontier parents src[8k..8k+8) into their 16 children with one batch.
406 // Parent p has children 2p and 2p + 1, so the children of a batch interleave in pairs.
407 void Expand8(const int4 *src, int4 *dst, size_t k, int4 cw) {
408 int4 parents[8];
409 cuda::std::array<int4, 8> seeds;
410 for (int j = 0; j < 8; ++j) {
411 parents[j] = src[8 * k + j];
412 seeds[j] = util::Xor(hash_key, parents[j]);
413 }
414 auto hs = detail::GenBatch<1>(prg, seeds);
415 for (int j = 0; j < 8; ++j) {
416 int4 left = MaskXor(hs[j][0], cw, -static_cast<int>(util::GetLsb(parents[j])));
417 dst[16 * k + 2 * j] = left;
418 dst[16 * k + 2 * j + 1] = util::Xor(left, parents[j]);
419 }
420 }
421
422 // Last-level Expand8() with the leaf conversion fused in: the children are the level
423 // kDepth nodes, so they are hashed straight from registers into two more batches
424 // (even and odd children), fcw applies through the child LSB mask, and each final block
425 // is stored exactly once.
426 void ExpandConvert8(const int4 *src, int4 *ys, size_t k, int4 cw, int4 fcw) {
427 int4 parents[8];
428 cuda::std::array<int4, 8> seeds;
429 for (int j = 0; j < 8; ++j) {
430 parents[j] = src[8 * k + j];
431 seeds[j] = util::Xor(hash_key, parents[j]);
432 }
433 auto hs = detail::GenBatch<1>(prg, seeds);
434
435 int4 children[16];
436 cuda::std::array<int4, 8> conv_seeds[2];
437 for (int j = 0; j < 8; ++j) {
438 int4 left = MaskXor(hs[j][0], cw, -static_cast<int>(util::GetLsb(parents[j])));
439 children[2 * j] = left;
440 children[2 * j + 1] = util::Xor(left, parents[j]);
441 conv_seeds[0][j] = util::Xor(hash_key, children[2 * j]);
442 conv_seeds[1][j] = util::Xor(hash_key, children[2 * j + 1]);
443 }
444 auto leaves_even = detail::GenBatch<1>(prg, conv_seeds[0]);
445 auto leaves_odd = detail::GenBatch<1>(prg, conv_seeds[1]);
446 for (int j = 0; j < 8; ++j) {
447 ys[16 * k + 2 * j] = MaskXor(leaves_even[j][0], fcw, -static_cast<int>(util::GetLsb(children[2 * j])));
448 ys[16 * k + 2 * j + 1] =
449 MaskXor(leaves_odd[j][0], fcw, -static_cast<int>(util::GetLsb(children[2 * j + 1])));
450 }
451 }

453 __host__ __device__ int4 ConvertLeaf(int4 node, int4 fcw) {
454 int4 leaf = prg.Gen(util::Xor(hash_key, node))[0];
455 if (util::GetLsb(node)) leaf = util::Xor(leaf, fcw);
456 return leaf;
