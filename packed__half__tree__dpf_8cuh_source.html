88// Split Cws view shared by the host and device split paths. The packed
89// construction carries no control bits, so layer l is just cws[l*stride] and the
90// terminal entry holds the packed-output correction word at cws[kDepth*stride].
91// operator[] recovers the Cw the shared algorithms consume, so their bodies stay
92// identical on the Cw and split paths.
93template <typename Cw>
94struct PackedHalfTreeCwsView {
95 const int4 *cws;
96 size_t stride;
97
98 __host__ __device__ inline Cw operator[](int l) const {
99 return Cw{cws[static_cast<size_t>(l) * stride]};
100 }
101};

475template <int in_bits, int out_bits, typename Hash, typename In>
476 requires(ValidForInputBits<In, in_bits> &&
477 CcrHashable<Hash> && out_bits >= 1 && out_bits <= 64 && (out_bits & (out_bits - 1)) == 0)
478__host__ __device__ inline int4 PackedHalfTreeDpf<in_bits, out_bits, Hash, In>::PackBeta(
479 In a, uint64_t beta) {
480 int lane = static_cast<int>(a) & (kLanes - 1);
481 int4 packed = {0, 0, 0, 0};
482 if constexpr (out_bits == 64) {
483 packed = lane == 0 ? int4{static_cast<int>(beta), static_cast<int>(beta >> 32), 0, 0}
484 : int4{0, 0, static_cast<int>(beta), static_cast<int>(beta >> 32)};
485 } else {
486 constexpr int lanes_per_word = 32 / out_bits;
487 int word = lane / lanes_per_word;
488 int off = (lane % lanes_per_word) * out_bits;
489 unsigned int v = static_cast<unsigned int>(beta) << off;
490 if (word == 0) packed.x = static_cast<int>(v);
491 else if (word == 1) packed.y = static_cast<int>(v);
492 else if (word == 2) packed.z = static_cast<int>(v);
493 else packed.w = static_cast<int>(v);
494 }
495 return packed;
496}

498template <int in_bits, int out_bits, typename Hash, typename In>
499 requires(ValidForInputBits<In, in_bits> &&
500 CcrHashable<Hash> && out_bits >= 1 && out_bits <= 64 && (out_bits & (out_bits - 1)) == 0)
501__host__ __device__ inline uint64_t PackedHalfTreeDpf<in_bits, out_bits, Hash, In>::Extract(
502 int4 block, In x) {
503 int lane = static_cast<int>(x) & (kLanes - 1);
504 if constexpr (out_bits == 64) {
505 uint64_t lo = lane == 0 ? static_cast<unsigned int>(block.x) : static_cast<unsigned int>(block.z);
506 uint64_t hi = lane == 0 ? static_cast<unsigned int>(block.y) : static_cast<unsigned int>(block.w);
507 return lo | (hi << 32);
508 } else {
509 constexpr int lanes_per_word = 32 / out_bits;
510 int word = lane / lanes_per_word;
511 int off = (lane % lanes_per_word) * out_bits;
512 unsigned int w = word == 0 ? static_cast<unsigned int>(block.x)
513 : word == 1 ? static_cast<unsigned int>(block.y)
514 : word == 2 ? static_cast<unsigned int>(block.z)
515 : static_cast<unsigned int>(block.w);
516 return (w >> off) & static_cast<unsigned int>((1ULL << out_bits) - 1);
517 }
518}

536template <int in_bits, int out_bits, typename Hash, typename In>
537 requires(ValidForInputBits<In, in_bits> &&
538 CcrHashable<Hash> && out_bits >= 1 && out_bits <= 64 && (out_bits & (out_bits - 1)) == 0)
539template <typename Writer>
540__host__ __device__ inline void PackedHalfTreeDpf<in_bits, out_bits, Hash, In>::Generate(
541 Writer writer, const int4 *s0s, In a, uint64_t beta) {
542 util::RequireArg(beta <= kBetaMax, "PackedHalfTreeDpf::Gen: beta exceeds kBetaMax");
543
544 int4 packed_beta = PackBeta(a, beta);
545
546 // node1 starts with the control bit set, so the path difference delta keeps LSB 1 and
547 // exactly one party has t=1 on the path at every level.
548 int4 node0 = util::SetLsb(s0s[0], false);
549 int4 node1 = util::SetLsb(s0s[1], true);
550 int4 delta = util::Xor(node0, node1); // LSB = 0^1 = 1
551
552 for (int i = 0; i < kDepth; ++i) {
553 int4 h0 = hash_.Hash(util::Xor(instance_seed_, node0));
554 int4 h1 = hash_.Hash(util::Xor(instance_seed_, node1));
555
556 bool a_bit = (a >> (in_bits - 1 - i)) & 1;
557
558 // CW = h0 ^ h1 ^ (!a_bit ? delta : 0)
559 // When a_bit=0 (go left): non-alpha is right, CW = h0^h1^delta makes right0=right1
560 // When a_bit=1 (go right): non-alpha is left, CW = h0^h1 makes left0=left1
561 int4 cw = util::Xor(h0, h1);
562 if (!a_bit) cw = util::Xor(cw, delta);
563
564 writer.Write(i, cw);
565
566 bool t0 = util::GetLsb(node0);
567 bool t1 = util::GetLsb(node1);
568
569 // node_b = h_b ^ (a_bit ? node_b : 0) ^ (t_b ? cw : 0)
570 int4 zero4 = {0, 0, 0, 0};
571 int4 ab_mask0 = a_bit ? node0 : zero4;
572 int4 ab_mask1 = a_bit ? node1 : zero4;
573 int4 t0_mask = t0 ? cw : zero4;
574 int4 t1_mask = t1 ? cw : zero4;
575
576 node0 = util::Xor(util::Xor(h0, ab_mask0), t0_mask);
577 node1 = util::Xor(util::Xor(h1, ab_mask1), t1_mask);
578
579 // delta = node0 ^ node1 for the next level; stays the root seed XOR
580 delta = util::Xor(node0, node1);
581 }
582
583 // The leaf hash masks the constant path difference, see the file-level details.
584 int4 leaf0 = hash_.Hash(util::Xor(instance_seed_, node0));
585 int4 leaf1 = hash_.Hash(util::Xor(instance_seed_, node1));
586 int4 fcw = util::Xor(util::Xor(packed_beta, leaf0), leaf1);
587 writer.Finish(fcw);
588}

614template <int in_bits, int out_bits, typename Hash, typename In>
615 requires(ValidForInputBits<In, in_bits> &&
616 CcrHashable<Hash> && out_bits >= 1 && out_bits <= 64 && (out_bits & (out_bits - 1)) == 0)
617template <typename Cws, typename Visit>
618__host__ __device__ inline int4 PackedHalfTreeDpf<in_bits, out_bits, Hash, In>::Walk(
619 bool b, int4 s0, Cws cws, In x, Visit &&visit) {
620 int4 node = util::SetLsb(s0, b);
621
622 for (int i = 0; i < kDepth; ++i) {
623 bool x_bit = (x >> (in_bits - 1 - i)) & 1;
624 bool t = util::GetLsb(node);
625 visit(t, x_bit);
626
627 node = PathStep(hash_, instance_seed_, node, cws[i].s, x_bit);
628 }
629
630 return node;
631}

690template <int in_bits, int out_bits, typename Hash, typename In>
691 requires(ValidForInputBits<In, in_bits> &&
692 CcrHashable<Hash> && out_bits >= 1 && out_bits <= 64 && (out_bits & (out_bits - 1)) == 0)
693template <typename Cws>
694inline void PackedHalfTreeDpf<in_bits, out_bits, Hash, In>::EvalRangeCws(
695 bool b, In first, In last, int4 s0, Cws cws,
696 cuda::std::span<int4> ys, int par_depth) {
697 assert(par_depth >= 0);
698 In domain_last = static_cast<In>(~In{0});
699 if constexpr (in_bits < sizeof(In) * 8) {
700 domain_last = static_cast<In>((In{1} << in_bits) - 1);
701 }
702 if (first > last)
703 throw std::invalid_argument("PackedHalfTreeDpf::EvalRange: first must not exceed last");
704 if (last > domain_last)
705 throw std::invalid_argument("PackedHalfTreeDpf::EvalRange: last must stay inside the input domain");
706 const __uint128_t distance = static_cast<__uint128_t>(last) - first;
707 const __uint128_t block_distance = distance / kLanes;
708 // The caller controls the interval, so this is an argument check, not an
709 // invariant: with a 128-bit input domain the block distance can exceed
710 // size_t, and narrowing it first would shrink the output-size check.
711 if (block_distance >= std::numeric_limits<size_t>::max() / sizeof(int4))
712 throw std::invalid_argument("PackedHalfTreeDpf::EvalRange: the interval is too large for size_t");
713 const size_t blocks = static_cast<size_t>(block_distance) + 1;
714 if (ys.size() < blocks || ys.data() == nullptr)
715 throw std::invalid_argument("PackedHalfTreeDpf::EvalRange: ys must hold the covered blocks");
716
717 const size_t partitions = detail::ResolveParPartitions(par_depth, blocks);
718 detail::ParallelFor(par_depth, partitions, [&](size_t partition) {
719 const size_t begin = partition * (blocks / partitions) +
720 (partition < blocks % partitions ? partition : blocks % partitions);
721 const size_t end = begin + blocks / partitions + (partition < blocks % partitions);
722 for (size_t index = begin; index < end; ++index) ys[index] = {0, 0, 0, 0};
723 const __uint128_t begin_offset = static_cast<__uint128_t>(begin) * kLanes;
724 const __uint128_t end_offset = end == blocks ? distance :
725 static_cast<__uint128_t>(end) * kLanes - 1;
726 EvalRangeTree(util::SetLsb(s0, b), cws, ys.data(),
727 static_cast<In>(first + begin_offset), static_cast<In>(first + end_offset),
728 first, In{0}, domain_last, 0);
729 });
730}

738 In output_first, In low, In high, int level) {
739 if (level == kDepth) {
740 const int4 block = ConvertLeaf(node, cws[kDepth].s);
741 const In begin = first > low ? first : low;
742 const In end = last < high ? last : high;
743 const size_t lanes = static_cast<size_t>(end - begin) + 1;
744 for (size_t lane = 0; lane < lanes; ++lane) {
745 const In x = static_cast<In>(begin + lane);
746 const In offset = static_cast<In>(x - output_first);
747 const size_t index = static_cast<size_t>(offset / kLanes);
748 ys[index] = util::Xor(ys[index], PackBeta(offset, Extract(block, x)));
749 }
750 return;
751 }
752 const In middle = static_cast<In>(low + ((high - low) >> 1));
753 int4 left, right;
754 ExpandNode(node, cws[level].s, left, right);
755 if (first <= middle) {
756 EvalRangeTree(left, cws, ys, first, last, output_first, low, middle, level + 1);
757 }
758 if (last > middle) {
759 EvalRangeTree(right, cws, ys, first, last, output_first,
760 static_cast<In>(middle + 1), high, level + 1);
761 }
762}

780template <int in_bits, int out_bits, typename Hash, typename In>
781 requires(ValidForInputBits<In, in_bits> &&
782 CcrHashable<Hash> && out_bits >= 1 && out_bits <= 64 && (out_bits & (out_bits - 1)) == 0)
783template <typename Cws>
784inline void PackedHalfTreeDpf<in_bits, out_bits, Hash, In>::EvalAllFrontier(
785 bool b, int4 s0, Cws cws, int4 fcw, int4 ys[], int par_depth) {
786 constexpr size_t num_blocks = detail::FullDomainSize<int4, kDepth>();
787
788 // Frontier ping-pong scratch, grown on demand and reused across calls of this thread.
789 // Two halves of num_blocks / 2 each: no frontier before the last level exceeds that size.
790 // thread_local isolates concurrent calls: each thread pays the first-call allocation and
791 // retains the peak 16 * 2 ** kDepth bytes, about one packed output buffer, until it exits.
792 static thread_local std::vector<int4> scratch;
793 if (scratch.size() < num_blocks) scratch.resize(num_blocks);
794 int4 *bufs[2] = {scratch.data(), scratch.data() + num_blocks / 2};
795
796 // Scalar top: the root becomes the 2^kTopDepth level-kTopDepth frontier in bufs[kTopDepth & 1].
797 bufs[0][0] = util::SetLsb(s0, b);
798 for (int i = 0; i < kTopDepth; ++i) {
799 size_t count = size_t{1} << i;
800 for (size_t j = 0; j < count; ++j) {
801 ExpandNode(bufs[i & 1][j], cws[i].s, bufs[(i + 1) & 1][2 * j], bufs[(i + 1) & 1][2 * j + 1]);
802 }
803 }
804
805 // Each level bounds its independent partitions by 2^par_depth. The
806 // sequential branch avoids an inactive OpenMP region as well as workers.
807 int depth = fss::detail::ResolveParDepth(par_depth, kDepth);
808 if (depth == 0) {
809 for (int i = kTopDepth; i < kDepth; ++i) {
810 const int4 *src = bufs[i & 1];
811 int4 *dst = i + 1 == kDepth ? ys : bufs[(i + 1) & 1];
812 size_t batches = (size_t{1} << i) / 8;
813 for (size_t k = 0; k < batches; ++k) {
814 if (i + 1 == kDepth) ExpandConvert8(src, dst, k, cws[i].s, fcw);
815 else Expand8(src, dst, k, cws[i].s);
816 }
817 }
818 return;
819 }
820 int threads = fss::detail::ResolveParThreads(depth, num_blocks / 16);
821#pragma omp parallel num_threads(threads)
822 {
823 for (int i = kTopDepth; i < kDepth; ++i) {
824 const int4 *src = bufs[i & 1];
825 int4 *dst = i + 1 == kDepth ? ys : bufs[(i + 1) & 1];
826 size_t batches = (size_t{1} << i) / 8;
827 size_t partitions = fss::detail::ResolveParPartitions(depth, batches);
828#pragma omp for schedule(static)
829 for (size_t partition = 0; partition < partitions; ++partition) {
830 size_t begin = partition * (batches / partitions) +
831 (partition < batches % partitions ? partition : batches % partitions);
832 size_t end = begin + batches / partitions + (partition < batches % partitions);
833 for (size_t k = begin; k < end; ++k) {
834 if (i + 1 == kDepth) ExpandConvert8(src, dst, k, cws[i].s, fcw);
835 else Expand8(src, dst, k, cws[i].s);
836 }
837 }
838 }
839 }
840}

862 requires(ValidForInputBits<In, in_bits> &&
863 CcrHashable<Hash> && out_bits >= 1 && out_bits <= 64 && (out_bits & (out_bits - 1)) == 0)
864inline void PackedHalfTreeDpf<in_bits, out_bits, Hash, In>::ExpandConvert8(
865 const int4 *src, int4 *ys, size_t k, int4 cw, int4 fcw) {
866 int4 parents[8];
867 cuda::std::array<int4, 8> seeds;
868 for (int j = 0; j < 8; ++j) {
869 parents[j] = src[8 * k + j];
870 seeds[j] = util::Xor(instance_seed_, parents[j]);
871 }
872 auto hs = detail::HashBatch(hash_, seeds);
873
874 int4 children[16];
875 cuda::std::array<int4, 8> conv_seeds[2];
876 for (int j = 0; j < 8; ++j) {
877 int4 left = MaskXor(hs[j], cw, -static_cast<int>(util::GetLsb(parents[j])));
878 children[2 * j] = left;
879 children[2 * j + 1] = util::Xor(left, parents[j]);
880 conv_seeds[0][j] = util::Xor(instance_seed_, children[2 * j]);
881 conv_seeds[1][j] = util::Xor(instance_seed_, children[2 * j + 1]);
882 }
883 auto leaves_even = detail::HashBatch(hash_, conv_seeds[0]);
884 auto leaves_odd = detail::HashBatch(hash_, conv_seeds[1]);
885 for (int j = 0; j < 8; ++j) {
886 ys[16 * k + 2 * j] = MaskXor(leaves_even[j], fcw, -static_cast<int>(util::GetLsb(children[2 * j])));
887 ys[16 * k + 2 * j + 1] =
888 MaskXor(leaves_odd[j], fcw, -static_cast<int>(util::GetLsb(children[2 * j + 1])));
889 }

1045// Point-evaluate packed HalfTree keys, one thread per key: ys[tid] is x's output
1046// block, or with cmp the 1[alpha <= x] share of HalfTreeGrottoDcf::Eval(). cws
1047// holds dense level-major correction words, the layout the split form of Gen()
1048// writes: layer i of key k is cws[i * nkeys + k] and the final correction word
1049// is the tail row, cws[kDepth * nkeys + k].
1050template <int in_bits, int unroll, bool b, bool cmp, typename Scheme, typename In, typename Query, typename Y>
1051__global__ void PackedHalfTreeDpfEvalPointKernel(Y *ys, const int4 *seeds, const int4 *cws, const Query *xs,
1052 int nkeys, typename HalfTreeGpuTraits<Scheme>::Backend hash, int4 instance_seed) {
1053 int tid = blockIdx.x * blockDim.x + threadIdx.x;
1054 if (tid >= nkeys) return;
1055 int4 node = fss::util::SetLsb(seeds[tid], b);
1056 In x = static_cast<In>(xs[tid]);
1057 fss::detail::GrottoTurns turns;
1058
1059#pragma unroll unroll
1060 for (int i = 0; i < Scheme::kDepth; ++i) {
1061 bool x_bit = GetBit(x, in_bits - 1 - i);
1062 bool t = fss::util::GetLsb(node);
1063 if constexpr (cmp) turns(t, x_bit);
1064 int4 h = hash.Hash(fss::util::Xor(instance_seed, node));
1065 int4 cw = cws[(size_t)i * nkeys + tid];
1066 node = fss::util::Xor(fss::util::Xor(h, x_bit ? node : Zero4()), t ? cw : Zero4());
1067 }
1068
1069 int4 leaf = hash.Hash(fss::util::Xor(instance_seed, node));
1070 if (fss::util::GetLsb(node)) leaf = fss::util::Xor(leaf, cws[(size_t)Scheme::kDepth * nkeys + tid]);
1071 if constexpr (cmp) {
1072 ys[tid] = turns.share ^ fss::detail::SchemeAccess::HalfTreeGrottoLeafParity<Scheme>(leaf, x);
1073 } else {
1074 ys[tid] = leaf;
1075 }
1076}
