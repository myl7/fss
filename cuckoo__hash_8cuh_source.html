143constexpr double Log2(double x) {
144 if (x <= 0) return -1e308;
145 int e = 0;
146 double m = x;
147 while (m >= 2.0) {
148 m /= 2.0;
149 ++e;
150 }
151 while (m < 1.0) {
152 m *= 2.0;
153 --e;
154 }
155 double y = (m - 1.0) / (m + 1.0);
156 double y2 = y * y, sum = 0, term = y;
157 for (int k = 0; k < 40; ++k) {
158 sum += term / (2 * k + 1);
159 term *= y2;
160 }
161 return e + 2.0 * sum / 0.6931471805599453;
162}

245constexpr int ChBucket(int t, int lambda) {
246 // Remark 1 applies for t >= 30.
247 // For smaller t, the full Lemma 5 formula with Normal CDF factors is needed,
248 // but the paper considers t >= 30 to capture nearly all practical use cases.
249 if (t < 30)
250 throw std::invalid_argument("cuckoo_hash::ChBucket: t must be at least 30");
251 double td = static_cast<double>(t);
252 double e = (static_cast<double>(lambda) + 130.0 + detail::Log2(td)) / 123.5;
253 return static_cast<int>(detail::Ceil(e * td));
254}

258template <typename Prp, typename In>
259 requires Permutable<Prp>
260inline __host__ __device__ LocateResult PrpHash<Prp, In>::Locate(
261 int4 sigma, In x, int k, __uint128_t n, __uint128_t b_size, int choices) {
262 __uint128_t val = static_cast<__uint128_t>(x) + n * k;
263 __uint128_t domain_size = n * static_cast<__uint128_t>(choices);
264 __uint128_t y = prp.Permu(sigma, val, domain_size);
265
266 // Compare the quotient in __uint128_t: a zero b_size has no bucket at all,
267 // and an over-int quotient must saturate rather than wrap negative through
268 // the cast, because the callers filter buckets with `bucket >= bound` and
269 // would accept a negative value before indexing the per-bucket key arrays.
270 if (b_size == 0) {
271 return {kRejectedBucket, 0};
272 }
273 __uint128_t bucket = y / b_size;
274 if (bucket > static_cast<__uint128_t>(kRejectedBucket)) {
275 return {kRejectedBucket, 0};
276 }
277 return {static_cast<int>(bucket), y % b_size};
278}

280template <typename Prp, typename In>
281 requires Permutable<Prp>
282inline int Compact<Prp, In>::Run(std::span<const In> as, int m, int4 sigma, __uint128_t n,
283 __uint128_t b_size, int max_evictions, std::span<std::pair<int, int>> table) {
284 PrpHash<Prp, In> hasher{prp};
285 int t = static_cast<int>(as.size());
286
287 // Initialize table to empty.
288 for (int i = 0; i < m; ++i) {
289 table[i] = {-1, -1};
290 }
291
292 std::mt19937 rng(42);
293
294 for (int omega = 0; omega < t; ++omega) {
295 int cur_idx = omega;
296 int cur_k = static_cast<int>(rng() % 3);
297 int evictions = 0;
298
299 for (;;) {
300 auto [bucket, _index] = hasher.Locate(sigma, as[cur_idx], cur_k, n, b_size);
301 // Clamp bucket to [0, m).
302 bucket = bucket % m;
303
304 if (table[bucket].first == -1) {
305 // Empty slot found.
306 table[bucket] = {cur_idx, cur_k};
307 break;
308 }
309
310 // Evict the existing entry and take its place.
311 int evicted_idx = table[bucket].first;
312 table[bucket] = {cur_idx, cur_k};
313
314 // The evicted element becomes the current element with a new random hash.
315 cur_idx = evicted_idx;
316 cur_k = static_cast<int>(rng() % 3);
317
318 ++evictions;
319 if (evictions > max_evictions) {
320 return 1;
321 }
322 }
323 }
324
325 return 0;
326}
