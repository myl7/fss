71 __host__ __device__ void Gen(Key &k0, Key &k1, cuda::std::span<const int4, 2> s0s, In a) {

125 PreprocessedTree Preprocess(bool b, const Key &key) {
126 constexpr size_t N = size_t{1} << in_bits;
127
128 PreprocessedTree pt;
129 pt.p = std::make_unique<bool[]>(2 * N - 1);
130 pt.b = b;
131
132 // Phase 1: expand tree, write leaf control bits to pt.p[N-1 .. 2N-2]
133 ExpandTree(b, key.s0, key.cws, pt.p.get() + (N - 1));
134
135 // Phase 2a: build parity segment tree bottom-up
136 for (size_t j = N - 2; j < N - 1; --j) {
137 pt.p[j] = pt.p[2 * j + 1] ^ pt.p[2 * j + 2];
138 }
139
140 return pt;
141 }

155 __host__ __device__ static bool Eval(const ParityTree &pt, In x) {
156 constexpr size_t N = 1ULL << in_bits;
157 In e = static_cast<In>(x) + 1;
158
159 // e == 0 means x + 1 overflowed, i.e., e = N (entire domain)
160 if (e == 0 || e == N) return pt.p[0];
161
162 bool pi = false;
163 size_t cur = 0;
164 for (int i = 0; i < in_bits; ++i) {
165 bool e_bit = (e >> (in_bits - 1 - i)) & 1;
166 if (e_bit) {
167 pi ^= pt.p[2 * cur + 1];
168 cur = 2 * cur + 2;
169 } else {
170 cur = 2 * cur + 1;
171 }
172 }
173 return pi;
174 }

201 void EvalAll(bool b, const Key &key, cuda::std::span<bool> ys) {
202 constexpr size_t N = 1ULL << in_bits;
203
204 // Phase 1: expand tree to get leaf control bits into ys[]
205 ExpandTree(b, key.s0, key.cws, ys.data());
206
207 // Phase 2b: prefix-sum scan (running XOR)
208 // ys[x] currently holds leaf x's control bit.
209 // Transform to: ys[x] = XOR of control bits [0..x] = share of 1[alpha <= x].
210 for (size_t x = 1; x < N; ++x) {
211 ys[x] = ys[x] ^ ys[x - 1];
212 }
213 }

216 // Expand the DPF tree and write leaf control bits.
217 void ExpandTree(bool b, int4 s0, const Cw cws[], bool t[]) {
218 int4 st = s0;
219 st = util::SetLsb(st, b);
220
221 assert(in_bits < sizeof(size_t) * 8);
222 size_t l = 0;
223 size_t r = 1ULL << in_bits;
224 int i = 0;
225
226 int par_depth_ = util::ResolveParDepth(par_depth);
227
228#pragma omp parallel
229#pragma omp single
230 ExpandTreeRec(st, cws, t, l, r, i, par_depth_);

233 void ExpandTreeRec(int4 st, const Cw cws[], bool t[], size_t l, size_t r, int i, int par_depth_) {
234 bool tc = util::GetLsb(st);
235 int4 s = st;
236 s = util::SetLsb(s, false);
237
238 if (i == in_bits) {
239 assert(l + 1 == r);
240 t[l] = tc;
241 return;
242 }
243
244 Cw cw = cws[i];
245 int4 s_cw = cw.s;
246 bool tl_cw = util::GetLsb(s_cw);
247 s_cw = util::SetLsb(s_cw, false);
248 bool tr_cw = cw.tr;
249
250 auto [sl, sr] = prg.Gen(s);
251
252 bool tl = util::GetLsb(sl);
253 sl = util::SetLsb(sl, false);
254 bool tr = util::GetLsb(sr);
255 sr = util::SetLsb(sr, false);
256
257 if (tc) {
258 sl = util::Xor(sl, s_cw);
259 sr = util::Xor(sr, s_cw);
260 tl = tl ^ tl_cw;
261 tr = tr ^ tr_cw;
262 }
263
264 int4 stl = sl;
265 stl = util::SetLsb(stl, tl);
266 int4 str = sr;
267 str = util::SetLsb(str, tr);
268
269 size_t mid = (l + r) / 2;
270
271 if (i < par_depth_) {
272#pragma omp task
273 ExpandTreeRec(stl, cws, t, l, mid, i + 1, par_depth_);
274#pragma omp task
275 ExpandTreeRec(str, cws, t, mid, r, i + 1, par_depth_);
276#pragma omp taskwait
277 } else {
278 ExpandTreeRec(stl, cws, t, l, mid, i + 1, par_depth_);
279 ExpandTreeRec(str, cws, t, mid, r, i + 1, par_depth_);
280 }
281 }
