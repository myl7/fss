68// Split correction-word view shared by the host and device split paths. The key is an
69// underlying DPF key with beta equal to zero, so layer l is read from
70// cws[l*stride] with the right-control correction in bit l of packed_tr. Bits at
71// or above in_bits are ignored because no control correction is read from the
72// terminal entry. operator[] recovers the Cw the shared algorithms consume, so
73// their bodies stay identical on the Cw and split paths.
74template <typename Cw, typename In>
75struct GrottoDcfCwsView {
76 using Control = In;
77
78 const int4 *cws;
79 In packed_tr;
80 size_t stride;
81
82 __host__ __device__ inline Cw operator[](int l) const {
83 return Cw{cws[static_cast<size_t>(l) * stride], ((packed_tr >> l) & 1) != 0};
84 }
85};

504template <int tree_depth, int frontier_depth, int threads_per_block>
505struct GrottoGpuLaunchShape {
506 static_assert(frontier_depth >= 0, "gpu frontier depth must be nonnegative");
507 static_assert(1 <= threads_per_block && threads_per_block <= 1024 &&
508 (threads_per_block & (threads_per_block - 1)) == 0,
509 "gpu block size must be a power of two in [1,1024]");
510 static constexpr int kFrontierDepth = frontier_depth < tree_depth ? frontier_depth : tree_depth;
511 static_assert(tree_depth - kFrontierDepth <= 8,
512 "gpu frontier leaves more than eight tail levels; specify a larger frontier_depth");
513 static constexpr int kThreadsPerBlockLog2 = [] {
514 int log2 = 0;
515 for (int count = threads_per_block; count > 1; count >>= 1) ++log2;
516 return log2 < kFrontierDepth ? log2 : kFrontierDepth;
517 }();
518 static constexpr int kThreadsPerBlock = 1 << kThreadsPerBlockLog2;
519 static constexpr int kBlockRootDepth = kFrontierDepth - kThreadsPerBlockLog2;
520 static_assert(kBlockRootDepth <= 30, "gpu block index exceeds the signed-int range");
521 static constexpr size_t kBlocks = size_t{1} << (kBlockRootDepth <= 30 ? kBlockRootDepth : 0);
522};

530static __global__ void GrottoUnpackBitsKernel(const int4 *blocks, bool *leaves, size_t count) {
531 size_t i = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
532 if (i >= count) return;
533 int4 block = blocks[i / 128];
534 unsigned word = (i % 128) / 32;
535 unsigned value = word == 0 ? static_cast<unsigned>(block.x) :
536 word == 1 ? static_cast<unsigned>(block.y) : word == 2 ? static_cast<unsigned>(block.z) : static_cast<unsigned>(block.w);
537 leaves[i] = ((value >> (i % 32)) & 1u) != 0;
538}

540inline cudaError_t GrottoUnpackGrid(size_t count, int threads, int &grid) {
541 const size_t blocks = count / threads + (count % threads != 0);
542 if (blocks > static_cast<size_t>(std::numeric_limits<int>::max()))
543 return cudaErrorInvalidConfiguration;
544 grid = static_cast<int>(blocks);
545 return cudaSuccess;
546}

548// Split-layout leaf-bits kernel. The correction buffer
549// arrives type-erased and the boundary view reconstructs the per-layer Cw
550// entries, so the traversal stays identical on both paths.
551template <int input_bits, int frontier_depth, int block_root_depth, typename Dpf, typename Prg,
552 typename Cws>
553__device__ void GrottoSplitLeafBitsWork(bool party, int4 s0, const Cws &view, bool *leaves,
554 Prg prg) {
555 constexpr int kTailDepth = input_bits - frontier_depth;
556 constexpr int kBlockSubtreeDepth = frontier_depth - block_root_depth;
557 const int64_t thread_index = static_cast<int64_t>(blockIdx.x) * blockDim.x + threadIdx.x;
558 __shared__ int4 shared_frontier[1 << kBlockSubtreeDepth];
559 if (threadIdx.x == 0) {
560 int4 state = util::SetLsb(s0, party);
561#pragma unroll
562 for (int level = 0; level < block_root_depth; ++level) {
563 bool take_right = (blockIdx.x >> (block_root_depth - 1 - level)) & 1;
564 auto node_prg = prg;
565 const auto cw = view[level];
566 state = fss::detail::SchemeAccess::DpfPathStep<Dpf>(
567 node_prg, state, cw.s, cw.tr, take_right);
568 }
569 shared_frontier[0] = state;
570 }
571 __syncthreads();
572
573#pragma unroll
574 for (int level = 0; level < kBlockSubtreeDepth; ++level) {
575 const bool active = threadIdx.x < (1 << (level + 1));
576 int4 state = {0, 0, 0, 0};
577 if (active) state = shared_frontier[threadIdx.x >> 1];
578 __syncthreads();
579 if (active) {
580 int4 left, right;
581 {
582 auto node_prg = prg;
583 const auto cw = view[block_root_depth + level];
584 fss::detail::SchemeAccess::DpfExpandNode<Dpf>(node_prg, state, cw.s, cw.tr, left, right);
585 }
586 shared_frontier[threadIdx.x] = (threadIdx.x & 1) ? right : left;
587 }
588 __syncthreads();
589 }
590
591 int4 nodes[1 << kTailDepth];
592 nodes[0] = shared_frontier[threadIdx.x];
593 int count = 1;
594#pragma unroll
595 for (int level = frontier_depth; level < input_bits; ++level) {
596#pragma unroll
597 for (int i = count - 1; i >= 0; --i) {
598 int4 left, right;
599 {
600 auto node_prg = prg;
601 const auto cw = view[level];
602 fss::detail::SchemeAccess::DpfExpandNode<Dpf>(node_prg, nodes[i], cw.s, cw.tr, left, right);
603 }
604 nodes[2 * i] = left;
605 nodes[2 * i + 1] = right;
606 }
607 count *= 2;
608 }
609
610 const size_t output_offset = static_cast<size_t>(thread_index) << kTailDepth;
611#pragma unroll
612 for (int i = 0; i < count; ++i) leaves[output_offset + i] = util::GetLsb(nodes[i]);
613}
614template <int input_bits, int frontier_depth, int block_root_depth, typename Dpf, typename Prg,
615 typename Cws>
616__global__ void GrottoSplitLeafBitsKernel(bool party, int4 s0, const void *cws_buf,
617 typename Cws::Control packed_tr, size_t stride, bool *leaves, Prg prg) {
618 const auto *cws = static_cast<const int4 *>(cws_buf);
619 GrottoSplitLeafBitsWork<input_bits, frontier_depth, block_root_depth, Dpf, Prg, Cws>(
620 party, s0, Cws{cws, packed_tr, stride}, leaves, prg);
621}

635static __global__ void GrottoParityLevelKernel(bool *tree, size_t count) {
636 size_t i = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
637 if (i >= count) return;
638 size_t index = count - 1 + i;
639 tree[index] = tree[2 * index + 1] ^ tree[2 * index + 2];
640}
641
642template <int top_level>
643__global__ void GrottoParityTopKernel(bool *tree) {
644#pragma unroll
645 for (int level = top_level; level >= 0; --level) {
646 int count = 1 << level;
647 if (threadIdx.x < count) {
648 int index = count - 1 + threadIdx.x;
649 tree[index] = tree[2 * index + 1] ^ tree[2 * index + 2];
650 }
651 __syncthreads();
652 }
653}

655static __global__ void GrottoPackedParityKernel(const int4 *blocks, bool *leaves, size_t count) {
656 size_t i = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
657 if (i >= count) return;
658 int4 block = blocks[i];
659 unsigned value = static_cast<unsigned>(block.x) ^ static_cast<unsigned>(block.y) ^
660 static_cast<unsigned>(block.z) ^ static_cast<unsigned>(block.w);
661 leaves[i] = (__popc(value) & 1) != 0;
662}

664template <int leaf_bits, typename LaunchShape>
665inline cudaError_t BuildGrottoParityTree(bool *tree, cudaStream_t stream) {
666 constexpr int kThreads = LaunchShape::kThreadsPerBlock;
667 constexpr int kTop = LaunchShape::kThreadsPerBlockLog2 < leaf_bits - 1
668 ? LaunchShape::kThreadsPerBlockLog2 : leaf_bits - 1;
669 for (int level = leaf_bits - 1; level > kTop; --level) {
670 size_t count = size_t{1} << level;
671 size_t grid = count / kThreads + (count % kThreads != 0);
672 GrottoParityLevelKernel<<<grid, kThreads, 0, stream>>>(tree, count);
673 cudaError_t status = cudaGetLastError();
674 if (status != cudaSuccess) return status;
675 }
676 if constexpr (leaf_bits > 0) {
677 GrottoParityTopKernel<kTop><<<1, kThreads, 0, stream>>>(tree);
678 return cudaGetLastError();
679 }
680 return cudaSuccess;
681}

754template <int in_bits, typename Prg, typename In>
755 requires(ValidForInputBits<In, in_bits> && Prgable<Prg, 2>)
756__host__ __device__ inline bool GrottoDcf<in_bits, Prg, In>::Eval(ParityTreeView pt, In x) {
757 static_assert(detail::FitsBinaryTree<bool, in_bits>(),
758 "parity-tree count and byte size must fit in size_t");
759 constexpr size_t N = detail::FullDomainSize<bool, in_bits>();
760 In e = static_cast<In>(x) + 1;
761
762 // e == 0 means x + 1 overflowed, i.e., e = N (entire domain)
763 if (e == 0 || e == N) return pt.p[0];
764
765 bool pi = false;
766 size_t cur = 0;
767 for (int i = 0; i < in_bits; ++i) {
768 bool e_bit = (e >> (in_bits - 1 - i)) & 1;
769 if (e_bit) {
770 pi ^= pt.p[2 * cur + 1];
771 cur = 2 * cur + 2;
772 } else {
773 cur = 2 * cur + 1;
774 }
775 }
776 return pi;
777}

800 const size_t required = static_cast<size_t>(in_bits) * stride + 1;
801 if (cws.size() < required)
802 throw std::invalid_argument("GrottoDcf::EvalAll: cws must hold in_bits*stride+1 elements");
803 EvalAllCws(b, s0, CwsView{cws.data(), tr, stride}, ys, par_depth);
804}

806template <int in_bits, typename Prg, typename In>
807 requires(ValidForInputBits<In, in_bits> && Prgable<Prg, 2>)
808template <typename Cws>
809inline void GrottoDcf<in_bits, Prg, In>::EvalAllCws(
810 bool b, int4 s0, Cws cws, bool ys[], int par_depth) {
811 assert(par_depth >= 0);
812 constexpr size_t N = detail::FullDomainSize<bool, in_bits>();
813
814 // Phase 1: expand tree to get leaf control bits into ys[]
815 ExpandTree(b, s0, cws, ys, par_depth);
816
817 // Phase 2b: prefix-sum scan (running XOR)
818 // ys[x] currently holds leaf x's control bit.
819 // Transform to: ys[x] = XOR of control bits [0..x] = share of 1[alpha <= x].
820 for (size_t x = 1; x < N; ++x) {
821 ys[x] = ys[x] ^ ys[x - 1];
822 }
823}

825template <int in_bits, typename Prg, typename In>
826 requires(ValidForInputBits<In, in_bits> && Prgable<Prg, 2>)
827template <typename Cws>
828inline void GrottoDcf<in_bits, Prg, In>::ExpandTree(
829 bool b, int4 s0, Cws cws, bool t[], int par_depth) {
830 int4 st = s0;
831 st = util::SetLsb(st, b);
832
833 size_t l = 0;
834 constexpr size_t r = detail::FullDomainSize<bool, in_bits>();
835 int i = 0;
836
837 int par_depth_ = fss::detail::ResolveParDepth(par_depth, in_bits);
838
839 if (par_depth_ == 0) {
840 ExpandTreeRec(st, cws, t, l, r, i, 0);
841 return;
842 }
843 int threads = fss::detail::ResolveParThreads(par_depth_, r);
844#pragma omp parallel num_threads(threads)
845#pragma omp single
846 ExpandTreeRec(st, cws, t, l, r, i, par_depth_);
847}

849template <int in_bits, typename Prg, typename In>
850 requires(ValidForInputBits<In, in_bits> && Prgable<Prg, 2>)
851template <typename Cws>
852inline void GrottoDcf<in_bits, Prg, In>::ExpandTreeRec(
853 int4 st, Cws cws, bool t[], size_t l, size_t r, int i, int par_depth_) {
854 bool tc = util::GetLsb(st);
855 int4 s = st;
856 s = util::SetLsb(s, false);
857
858 if (i == in_bits) {
859 assert(l + 1 == r);
860 t[l] = tc;
861 return;
862 }
863
864 const Cw cw = cws[i];
865 int4 s_cw = cw.s;
866 bool tl_cw = util::GetLsb(s_cw);
867 s_cw = util::SetLsb(s_cw, false);
868 bool tr_cw = cw.tr;
869
870 auto [sl, sr] = prg_.Gen(s);
871
872 bool tl = util::GetLsb(sl);
873 sl = util::SetLsb(sl, false);
874 bool tr = util::GetLsb(sr);
875 sr = util::SetLsb(sr, false);
876
877 if (tc) {
878 sl = util::Xor(sl, s_cw);
879 sr = util::Xor(sr, s_cw);
880 tl = tl ^ tl_cw;
881 tr = tr ^ tr_cw;
882 }
883
884 int4 stl = sl;
885 stl = util::SetLsb(stl, tl);
886 int4 str = sr;
887 str = util::SetLsb(str, tr);
888
889 size_t mid = (l + r) / 2;
890
891 if (i < par_depth_) {
892#pragma omp task
893 ExpandTreeRec(stl, cws, t, l, mid, i + 1, par_depth_);
894#pragma omp task
895 ExpandTreeRec(str, cws, t, mid, r, i + 1, par_depth_);
896#pragma omp taskwait
897 } else {
898 ExpandTreeRec(stl, cws, t, l, mid, i + 1, par_depth_);
899 ExpandTreeRec(str, cws, t, mid, r, i + 1, par_depth_);
900 }
901}

1044// Point-evaluate in_bits-level Grotto DCF keys: one thread per key, one
1045// single-path walk with in_bits PRG calls and no preprocessing. ys[k] is the
1046// party's share of 1[alpha <= xs[k]]. cws holds dense level-major correction
1047// words, the layout the split form of Gen() writes: layer i of key k is
1048// cws[i * nkeys + k]. tr[k] carries key k's packed right-control bits, bit i
1049// for layer i.
1050template <int in_bits, int unroll, bool b, typename Prg, typename In, typename Query>
1051__global__ void GrottoDcfEvalPointKernel(bool *ys, const int4 *seeds, const int4 *cws, const In *tr,
1052 const Query *xs, int nkeys, Prg prg) {
1053 static_assert(in_bits <= sizeof(In) * 8, "In too narrow for in_bits");
1054 int tid = blockIdx.x * blockDim.x + threadIdx.x;
1055 if (tid >= nkeys) return;
1056 int4 s = fss::util::SetLsb(seeds[tid], false);
1057 bool t = b;
1058 In x = static_cast<In>(xs[tid]);
1059 In tr_k = tr[tid];
1060 fss::detail::GrottoTurns turns;
1061
1062#pragma unroll unroll
1063 for (int i = 0; i < in_bits; ++i) {
1064 bool x_bit = GetBit(x, in_bits - 1 - i);
1065 turns(t, x_bit);
1066
1067 int4 s_cw = cws[(size_t)i * nkeys + tid];
1068 bool tr_cw = GetBit(tr_k, i);
1069 bool tl_cw = fss::util::GetLsb(s_cw);
1070 s_cw = fss::util::SetLsb(s_cw, false);
1071
1072 auto g = prg.Gen(s);
1073 int4 sl = g[0], sr = g[1];
1074 bool tl = fss::util::GetLsb(sl);
1075 sl = fss::util::SetLsb(sl, false);
1076 bool tr = fss::util::GetLsb(sr);
1077 sr = fss::util::SetLsb(sr, false);
1078
1079 if (t) {
1080 sl = fss::util::Xor(sl, s_cw);
1081 sr = fss::util::Xor(sr, s_cw);
1082 tl = tl ^ tl_cw;
1083 tr = tr ^ tr_cw;
1084 }
1085
1086 if (x_bit) {
1087 s = sr;
1088 t = tr;
1089 } else {
1090 s = sl;
1091 t = tl;
1092 }
1093 }
1094
1095 ys[tid] = turns.PointLeaf(t);
1096}

1252// device copy exists. The old Eval() never read the party tag, so the query
1253// goes through ParityTreeView directly.
1254template <int in_bits, typename Prg, typename In>
1255 requires(ValidForInputBits<In, in_bits> && Prgable<Prg, 2>)
1256__host__ __device__ inline void GrottoDcf<in_bits, Prg, In>::Gen(
1257 compat::detail::CwsKeyRecord<GrottoDcf, Cw, in_bits + 1> &k0,
1258 compat::detail::CwsKeyRecord<GrottoDcf, Cw, in_bits + 1> &k1,
1259 cuda::std::span<const int4, 2> s0s, In a) {
1260 int4 seed[in_bits + 1];
1261 In tr;
1262 Gen(cuda::std::span<int4>(seed, in_bits + 1), tr, s0s, a);
1263 for (int i = 0; i < in_bits; ++i) {
1264 const Cw layer = {seed[i], ((tr >> i) & 1) != 0};
1265 k0.cws[i] = layer;
1266 k1.cws[i] = layer;
1267 }
1268 const Cw terminal = {seed[in_bits], false};
1269 k0.cws[in_bits] = terminal;
1270 k1.cws[in_bits] = terminal;
1271 k0.s0 = s0s[0];
1272 k1.s0 = s0s[1];
1273}

1276 requires(ValidForInputBits<In, in_bits> && Prgable<Prg, 2>)
1277inline compat::detail::PreprocessedTreeRecord<GrottoDcf<in_bits, Prg, In>>
1278GrottoDcf<in_bits, Prg, In>::Preprocess(
1279 bool b, const compat::detail::CwsKeyRecord<GrottoDcf, Cw, in_bits + 1> &key) {
1280 int4 seed[in_bits + 1];
1281 In tr = 0;
1282 for (int i = 0; i <= in_bits; ++i) seed[i] = key.cws[i].s;
1283 for (int i = 0; i < in_bits; ++i) tr |= static_cast<In>(key.cws[i].tr) << i;
1284
1285 compat::detail::PreprocessedTreeRecord<GrottoDcf> pt;
1286 constexpr size_t kTreeSize = (size_t{2} << in_bits) - 1;
1287 pt.p = std::make_unique<bool[]>(kTreeSize);
1288 pt.b = b;
1289 Preprocess(b, key.s0, cuda::std::span<const int4>(seed, in_bits + 1), tr, pt.p.get());
1290 return pt;
1291}
