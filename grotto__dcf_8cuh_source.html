51// State of the prefix-parity algorithm (Section 4 of the paper) on x's path. Called with the
52// control bit t of every expanded node and the branch x_bit taken from it, share accumulates
53// the control bits of the nodes where the path turns, taking the edge into the root as a left
54// edge. GrottoDcf and HalfTreeGrottoDcf share it and differ only in the leaf step.
55struct GrottoTurns {
56 bool share = false;
57 bool prev_bit = false;
58
59 __host__ __device__ void operator()(bool t, bool x_bit) {
60 if (x_bit != prev_bit) share ^= t;
61 prev_bit = x_bit;
62 }
63
64 // Leaf step when each leaf is a single input: XOR the leaf's control bit if the path ends
65 // with a left edge.
66 __host__ __device__ bool PointLeaf(bool t) const {
67 return share != (!prev_bit && t);
68 }
69};

194 // Caller-owned v1.2.0 form. pt.p contains 2 * 2^in_bits - 1 entries.
195 void Preprocess(ParityTree &pt, int4 s0, const Cw cws[]) {
196 constexpr size_t n = size_t{1} << in_bits;
197 ExpandTree(pt.b, s0, cws, pt.p + n - 1);
198 for (size_t j = n - 2; j < n - 1; --j) {
199 pt.p[j] = pt.p[2 * j + 1] ^ pt.p[2 * j + 2];
200 }
201 }

215 __host__ __device__ static bool Eval(const ParityTree &pt, In x) {
216 constexpr size_t N = 1ULL << in_bits;
217 In e = static_cast<In>(x) + 1;
218
219 // e == 0 means x + 1 overflowed, i.e., e = N (entire domain)
220 if (e == 0 || e == N) return pt.p[0];
221
222 bool pi = false;
223 size_t cur = 0;
224 for (int i = 0; i < in_bits; ++i) {
225 bool e_bit = (e >> (in_bits - 1 - i)) & 1;
226 if (e_bit) {
227 pi ^= pt.p[2 * cur + 1];
228 cur = 2 * cur + 2;
229 } else {
230 cur = 2 * cur + 1;
231 }
232 }
233 return pi;
234 }

265 void EvalAll(bool b, int4 s0, const Cw cws[], bool ys[]) {
266 constexpr size_t N = 1ULL << in_bits;
267
268 // Phase 1: expand tree to get leaf control bits into ys[]
269 ExpandTree(b, s0, cws, ys);
270
271 // Phase 2b: prefix-sum scan (running XOR)
272 // ys[x] currently holds leaf x's control bit.
273 // Transform to: ys[x] = XOR of control bits [0..x] = share of 1[alpha <= x].
274 for (size_t x = 1; x < N; ++x) {
275 ys[x] = ys[x] ^ ys[x - 1];
276 }
277 }

280 // Expand the DPF tree and write leaf control bits.
281 void ExpandTree(bool b, int4 s0, const Cw cws[], bool t[]) {
282 int4 st = s0;
283 st = util::SetLsb(st, b);
284
285 assert(in_bits < sizeof(size_t) * 8);
286 size_t l = 0;
287 size_t r = 1ULL << in_bits;
288 int i = 0;
289
290 int par_depth_ = util::ResolveParDepth(par_depth, in_bits);
291
292 if (par_depth_ == 0) {
293 ExpandTreeRec(st, cws, t, l, r, i, 0);
294 return;
295 }
296 int threads = util::ResolveParThreads(par_depth_, r);
297#pragma omp parallel num_threads(threads)
298#pragma omp single
299 ExpandTreeRec(st, cws, t, l, r, i, par_depth_);
300 }
301
302 void ExpandTreeRec(int4 st, const Cw cws[], bool t[], size_t l, size_t r, int i, int par_depth_) {
303 bool tc = util::GetLsb(st);
304 int4 s = st;
305 s = util::SetLsb(s, false);
306
307 if (i == in_bits) {
308 assert(l + 1 == r);
309 t[l] = tc;
310 return;
311 }
312
313 Cw cw = cws[i];
314 int4 s_cw = cw.s;
315 bool tl_cw = util::GetLsb(s_cw);
316 s_cw = util::SetLsb(s_cw, false);
317 bool tr_cw = cw.tr;
318
319 auto [sl, sr] = prg.Gen(s);
320
321 bool tl = util::GetLsb(sl);
322 sl = util::SetLsb(sl, false);
323 bool tr = util::GetLsb(sr);
324 sr = util::SetLsb(sr, false);
325
326 if (tc) {
327 sl = util::Xor(sl, s_cw);
328 sr = util::Xor(sr, s_cw);
329 tl = tl ^ tl_cw;
330 tr = tr ^ tr_cw;
331 }
332
333 int4 stl = sl;
334 stl = util::SetLsb(stl, tl);
335 int4 str = sr;
336 str = util::SetLsb(str, tr);
337
338 size_t mid = (l + r) / 2;
339
340 if (i < par_depth_) {
341#pragma omp task
342 ExpandTreeRec(stl, cws, t, l, mid, i + 1, par_depth_);
343#pragma omp task
344 ExpandTreeRec(str, cws, t, mid, r, i + 1, par_depth_);
345#pragma omp taskwait
346 } else {
347 ExpandTreeRec(stl, cws, t, l, mid, i + 1, par_depth_);
348 ExpandTreeRec(str, cws, t, mid, r, i + 1, par_depth_);
349 }
350 }
