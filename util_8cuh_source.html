43// Resolve a tree split depth without shifting beyond a machine word.
44inline int ResolveParDepth(int par_depth, int useful_depth = sizeof(size_t) * CHAR_BIT - 1) {
45 if (par_depth < 0) {
46 par_depth = 0;
47 unsigned threads = static_cast<unsigned>(omp_get_max_threads());
48 size_t partitions = 1;
49 while (partitions < threads && par_depth < useful_depth) {
50 partitions *= 2;
51 ++par_depth;
52 }
53 }
54 return par_depth < useful_depth ? par_depth : useful_depth;
55}

57inline size_t ResolveParPartitions(int par_depth, size_t count) {
58 if (count == 0) return 0;
59 int depth = ResolveParDepth(par_depth);
60 size_t partitions = 1;
61 while (depth-- > 0 && partitions < count) {
62 if (partitions > count / 2) return count;
63 partitions *= 2;
64 }
65 return partitions;
66}

68inline int ResolveParThreads(int par_depth, size_t count) {
69 size_t partitions = ResolveParPartitions(par_depth, count);
70 int threads = omp_get_max_threads();
71 return partitions < static_cast<size_t>(threads) ? static_cast<int>(partitions) : threads;
72}

74// Each scheduled partition owns a contiguous range. A depth of zero never
75// enters an OpenMP region, including when called from a shallow fallback path.
76template <typename Function>
77inline void ParallelFor(int par_depth, size_t count, Function function) {
78 size_t partitions = ResolveParPartitions(par_depth, count);
79 if (partitions <= 1) {
80 for (size_t i = 0; i < count; ++i) function(i);
81 return;
82 }
83 int threads = ResolveParThreads(par_depth, count);
84#pragma omp parallel for num_threads(threads)
85 for (size_t partition = 0; partition < partitions; ++partition) {
86 size_t begin = partition * (count / partitions) + (partition < count % partitions ? partition : count % partitions);
87 size_t end = begin + count / partitions + (partition < count % partitions);
88 for (size_t i = begin; i < end; ++i) function(i);
89 }
90}

92// Stored seed words are signed int32 values. Increment their bit pattern modulo
93// 2^32, then convert back without an out-of-range signed conversion.
94__host__ __device__ inline int IncrementWord(int value) {
95 uint32_t bits = static_cast<uint32_t>(value) + uint32_t{1};
96 return bits <= static_cast<uint32_t>(INT_MAX) ? static_cast<int>(bits)
97 : -1 - static_cast<int>(UINT32_MAX - bits);
98}

100template <typename In>
101__host__ __device__ inline int4 Pack(In val) {
102 int4 buf = {0, 0, 0, 0};
103 if constexpr (sizeof(In) <= 4) {
104 buf.x = static_cast<int>(val);
105 } else if constexpr (sizeof(In) <= 8) {
106 auto v = static_cast<uint64_t>(val);
107 buf.x = static_cast<int>(v & 0xFFFFFFFF);
108 buf.y = static_cast<int>((v >> 32) & 0xFFFFFFFF);
109 } else {
110 auto v = static_cast<__uint128_t>(val);
111 buf.x = static_cast<int>(v & 0xFFFFFFFF);
112 buf.y = static_cast<int>((v >> 32) & 0xFFFFFFFF);
113 buf.z = static_cast<int>((v >> 64) & 0xFFFFFFFF);
114 buf.w = static_cast<int>((v >> 96) & 0xFFFFFFFF);
115 }
116 return buf;
117}
