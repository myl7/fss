50 __host__ __device__ Uint128 operator+(Uint128 rhs) const {
51 if (val_ >= kMod - rhs.val_) return {val_ + rhs.val_ - kMod};
52 else return {val_ + rhs.val_};
53 }
54
56 __host__ __device__ Uint128 operator-() const {
57 if (val_ == 0) return {0};
58 else return {kMod - val_};
59 }

65 __host__ __device__ static Uint128 From(int4 buf) {
66 assert((buf.w & 1) == 0);
67
68 __uint128_t value = static_cast<__uint128_t>(static_cast<unsigned int>(buf.x)) |
69 static_cast<__uint128_t>(static_cast<unsigned int>(buf.y)) << 32 |
70 static_cast<__uint128_t>(static_cast<unsigned int>(buf.z)) << 64 |
71 // The LSB of buf.w is the clamped bit
72 static_cast<__uint128_t>(static_cast<unsigned int>(buf.w) >> 1) << 96;
73
74 return {value % kMod};
75 }

82 buf.z = static_cast<int>((val_ >> 64) & 0xffffffff);
83 // The LSB of buf.w is the clamped bit
84 buf.w = static_cast<int>((val_ >> 96) << 1);
