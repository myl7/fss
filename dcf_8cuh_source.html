157private:
158 // EvalAll() shape (mirroring Dpf): a scalar prefix of kBatchPrefix levels grows the
159 // frontier to 8 nodes, each carrying the seed (t in LSB) beside the running group
160 // value, detail::GenBatch() expands 8 parents per level (pipelined through Gen8()
161 // when the PRG offers a batched entry), and a kBatchTail-level tail converts each
162 // 8-subtree group of 8 << kBatchTail leaves in local memory. Domains below
163 // kBatchPrefix + kBatchTail levels keep the recursive EvalTree() walk.
164 static constexpr int kBatchPrefix = 3;
165 static constexpr int kBatchTail = 3;
166
167 // The v-channel correction has branchless masked fast paths for the XOR group and
168 // the clamped 127-bit integer group; every other group runs the generic operators.
169 static constexpr bool kGroupIsBytes = std::is_same_v<Group, group::Bytes>;
170 static constexpr bool kGroupIsUint127 =
171 std::is_same_v<Group, group::Uint<__uint128_t, (static_cast<__uint128_t>(1) << 127)>>;

173public:
174 __host__ __device__ void Gen(Cw cws[], const int4 s0s[2], In a, int4 b_buf) {
175 int4 s0 = s0s[0];
176 s0 = util::SetLsb(s0, false);
177 int4 s1 = s0s[1];
178 s1 = util::SetLsb(s1, false);
179 bool t0 = false;
180 bool t1 = true;
181 Group v;
182 b_buf = util::SetLsb(b_buf, false);
183
184 for (int i = 0; i < in_bits; ++i) {
185 auto [s0l, v0l_buf, s0r, v0r_buf] = prg.Gen(s0);
186 auto [s1l, v1l_buf, s1r, v1r_buf] = prg.Gen(s1);
187
188 bool t0l = util::GetLsb(s0l);
189 s0l = util::SetLsb(s0l, false);
190 v0l_buf = util::SetLsb(v0l_buf, false);
191 auto v0l = Group::From(v0l_buf);
192 bool t0r = util::GetLsb(s0r);
193 s0r = util::SetLsb(s0r, false);
194 v0r_buf = util::SetLsb(v0r_buf, false);
195 auto v0r = Group::From(v0r_buf);
196 bool t1l = util::GetLsb(s1l);
197 s1l = util::SetLsb(s1l, false);
198 v1l_buf = util::SetLsb(v1l_buf, false);
199 auto v1l = Group::From(v1l_buf);
200 bool t1r = util::GetLsb(s1r);
201 s1r = util::SetLsb(s1r, false);
202 v1r_buf = util::SetLsb(v1r_buf, false);
203 auto v1r = Group::From(v1r_buf);
204
205 bool a_bit = (a >> (in_bits - 1 - i)) & 1;
206
207 int4 s_cw;
208 if (!a_bit) s_cw = util::Xor(s0r, s1r);
209 else s_cw = util::Xor(s0l, s1l);
210
211 Group v_cw = (-v);
212 if (!a_bit) {
213 v_cw = v_cw + v1r + (-v0r);
214 if constexpr (pred == DcfPred::kGt) v_cw = v_cw + Group::From(b_buf);
215 } else {
216 v_cw = v_cw + v1l + (-v0l);
217 if constexpr (pred == DcfPred::kLt) v_cw = v_cw + Group::From(b_buf);
218 }
219 if (t1) v_cw = -v_cw;
220
221 if (!a_bit) v = v + (-v1l) + v0l;
222 else v = v + (-v1r) + v0r;
223 if (t1) v = v + (-v_cw);
224 else v = v + v_cw;
225
226 bool tl_cw = t0l ^ t1l ^ a_bit ^ 1;
227 bool tr_cw = t0r ^ t1r ^ a_bit;
228
229 if (!a_bit) {
230 s0 = s0l;
231 if (t0) s0 = util::Xor(s0, s_cw);
232 s1 = s1l;
233 if (t1) s1 = util::Xor(s1, s_cw);
234
235 if (t0) t0 = t0l ^ tl_cw;
236 else t0 = t0l;
237 if (t1) t1 = t1l ^ tl_cw;
238 else t1 = t1l;
239 } else {
240 s0 = s0r;
241 if (t0) s0 = util::Xor(s0, s_cw);
242 s1 = s1r;
243 if (t1) s1 = util::Xor(s1, s_cw);
244
245 if (t0) t0 = t0r ^ tr_cw;
246 else t0 = t0r;
247 if (t1) t1 = t1r ^ tr_cw;
248 else t1 = t1r;
249 }
250
251 s_cw = util::SetLsb(s_cw, tl_cw);
252 int4 v_buf = v_cw.Into();
253 v_buf = util::SetLsb(v_buf, tr_cw);
254 cws[i] = {s_cw, v_buf};
255 }
256
257 auto v_cw_np1 = Group::From(s1) + (-Group::From(s0)) + (-v);
258 if (t1) v_cw_np1 = -v_cw_np1;
259 cws[in_bits] = {{0, 0, 0, 0}, v_cw_np1.Into()};
260 }

262 // Seed + correction-words form used by the internal recursion and tests.
263 __host__ __device__ int4 Eval(bool b, int4 s0, const Cw cws[], In x) {
264 int4 s = s0;
265 s = util::SetLsb(s, false);
266 Group v;
267 bool t = b;
268
269 for (int i = 0; i < in_bits; ++i) {
270 auto cw = cws[i];
271
272 int4 s_cw = cw.s;
273 bool tl_cw = util::GetLsb(s_cw);
274 s_cw = util::SetLsb(s_cw, false);
275
276 int4 v_cw_buf = cw.v;
277 bool tr_cw = util::GetLsb(v_cw_buf);
278 v_cw_buf = util::SetLsb(v_cw_buf, false);
279 auto v_cw = Group::From(v_cw_buf);
280
281 auto [sl, vl_buf, sr, vr_buf] = prg.Gen(s);
282
283 bool tl = util::GetLsb(sl);
284 sl = util::SetLsb(sl, false);
285 vl_buf = util::SetLsb(vl_buf, false);
286 auto vl = Group::From(vl_buf);
287
288 bool tr = util::GetLsb(sr);
289 sr = util::SetLsb(sr, false);
290 vr_buf = util::SetLsb(vr_buf, false);
291 auto vr = Group::From(vr_buf);
292
293 if (t) {
294 sl = util::Xor(sl, s_cw);
295 sr = util::Xor(sr, s_cw);
296 tl = tl ^ tl_cw;
297 tr = tr ^ tr_cw;
298 }
299
300 bool x_bit = (x >> (in_bits - 1 - i)) & 1;
301
302 if (b) {
303 if (!x_bit) v = v + (-vl);
304 else v = v + (-vr);
305 if (t) v = v + (-v_cw);
306 } else {
307 if (!x_bit) v = v + vl;
308 else v = v + vr;
309 if (t) v = v + v_cw;
310 }
311
312 if (!x_bit) {
313 s = sl;
314 t = tl;
315 } else {
316 s = sr;
317 t = tr;
318 }
319 }
320
321 int4 v_cw_np1_buf = cws[in_bits].v;
322 assert((v_cw_np1_buf.w & 1) == 0);
323 auto v_cw_np1 = Group::From(v_cw_np1_buf);
324
325 if (b) {
326 v = v + (-Group::From(s));
327 if (t) v = v + (-v_cw_np1);
328 } else {
329 v = v + Group::From(s);
330 if (t) v = v + v_cw_np1;
331 }
332
333 return v.Into();
334 }

336 void EvalAll(bool b, int4 s0, const Cw cws[], int4 ys[]) {
337 assert(in_bits < sizeof(size_t) * 8);
338
339 int par_depth_ = util::ResolveParDepth(par_depth, in_bits);
340
341 int4 st = s0;
342 st = util::SetLsb(st, b);
343 Group v;
344
345 if constexpr (in_bits < kBatchPrefix + kBatchTail) {
346 // Domains too small to fill an 8wide frontier keep the recursive tree walk.
347 if (par_depth_ == 0) {
348 EvalTree(b, st, cws, ys, 0, size_t{1} << in_bits, 0, 0, v);
349 } else {
350 int threads = util::ResolveParThreads(par_depth_, size_t{1} << in_bits);
351#pragma omp parallel num_threads(threads)
352#pragma omp single
353 EvalTree(b, st, cws, ys, 0, size_t{1} << in_bits, 0, par_depth_, v);
354 }
355 return;
356 }
357
358 int split_depth = par_depth_;
359 if (split_depth > in_bits - kBatchPrefix - kBatchTail) split_depth = in_bits - kBatchPrefix - kBatchTail;
360 if (split_depth > 0) {
361 // Split at a frontier of 2 ** split_depth subtree roots instead of per-node tasks,
362 // so the sequential batched walk stays the fast default path.
363 size_t num_tasks = size_t{1} << split_depth;
364 std::vector<int4> st_front(num_tasks);
365 std::vector<Group> v_front(num_tasks);
366 st_front[0] = st;
367 v_front[0] = v;
368 for (int lv = 0; lv < split_depth; ++lv) ExpandFrontier(b, st_front.data(), v_front.data(), size_t{1} << lv, cws, lv);
369 size_t span = (size_t{1} << in_bits) / num_tasks;
370 util::ParallelFor(par_depth_, num_tasks, [&](size_t task) {
371 EvalBatched8(b, st_front[task], v_front[task], cws, ys, task * span, split_depth);
372 });
373 } else {
374 EvalBatched8(b, st, v, cws, ys, 0, 0);
375 }
376 }

378private:
379 void EvalTree(bool b, int4 st, const Cw cws[], int4 ys[], size_t l, size_t r, int i, int par_depth_, Group v) {
380 bool t = util::GetLsb(st);
381 int4 s = st;
382 s = util::SetLsb(s, false);
383
384 if (i == in_bits) {
385 int4 v_cw_np1_buf = cws[in_bits].v;
386 assert((v_cw_np1_buf.w & 1) == 0);
387 auto term = Group::From(s);
388 if (t) term = term + Group::From(v_cw_np1_buf);
389 if (b) term = -term;
390 v = v + term;
391 assert(l + 1 == r);
392 ys[l] = v.Into();
393 return;
394 }
395
396 Cw cw = cws[i];
397 int4 s_cw = cw.s;
398 bool tl_cw = util::GetLsb(s_cw);
399 s_cw = util::SetLsb(s_cw, false);
400 int4 v_cw_buf = cw.v;
401 bool tr_cw = util::GetLsb(v_cw_buf);
402 v_cw_buf = util::SetLsb(v_cw_buf, false);
403 auto v_cw = Group::From(v_cw_buf);
404
405 auto [sl, vl_buf, sr, vr_buf] = prg.Gen(s);
406
407 bool tl = util::GetLsb(sl);
408 sl = util::SetLsb(sl, false);
409 vl_buf = util::SetLsb(vl_buf, false);
410 auto vl = Group::From(vl_buf);
411
412 bool tr = util::GetLsb(sr);
413 sr = util::SetLsb(sr, false);
414 vr_buf = util::SetLsb(vr_buf, false);
415 auto vr = Group::From(vr_buf);
416
417 if (t) {
418 sl = util::Xor(sl, s_cw);
419 sr = util::Xor(sr, s_cw);
420 tl = tl ^ tl_cw;
421 tr = tr ^ tr_cw;
422 vl = vl + v_cw;
423 vr = vr + v_cw;
424 }
425 if (b) {
426 vl = -vl;
427 vr = -vr;
428 }
429
430 vl = vl + v;
431 vr = vr + v;
432
433 int4 stl = sl;
434 stl = util::SetLsb(stl, tl);
435 int4 str = sr;
436 str = util::SetLsb(str, tr);
437
438 size_t mid = (l + r) / 2;
439
440 if (i < par_depth_) {
441#pragma omp task
442 EvalTree(b, stl, cws, ys, l, mid, i + 1, par_depth_, vl);
443#pragma omp task
444 EvalTree(b, str, cws, ys, mid, r, i + 1, par_depth_, vr);
445#pragma omp taskwait
446 } else {
447 EvalTree(b, stl, cws, ys, l, mid, i + 1, par_depth_, vl);
448 EvalTree(b, str, cws, ys, mid, r, i + 1, par_depth_, vr);
449 }
450 }

452 // Expands w scalar frontier nodes (t in LSB, v the running group value) by one level
453 // with cws[lv]. The children overwrite the frontier in place: iteration is in reverse
454 // so the writes at 2j, 2j+1 never clobber nodes still to be read.
455 void ExpandFrontier(bool b, int4 st_front[], Group v_front[], size_t w, const Cw cws[], int lv) {
456 int4 s_cw = util::SetLsb(cws[lv].s, false);
457 bool tl_cw = util::GetLsb(cws[lv].s);
458 bool tr_cw = util::GetLsb(cws[lv].v);
459 auto v_cw = Group::From(util::SetLsb(cws[lv].v, false));
460
461 for (size_t j = w; j-- > 0;) {
462 bool t = util::GetLsb(st_front[j]);
463 int4 s = util::SetLsb(st_front[j], false);
464 // The parent value must be read before the children overwrite the frontier:
465 // for j = 0 the left child lands exactly on the parent's slot.
466 Group v_parent = v_front[j];
467
468 auto [sl, vl_buf, sr, vr_buf] = prg.Gen(s);
469
470 int m = -static_cast<int>(t);
471 st_front[2 * j] = CorrectChild(sl, s_cw, m, m & static_cast<int>(tl_cw));
472 st_front[2 * j + 1] = CorrectChild(sr, s_cw, m, m & static_cast<int>(tr_cw));
473 v_front[2 * j] = CorrectVal(vl_buf, v_cw, t, b, v_parent);
474 v_front[2 * j + 1] = CorrectVal(vr_buf, v_cw, t, b, v_parent);
475 }
476 }

478 // Applies the level correction to one expanded child branchlessly.
479 // m is 0 or -1 from the parent control bit t, and t_cw is the cw control bit
480 // masked by t. The cw seed is clamped, so the XOR leaves the child LSB free
481 // for the packed control bit.
482 static int4 CorrectChild(int4 child, int4 s_cw, int m, int t_cw) {
483 int4 res = {child.x ^ (s_cw.x & m), child.y ^ (s_cw.y & m), child.z ^ (s_cw.z & m), child.w ^ (s_cw.w & m)};
484 res.w = (res.w & ~1) | ((child.w ^ t_cw) & 1);
485 return res;
486 }
487
488 // Applies the v-channel correction to one child value: the PRG word becomes the child
489 // increment (v_cw added when the parent control bit t is set, negated for party 1),
490 // which then adds into the parent's running value v. Bit-equal with the operator
491 // sequence in EvalTree().
492 static Group CorrectVal(int4 base_buf, Group v_cw, bool t, bool b, Group v) {
493 if constexpr (kGroupIsBytes) {
494 // The XOR group: negation is the identity, so the correction is three XORs with
495 // the cw masked by the parent control bit.
496 int m = -static_cast<int>(t);
497 int4 base = util::SetLsb(base_buf, false);
498 int4 res;
499 res.x = v.val.x ^ base.x ^ (v_cw.val.x & m);
500 res.y = v.val.y ^ base.y ^ (v_cw.val.y & m);
501 res.z = v.val.z ^ base.z ^ (v_cw.val.z & m);
502 res.w = v.val.w ^ base.w ^ (v_cw.val.w & m);
503 Group out;
504 out.val = res;
505 return out;
506 } else if constexpr (kGroupIsUint127) {
507 // The clamped 127-bit integer group: mod-2^127 addition and negation in branchless
508 // masked forms. Operands are < 2^127, so a uint128 sum loses at most the carry-out
509 // at bit 127 and masking it off is the exact reduction; the party-1 subtraction
510 // wraps the borrow the same way. Bit-equal with the generic operator sequence.
511 constexpr __uint128_t kMask127 = (static_cast<__uint128_t>(1) << 127) - 1;
512 __uint128_t m = t ? ~static_cast<__uint128_t>(0) : 0;
513 __uint128_t x = (Group::From(util::SetLsb(base_buf, false)).val + (v_cw.val & m)) & kMask127;
514 __uint128_t res;
515 if (b) res = (v.val - x) & kMask127;
516 else res = (v.val + x) & kMask127;
517 Group out;
518 out.val = res;
519 return out;
520 } else {
521 auto base = Group::From(util::SetLsb(base_buf, false));
522 if (t) base = base + v_cw;
523 if (b) base = -base;
524 return v + base;
525 }
526 }

528 // Full-domain evaluation of the subtree rooted at (st, v) (t in LSB) at level i through
529 // an 8wide batched frontier: a kBatchPrefix-level scalar prefix, batched expansions, and
530 // a kBatchTail-level tail carrying (st, v) pairs. Requires
531 // in_bits - i >= kBatchPrefix + kBatchTail.
532 void EvalBatched8(bool b, int4 st, Group v, const Cw cws[], int4 ys[], size_t l, int i) {
533 int4 st_front[8];
534 Group v_front[8];
535 st_front[0] = st;
536 v_front[0] = v;
537 for (int lv = i; lv < i + kBatchPrefix; ++lv) ExpandFrontier(b, st_front, v_front, size_t{1} << (lv - i), cws, lv);
538 EvalBatched8Group(b, st_front, v_front, cws, ys, l, i + kBatchPrefix);
539 }

541 // Evaluates a group of 8 consecutive frontier nodes (t in LSB, v the running group
542 // value) at level i, covering the leaves [l, l + 8 << (in_bits - i)) in order.
543 // Requires in_bits - i >= kBatchTail.
544 void EvalBatched8Group(
545 bool b, const int4 (&st_front)[8], const Group (&v_front)[8], const Cw cws[], int4 ys[], size_t l, int i) {
546 if (i == in_bits - kBatchTail) {
547 // Expand the group to its 8 << kBatchTail leaves in local memory, then convert.
548 int4 st_bufs[2][8 << kBatchTail];
549 Group v_bufs[2][8 << kBatchTail];
550 int4 *st_cur = st_bufs[0];
551 int4 *st_next = st_bufs[1];
552 Group *v_cur = v_bufs[0];
553 Group *v_next = v_bufs[1];
554 for (int j = 0; j < 8; ++j) {
555 st_cur[j] = st_front[j];
556 v_cur[j] = v_front[j];
557 }
558 int w = 8;
559 for (int lv = i; lv < in_bits; ++lv) {
560 int4 s_cw = util::SetLsb(cws[lv].s, false);
561 int tl_cw = util::GetLsb(cws[lv].s);
562 int tr_cw = util::GetLsb(cws[lv].v);
563 auto v_cw = Group::From(util::SetLsb(cws[lv].v, false));
564 for (int g = 0; g < w / 8; ++g) {
565 cuda::std::array<int4, 8> seeds;
566 int ts[8];
567 for (int j = 0; j < 8; ++j) {
568 ts[j] = static_cast<int>(util::GetLsb(st_cur[8 * g + j]));
569 seeds[j] = util::SetLsb(st_cur[8 * g + j], false);
570 }
571 auto out = detail::GenBatch<4>(prg, seeds);
572 for (int j = 0; j < 8; ++j) {
573 int m = -ts[j];
574 st_next[16 * g + 2 * j] = CorrectChild(out[j][0], s_cw, m, ts[j] & tl_cw);
575 st_next[16 * g + 2 * j + 1] = CorrectChild(out[j][2], s_cw, m, ts[j] & tr_cw);
576 v_next[16 * g + 2 * j] = CorrectVal(out[j][1], v_cw, ts[j] != 0, b, v_cur[8 * g + j]);
577 v_next[16 * g + 2 * j + 1] = CorrectVal(out[j][3], v_cw, ts[j] != 0, b, v_cur[8 * g + j]);
578 }
579 }
580 int4 *st_swap = st_cur;
581 st_cur = st_next;
582 st_next = st_swap;
583 Group *v_swap = v_cur;
584 v_cur = v_next;
585 v_next = v_swap;
586 w *= 2;
587 }
588
589 int4 v_cw_np1_buf = cws[in_bits].v;
590 assert((v_cw_np1_buf.w & 1) == 0);
591 auto v_cw_np1 = Group::From(v_cw_np1_buf);
592 for (int j = 0; j < w; ++j) {
593 bool t = util::GetLsb(st_cur[j]);
594 int4 s = util::SetLsb(st_cur[j], false);
595 auto y = CorrectVal(s, v_cw_np1, t, b, v_cur[j]);
596 ys[l + j] = y.Into();
597 }
598 return;
599 }
600
601 int4 s_cw = util::SetLsb(cws[i].s, false);
602 int tl_cw = util::GetLsb(cws[i].s);
603 int tr_cw = util::GetLsb(cws[i].v);
604 auto v_cw = Group::From(util::SetLsb(cws[i].v, false));
605
606 cuda::std::array<int4, 8> seeds;
607 int ts[8];
608 for (int j = 0; j < 8; ++j) {
609 ts[j] = static_cast<int>(util::GetLsb(st_front[j]));
610 seeds[j] = util::SetLsb(st_front[j], false);
611 }
612 auto out = detail::GenBatch<4>(prg, seeds);
613 // The 16 children interleave in leaf order: node j's children land at 2j, 2j+1 of
614 // the next level. De-interleave them into two consecutive groups of 8.
615 int4 st_low[8];
616 int4 st_high[8];
617 Group v_low[8];
618 Group v_high[8];
619 for (int j = 0; j < 4; ++j) {
620 int m = -ts[j];
